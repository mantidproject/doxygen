230 template <typename T> std::vector<std::shared_ptr<const T>> findDetectorsOfType() const {
231 std::queue<IComponent_const_sptr> compQueue; // Search queue
232 addInstrumentChildrenToQueue(compQueue);
233
234 std::vector<std::shared_ptr<const T>> detectors;
235
236 IComponent_const_sptr comp;
237
238 while (!compQueue.empty()) {
239 comp = compQueue.front();
240 compQueue.pop();
241
242 if (!validateComponentProperties(comp))
243 continue;
244
245 if (auto const detector = std::dynamic_pointer_cast<const T>(comp)) {
246 detectors.push_back(detector);
247 } else {
248 // If component is a ComponentAssembly, we add its children to the queue to check if they're type T
249 addAssemblyChildrenToQueue(compQueue, comp);
250 }
251 }
252 return detectors;
253 }
