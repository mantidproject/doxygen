310PyObject *stringNumpyArray(const Mantid::API::Column_const_sptr &column) {
311 // Find maximum string length in the column cells
312 size_t maxlen = 0;
313 for (size_t i = 0; i < column->size(); ++i) {
314 maxlen = std::max(maxlen, column->cell<std::string>(i).size());
315 }
316 // Create array descriptor
317 std::string dtype_str = "<U" + std::to_string(maxlen);
318 PyObject *dtype = Py_BuildValue("s", dtype_str.c_str());
319 PyArray_Descr *descr = nullptr;
320 if (!PyArray_DescrConverter(dtype, &descr)) {
321 Py_DECREF(dtype);
322 throw std::runtime_error("Failed to create Unicode dtype");
323 }
324 Py_DECREF(dtype);
325 npy_intp arrayDims[1] = {static_cast<npy_intp>(column->size())};
326 PyObject *nparray = PyArray_NewFromDescr(&PyArray_Type, descr, 1, arrayDims, nullptr, nullptr, 0, nullptr);
327 auto *dest = reinterpret_cast<PyArrayObject *>(nparray);
328 for (size_t i = 0; i < column->size(); ++i) {
329 // Create python string object and place it into array
330 PyObject *pystr = PyUnicode_FromString(column->cell<std::string>(i).c_str());
331 PyArray_SETITEM(dest, static_cast<char *>(PyArray_GETPTR1(dest, i)), pystr);
332 Py_DECREF(pystr);
333 }
334 return nparray;
335}

337size_t getStride(const Mantid::API::Column_const_sptr &column) {
338 const std::type_info &typeID = column->get_type_info();
339 if (typeID.hash_code() == typeid(Mantid::Kernel::V3D).hash_code()) {
340 return column->cell<Mantid::Kernel::V3D>(0).size();
341 }
342 if (typeID.hash_code() == typeid(std::vector<int>).hash_code()) {
343 return column->cell<std::vector<int>>(0).size();
344 }
345 if (typeID.hash_code() == typeid(std::vector<double>).hash_code()) {
346 return column->cell<std::vector<double>>(0).size();
347 }
348 return 1;
349}

358template <typename T, typename U> void copyTo2DArray(void *dest, const Mantid::API::Column_const_sptr &column) {
359 auto dest_p = static_cast<T *>(dest);
360 size_t stride = getStride(column);
361 for (size_t i = 0; i < column->size(); ++i) {
362 const auto src = static_cast<std::vector<T>>(column->cell<U>(i));
363 if (src.size() != stride) {
364 throw std::runtime_error(
365 "All rows in a vector column must have the same length to be converted into a numpy array");
366 }
367 std::copy(src.begin(), src.end(), dest_p + i * stride);
368 }
369 return;
370}

372const std::unordered_map<size_t, std::tuple<int, std::function<void(void *, Mantid::API::Column_const_sptr)>>>
373 array_handlers = {
374 {typeid(int).hash_code(), {NPY_INT, copyTo1DArray<int, int>}},
375 {typeid(Mantid::API::Boolean).hash_code(), {NPY_BOOL, copyTo1DArray<bool, Mantid::API::Boolean>}},
376 {typeid(int32_t).hash_code(), {NPY_INT32, copyTo1DArray<int32_t, int32_t>}},
377 {typeid(uint32_t).hash_code(), {NPY_UINT32, copyTo1DArray<uint32_t, uint32_t>}},
378 {typeid(int64_t).hash_code(), {NPY_INT64, copyTo1DArray<int64_t, int64_t>}},
379 {typeid(uint64_t).hash_code(), {NPY_UINT64, copyTo1DArray<uint64_t, uint64_t>}},
380 {typeid(double).hash_code(), {NPY_DOUBLE, copyTo1DArray<double, double>}},
381 {typeid(float).hash_code(), {NPY_FLOAT, copyTo1DArray<float, float>}},
382 {typeid(size_t).hash_code(),
383 {(sizeof(size_t) == 4) ? NPY_UINT32 : NPY_UINT64,
384 (sizeof(size_t) == 4) ? copyTo1DArray<uint32_t, uint32_t> : copyTo1DArray<uint64_t, uint64_t>}},
385 {typeid(std::vector<int>).hash_code(), {NPY_INT, copyTo2DArray<int, std::vector<int>>}},
386 {typeid(std::vector<double>).hash_code(), {NPY_DOUBLE, copyTo2DArray<double, std::vector<double>>}},
387 {typeid(Mantid::Kernel::V3D).hash_code(), {NPY_DOUBLE, copyTo2DArray<double, Mantid::Kernel::V3D>}}};

395PyObject *columnArray(const ITableWorkspace &self, const object &value) {
396 Mantid::API::Column_const_sptr column;
397 if (PyUnicode_Check(value.ptr())) {
398 column = self.getColumn(extract<std::string>(value)());
399 } else {
400 column = self.getColumn(extract<size_t>(value)());
401 }
402 const std::type_info &typeID = column->get_type_info();
403 if (typeID.hash_code() == typeid(std::string).hash_code()) {
404 return stringNumpyArray(column);
405 }
406
407 auto it = array_handlers.find(typeID.hash_code());
408 if (it == array_handlers.end()) {
409 throw std::invalid_argument(std::string("Column type not yet supported for array: ") + std::string(typeID.name()));
410 }
411 const size_t numRows = self.rowCount();
412 const size_t stride = getStride(column);
413 const bool isVectorLike = (stride != 1);
414 npy_intp arrayDims[2] = {static_cast<npy_intp>(numRows), static_cast<npy_intp>(isVectorLike ? stride : 0)};
415 const int nDims = isVectorLike ? 2 : 1;
416 auto &[npy_type, copy_func] = it->second;
417 PyObject *nparray = PyArray_NewFromDescr(&PyArray_Type, PyArray_DescrFromType(npy_type), nDims, arrayDims, nullptr,
418 nullptr, 0, nullptr);
419 if (!nparray) {
420 throw std::runtime_error("Failed to allocate NumPy array");
421 }
422 void *dest = PyArray_DATA(reinterpret_cast<PyArrayObject *>(nparray)); // HEAD of the contiguous numpy data array
423 copy_func(dest, std::move(column));
424 return nparray;
425}

433PyObject *row(ITableWorkspace &self, int row) {
434 if (row < 0)
435 throw std::invalid_argument("Cannot specify negative row number");
436 if (row >= static_cast<int>(self.rowCount()))
437 throw std::invalid_argument("Cannot specify row larger than number of rows");
438
439 auto numCols = static_cast<int>(self.columnCount());
440
441 PyObject *result = PyDict_New();
442
443 for (int columnIndex = 0; columnIndex < numCols; columnIndex++) {
444 Mantid::API::Column_const_sptr col = self.getColumn(columnIndex);
445 const std::type_info &typeID = col->get_type_info();
446
447 if (PyDict_SetItemString(result, col->name().c_str(), getValue(col, typeID, row)))
448 throw std::runtime_error("Error while building dict");
449 }
450
451 return result;
452}

481void addRowFromDict(ITableWorkspace &self, const dict &rowItems) {
482 // rowItems must contain an entry for every column
483 auto nitems = boost::python::len(rowItems);
484 if (nitems != static_cast<decltype(nitems)>(self.columnCount())) {
485 throw std::invalid_argument("Number of values given does not match the number of columns. "
486 "Expected: " +
487 std::to_string(self.columnCount()));
488 }
489
490 // Add a new row to populate with values
491 const auto rowIndex = static_cast<int>(self.rowCount());
492 self.appendRow();
493
494 // Declared in this scope so we can access them in catch block
495 Column_sptr col; // Column in table
496 object value; // Value from dictionary
497
498 try {
499 // Retrieve and set the value for each column
500 auto columns = self.getColumnNames();
501 for (auto &iter : columns) {
502 col = self.getColumn(iter);
503 value = rowItems[iter];
504 setValue(col, rowIndex, value);
505 }
506 } catch (error_already_set &) {
507 // One of the columns wasn't found in the dictionary
508 if (PyErr_ExceptionMatches(PyExc_KeyError)) {
509 std::ostringstream msg;
510 msg << "Missing key-value entry for column ";
511 msg << "<" << col->name() << ">";
512 PyErr_SetString(PyExc_KeyError, msg.str().c_str());
513 }
514
515 // Wrong type of data for one of the columns
516 if (PyErr_ExceptionMatches(PyExc_TypeError)) {
517 std::ostringstream msg;
518 msg << "Wrong datatype for column <" << col->name() << "> ";
519 msg << "(expected <" << col->type() << ">)";
520 PyErr_SetString(PyExc_TypeError, msg.str().c_str());
521 }
522
523 // Remove the new row since populating it has failed
524 self.removeRow(rowIndex);
525 throw;
526 }
527 self.modified();
528}

538void addRowFromSequence(ITableWorkspace &self, const object &rowItems) {

546

708 list columnTypes;

712 for (const auto &name : names) {

732 static void readMetaData(ITableWorkspace &ws, const dict &state) {
733 const auto &metaData = state["meta_data"];
734 const auto &columnNames = metaData["column_names"];
735 const auto &columnTypes = metaData["column_types"];
736
737 auto numColumns = len(columnNames);
738 for (decltype(numColumns) colIndex = 0; colIndex < numColumns; ++colIndex) {
739 const auto &key = columnNames[colIndex];
740 const auto &value = columnTypes[colIndex];
741 const auto &name = extract<std::string>(key);
742 const auto &type = extract<std::string>(value);
743 ws.addColumn(type, name);
744 }
