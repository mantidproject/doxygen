115 ElementType_sptr extractValue(const boost::any &value) const {
116// Despite the name and hash code being identical, operator== is returning false
117// in Release mode
118// with clang and libc++. The simplest workaround is to compare hash codes.
119#ifdef __clang__
120 if (value.type().hash_code() == dataitemTypeID().hash_code())
121#else
122 if (value.type() == dataitemTypeID())
123#endif
124 {
125 return extractFromDataItem(value);
126 } else {
127 return extractFromSharedPtr(value);
128 }
129 }
135 ElementType_sptr extractFromDataItem(const boost::any &value) const {
136 const DataItem_sptr data = boost::any_cast<DataItem_sptr>(value);
137 // First try and push it up to the type of the validator
138 ElementType_sptr typedValue = std::dynamic_pointer_cast<ElementType>(data);
139 if (!typedValue) {
140 throw std::invalid_argument("DataItem \"" + data->getName() + "\" is not of the expected type.");
141 }
142 return typedValue;
143 }
149 ElementType_sptr extractFromSharedPtr(const boost::any &value) const {
150 try {
151 return boost::any_cast<ElementType_sptr>(value);
152 } catch (boost::bad_any_cast &) {
153 throw std::invalid_argument("Value was not a shared_ptr type");
154 }
155 }
157 static const std::type_info &dataitemTypeID() { return typeid(std::shared_ptr<DataItem>); }
