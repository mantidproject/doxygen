215void ScriptEditor::saveAs() {
216 QString selectedFilter;
217 QString filter = "Scripts (*.py *.PY);;All Files (*)";
218 QString filename = QFileDialog::getSaveFileName(nullptr, "Save file...", "", filter, &selectedFilter);
219
220 if (filename.isEmpty()) {
221 throw SaveCancelledException();
222 }
223 if (QFileInfo(filename).suffix().isEmpty()) {
224 QString ext = selectedFilter.section('(', 1).section(' ', 0, 0);
225 ext.remove(0, 1);
226 if (ext != ")")
227 filename += ext;
228 }
229 saveScript(filename);

284void ScriptEditor::keyPressEvent(QKeyEvent *event) {
285 // The built-in shortcut Ctrl++ from QScintilla doesn't work for some reason
286 // Creating a new QShortcut makes Ctrl++ to zoom in on the IPython console
287 // stop working
288 // So here is where Ctrl++ is detected to zoom in
289 if (QApplication::keyboardModifiers() & Qt::ControlModifier &&
290 (event->key() == Qt::Key_Plus || event->key() == Qt::Key_Equal)) {
291 zoomIn();
292 emit textZoomedIn();
293 } else {
294 // Avoids a bug in QScintilla
295 forwardKeyPressToBase(event);
296 }
297
298 // There is a built in Ctrl+- shortcut for zooming out, but a signal is
299 // emitted here to tell the other editor tabs to also zoom out
300 if (QApplication::keyboardModifiers() & Qt::ControlModifier && (event->key() == Qt::Key_Minus)) {
301 emit textZoomedOut();
302 }
303}

410void ScriptEditor::showUpdatedProgressMarker() {
411 if (m_currentExecLine == m_last_currentExecLine && m_progress_marker_error == m_last_progress_marker_error) {
412 return;
413 }
414 m_last_currentExecLine = m_currentExecLine;
415 m_last_progress_marker_error = m_progress_marker_error;
416
417 if (m_last_progress_marker_error) {
418 setMarkerBackgroundColor(g_error_colour, m_progressArrowKey);
419 } else {
420 setMarkerBackgroundColor(g_success_colour, m_progressArrowKey);
421 }
422 markerDeleteAll();
423 // Check the lineno actually exists, -1 means delete
424 if (m_currentExecLine <= 0 || m_currentExecLine > this->lines())
425 return;
426
427 ensureLineVisible(m_last_currentExecLine);
428 markerAdd(m_currentExecLine - 1, m_progressArrowKey);
429 progressMade(m_last_currentExecLine);
430}

567void ScriptEditor::forwardKeyPressToBase(QKeyEvent *event) {
568 // Hack to get around a bug in QScitilla
569 // If you pressed ( after typing in a autocomplete command the calltip does
570 // not appear, you have to delete the ( and type it again
571 // This does that for you!
572 if (event->text() == "(") {
573 auto *backspEvent = new QKeyEvent(QEvent::KeyPress, Qt::Key_Backspace, Qt::NoModifier);
574 // QKeyEvent's copy constructor is protected in Qt6; rebuild from the original's fields
575 auto *bracketEvent = new QKeyEvent(event->type(), event->key(), event->modifiers(), event->text(),
576 event->isAutoRepeat(), static_cast<ushort>(event->count()));
577 QsciScintilla::keyPressEvent(bracketEvent);
578 QsciScintilla::keyPressEvent(backspEvent);
579
580 delete backspEvent;
581 delete bracketEvent;
582 }

584 QsciScintilla::keyPressEvent(event);

586// Only need to do this for Unix and for QScintilla version < 2.4.2. Moreover,
587// only Gnome but I don't think we can detect that
588#ifdef Q_OS_LINUX
589#if QSCINTILLA_VERSION < 0x020402
590 // If an autocomplete box has surfaced, correct the window flags.
591 // Unfortunately the only way to
592 // do this is to search through the child objects.
593 if (isListActive()) {
594 QObjectList children = this->children();
595 QListIterator<QObject *> itr(children);
596 // Search is performed in reverse order as we want the last one created
597 itr.toBack();
598 while (itr.hasPrevious()) {
599 QObject *child = itr.previous();
600 if (child->inherits("QListWidget")) {
601 QWidget *w = qobject_cast<QWidget *>(child);
602 w->setWindowFlags(Qt::ToolTip | Qt::WindowStaysOnTopHint);
603 w->show();
604 break;
605 }
606 }
607 }
608#endif
609#endif

612void ScriptEditor::replaceAll(const QString &searchString, const QString &replaceString, bool regex, bool caseSensitive,
613 bool matchWords, bool wrap, bool forward) {
614 int line(-1), index(-1), prevLine(-1), prevIndex(-1);
615
616 // Mark this as a set of actions that can be undone as one
617 this->beginUndoAction();
618 bool found = this->findFirst(searchString, regex, caseSensitive, matchWords, wrap, forward, 0, 0);
619 // If find first fails then there is nothing to replace
620 if (!found) {
621 QMessageBox::information(this, "Mantid - Find and Replace", "No matches found in current document.");
622 }
623
624 while (found) {
625 this->getCursorPosition(&prevLine, &prevIndex);
626 this->replace(replaceString);
627 found = this->findNext();
628 this->getCursorPosition(&line, &index);
629 // if the next match is on the previous line
630 // or if it is on the same line, but closer to the start
631 // it means we have wrapped around the text in the editor
632 if (line < prevLine || (line == prevLine && index <= prevIndex)) {
633 break;
634 }
635 }
636 this->endUndoAction();
637}
