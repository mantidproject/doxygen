81concept CovarianceMatrixProviderFor =
82 std::invocable<Provider, const InputArray &, const InputArray &> &&
83 std::convertible_to<std::invoke_result_t<Provider, const InputArray &, const InputArray &>, CovarianceMatrix>;

86public:
87 using DerType = Eigen::Matrix<double, N, 1>;
88 using InputArray = DerType;
89 using ADScalar = Eigen::AutoDiffScalar<DerType>;
90 using CovarianceMatrix = Eigen::Matrix<double, N, N>;
91};

93template <size_t IndependentVars, size_t DependentVars, typename... DependentFuncs> class CovarianceMatrixProvider {
94public:
95 static_assert(sizeof...(DependentFuncs) == DependentVars, "Number of dependent functions must match DependentVars");
96
97 static constexpr size_t TotalVars = IndependentVars + DependentVars;
98 using IndependentTypes = ErrorTypeHelper<IndependentVars>;
99 using Types = ErrorTypeHelper<TotalVars>;
100 using InputArray = typename Types::InputArray;
101 using CovarianceMatrix = typename Types::CovarianceMatrix;
102 using ADScalar = typename IndependentTypes::ADScalar;
103 using IndependentDerType = typename IndependentTypes::DerType;
104 using FunctionInput = std::array<ADScalar, TotalVars>;
105
106 explicit CovarianceMatrixProvider(DependentFuncs... dependentFuncs)
107 : m_dependentFuncs(std::move(dependentFuncs)...) {}
108
109 CovarianceMatrix operator()(const InputArray &inputValues, const InputArray &inputErrors) const {
110 CovarianceMatrix covariance = inputErrors.array().square().matrix().asDiagonal();
111 const auto functionInputs = makeFunctionInput(inputValues);
112 const auto derivatives = dependentDerivatives(functionInputs, std::make_index_sequence<DependentVars>{});
113
114 for (size_t dep = 0; dep < DependentVars; ++dep) {
115 const size_t depIndex = IndependentVars + dep;
116 for (size_t independent = 0; independent < IndependentVars; ++independent) {
117 const double covarianceWithIndependent =
118 derivatives[dep][independent] * inputErrors[independent] * inputErrors[independent];
119 covariance(independent, depIndex) = covarianceWithIndependent;
120 covariance(depIndex, independent) = covarianceWithIndependent;
121 }
122 }
123
124 if constexpr (DependentVars < 2)
125 return covariance;
126
127 // Preserve supplied dependent variances on the diagonal. Only dependent-dependent off-diagonal terms are inferred.
128 for (size_t depA = 0; depA < DependentVars; ++depA) {
129 const size_t depAIndex = IndependentVars + depA;
130 for (size_t depB = depA + 1; depB < DependentVars; ++depB) {
131 const size_t depBIndex = IndependentVars + depB;
132 double covarianceBetweenDependents = 0.0;
133 for (size_t independent = 0; independent < IndependentVars; ++independent) {
134 covarianceBetweenDependents += derivatives[depA][independent] * derivatives[depB][independent] *
135 inputErrors[independent] * inputErrors[independent];
136 }
137 covariance(depAIndex, depBIndex) = covarianceBetweenDependents;
138 covariance(depBIndex, depAIndex) = covarianceBetweenDependents;
139 }
140 }
141
142 return covariance;
143 }
144
145private:
146 FunctionInput makeFunctionInput(const InputArray &inputValues) const {
147 FunctionInput functionInputs;
148 for (size_t i = 0; i < IndependentVars; ++i) {
149 functionInputs[i] = ADScalar(inputValues[i], IndependentDerType::Unit(IndependentVars, i));
150 }
151 for (size_t i = IndependentVars; i < TotalVars; ++i) {
152 functionInputs[i] = ADScalar(inputValues[i], IndependentDerType::Zero());
153 }
154 return functionInputs;
155 }

157 template <typename Func>
158 ADScalar evaluateDependentFunction(Func const &func, const FunctionInput &functionInputs) const {
159 static_assert(std::invocable<Func, const FunctionInput &>,
160 "Dependent functions must accept the covariance provider input values");
161 return func(functionInputs);
162 }
163
164 template <size_t... Indices>
165 std::array<IndependentDerType, DependentVars> dependentDerivatives(const FunctionInput &functionInputs,
166 std::index_sequence<Indices...>) const {
167 return {evaluateDependentFunction(std::get<Indices>(m_dependentFuncs), functionInputs).derivatives()...};
168 }
169
170 std::tuple<DependentFuncs...> m_dependentFuncs;
171};

173// The first IndependentVars inputs are treated as independent. The following DependentVars inputs are treated as
174// quantities derived from the independent variables. If dependent variable d_a has derivatives J_ai = dd_a/dx_i, then
175// Cov(x_i, d_a) = J_ai Var(x_i) and Cov(d_a, d_b) = sum_i J_ai J_bi Var(x_i). Dependent functions must express each
176// dependency directly in terms of the independent variables; if one dependent quantity depends on another, inline that
177// dependency into the function. Use the input values' .value() when a fixed bin value is needed.
178template <size_t IndependentVars, size_t DependentVars, typename... DependentFuncs>
179auto makeCovarianceMatrixProvider(DependentFuncs &&...dependentFuncs) {
180 return CovarianceMatrixProvider<IndependentVars, DependentVars, std::decay_t<DependentFuncs>...>(
181 std::forward<DependentFuncs>(dependentFuncs)...);
182}

184template <size_t N, typename Func> class ErrorPropagation {
185public:
186 using Types = ErrorTypeHelper<N>;
187 using DerType = Types::DerType;
188 using ADScalar = Types::ADScalar;
189 using InputArray = Types::InputArray;
190 using CovarianceMatrix = Types::CovarianceMatrix;
191 ErrorPropagation(Func func) : computeFunc(std::move(func)) {}
192
193 struct AutoDevResult {
194 double value;
195 double error;
196 Eigen::Array<double, N, 1> derivatives;
197 };
198
199 AutoDevResult evaluate(const InputArray &values, const InputArray &errors) const {
200 return evaluateWithCovariance(values, covarianceMatrixFromErrors(errors));
201 }
202
203 AutoDevResult evaluateWithCovariance(const InputArray &values, const CovarianceMatrix &covariance) const {
204 std::array<ADScalar, N> x;
205 for (size_t i = 0; i < N; ++i) {
206 x[i] = ADScalar(values[i], DerType::Unit(N, i));
207 }
208 const ADScalar y = computeFunc(x);
209 const auto &derivatives = y.derivatives();
210 // First-order Taylor propagation with correlated inputs:
211 // Var(f) = J C J^T, where J is the row vector of derivatives df/dx_i and C is the covariance matrix.
212 const double variance = derivatives.dot(covariance * derivatives);
213 return {y.value(), std::sqrt(std::max(variance, 0.0)), derivatives};
214 }

240 static CovarianceMatrix covarianceMatrixFromErrors(const InputArray &errors) {
241 // Independent inputs have no off-diagonal covariance terms, so their covariance matrix is diagonal with
242 // C_ii = Var(x_i) = sigma_i^2.
243 return errors.array().square().matrix().asDiagonal();
244 }

253 template <typename Provider, std::same_as<API::MatrixWorkspace_sptr>... Ts>
254 requires CovarianceMatrixProviderFor<Provider, InputArray, CovarianceMatrix>
255 API::MatrixWorkspace_sptr evaluateWorkspacesImpl(std::optional<bool> outputWorkspaceDistribution,
256 Provider covarianceMatrixProvider, Ts... args) const {
257 const auto firstWs = std::get<0>(std::forward_as_tuple(args...));
258 API::MatrixWorkspace_sptr outWs = firstWs->clone();
259
260 if (outWs->id() == "EventWorkspace") {
261 outWs = convertToWorkspace2D(outWs);
262 }
263
264 const size_t numSpec = outWs->getNumberHistograms();
265 const size_t specSize = outWs->blocksize();
266
267 // cppcheck-suppress unreadVariable
268 const bool isThreadSafe = Kernel::threadSafe((*args)..., *outWs);
269 // cppcheck-suppress unreadVariable
270 const bool specOverBins = numSpec > specSize;
271
272 PARALLEL_FOR_IF(isThreadSafe && specOverBins)
273 for (int64_t i = 0; i < static_cast<int64_t>(numSpec); i++) {
274 auto &yOut = outWs->mutableY(i);
275 auto &eOut = outWs->mutableE(i);
276
277 PARALLEL_FOR_IF(isThreadSafe && !specOverBins)
278 for (int64_t j = 0; j < static_cast<int64_t>(specSize); ++j) {
279 const InputArray values{args->y(i)[j]...};
280 const InputArray errors(args->e(i)[j]...);
281 const CovarianceMatrix covariance = covarianceMatrixProvider(values, errors);
282 const auto result = evaluateWithCovariance(values, covariance);
283 yOut[j] = result.value;
284 eOut[j] = result.error;
285 }
286 }
287
288 if (outputWorkspaceDistribution.has_value()) {
289 outWs->setDistribution(outputWorkspaceDistribution.value());
290 }
291 return outWs;
292 }
