37bool containsOnlyRectDetectors(const Geometry::ComponentInfo &compInfo, const Geometry::DetectorInfo &detInfo) {
38 using Beamline::ComponentType;
39 std::queue<size_t> compQueue;
40 for (auto const child : compInfo.children(compInfo.root()))
41 compQueue.push(child);
42
43 bool foundRect = false;
44 while (!compQueue.empty()) {
45 size_t const index = compQueue.front();
46 compQueue.pop();
47
48 if ((compInfo.hasSource() && index == compInfo.source()) || (compInfo.hasSample() && index == compInfo.sample()))
49 continue;
50 if (compInfo.isDetector(index) && detInfo.isMonitor(index))
51 continue;
52 // skip choppers, slits and supermirrors - HACK!
53 auto const &name = compInfo.name(index);
54 if (name == "chopper-position" || name.substr(0, 4) == "slit" || name == "supermirror")
55 continue;
56
57 auto const type = compInfo.componentType(index);
58 if (type == ComponentType::Rectangular) {
59 foundRect = true;
60 } else if (type == ComponentType::Unstructured || type == ComponentType::Grid ||
61 type == ComponentType::Structured || type == ComponentType::OutlineComposite) {
62 // An assembly, so check its children
63 for (auto const child : compInfo.children(index))
64 compQueue.push(child);
65 } else {
66 // Found a non-rectangular detector or component
67 return false;
68 }
69 }
70 return foundRect;
71}

87DetectorSearcher::DetectorSearcher(const Geometry::ComponentInfo &compInfo, const Geometry::DetectorInfo &detInfo)
88 : m_usingFullRayTrace(containsOnlyRectDetectors(compInfo, detInfo)), m_crystallography_convention(getQSign()),
89 m_compInfo(compInfo), m_detInfo(detInfo) {
90
91 /* Choose the search strategy to use
92 * If the instrument uses rectangular detectors (e.g. TOPAZ) then it is faster
93 * to run a full ray trace starting from the top of the instrument. This is
94 * due to the speed up of looking up a single pixel in the rectangular
95 * detector.
96 *
97 * If the instrument does not use rectangular detectors (e.g. WISH, CORELLI)
98 * then it is faster to use a nearest neighbour search to find the closest
99 * pixels, then check them for intersection.
100 * */
101 if (!m_usingFullRayTrace) {
102 createDetectorCache();
103 } else {
104 m_rayTracer = std::make_unique<InstrumentRayTracer>(compInfo, detInfo);
105 }
106}

110void DetectorSearcher::createDetectorCache() {
111 std::vector<Eigen::Vector3d> points;
112 points.reserve(m_detInfo.size());
113 m_indexMap.reserve(m_detInfo.size());
114
115 const auto frame = m_compInfo.referenceFrame();
116 auto beam = frame->vecPointingAlongBeam();
117 auto up = frame->vecPointingUp();
118
119 for (size_t pointNo = 0; pointNo < m_detInfo.size(); ++pointNo) {
120 if (m_detInfo.isMonitor(pointNo) || m_detInfo.isMasked(pointNo))
121 continue; // detector is a monitor or masked so don't use
122
123 // Calculate a unit Q vector for each detector
124 // This follows a method similar to that used in IntegrateEllipsoids
125 const auto pos = normalize(m_detInfo.position(pointNo));
126 auto E1 = (pos - beam) * -m_crystallography_convention;
127 const auto norm = E1.norm();
128 if (norm == 0.) {
129 E1 = V3D(up) * -m_crystallography_convention;
130 } else {
131 E1 /= norm;
132 }
133
134 Eigen::Vector3d point(E1[0], E1[1], E1[2]);
135
136 // Ignore nonsensical points
137 if (point.hasNaN() || up.coLinear(beam, pos))
138 continue;
139
140 points.emplace_back(point);
141 m_indexMap.emplace_back(pointNo);
142 }
143
144 // create KDtree of cached detector Q vectors
145 m_detectorCacheSearch = std::make_unique<Kernel::NearestNeighbours<3>>(points);
146}

176DetectorSearcher::DetectorSearchResult DetectorSearcher::searchUsingInstrumentRayTracing(const V3D &q) {
177 const auto direction = convertQtoDirection(q);
178 m_rayTracer->traceFromSample(direction);
179 auto const detIndex = m_rayTracer->getDetectorResult();
180
181 if (detIndex == Geometry::ComponentInfo::invalidIndex || m_detInfo.isMasked(detIndex))
182 return std::make_tuple(false, 0);
183
184 return std::make_tuple(true, detIndex);

195DetectorSearcher::DetectorSearchResult DetectorSearcher::searchUsingNearestNeighbours(const V3D &q) {
196 const auto detectorDir = convertQtoDirection(q);
197 // find where this Q vector should intersect with "extended" space
198 // NOTE: increase the Neighbors from 11 to 21 to cover a wide extended space
199 const auto neighbours = m_detectorCacheSearch->findNearest(Eigen::Vector3d(q[0], q[1], q[2]), 21);
200
201 // check if neighboring is empty
202 if (neighbours.empty()) {
203 g_log.information() << "empty neighbor list for given Q\n";
204 return std::make_tuple(false, 0);
205 }
206
207 const auto result = checkInteceptWithNeighbours(detectorDir, neighbours);
208 const auto hitDetector = std::get<0>(result);
209 const auto index = std::get<1>(result);
210
211 if (hitDetector) {
212 return std::make_tuple(true, m_indexMap[index]);
213 } else if (m_compInfo.hasParameter(m_compInfo.root(), "tube-gap")) {
214 // Tube Gap Parameter specifically applies to tube instruments
215 return handleTubeGap(detectorDir, neighbours);
216 }
217
218 return std::make_tuple(false, 0);
219}

230DetectorSearcher::DetectorSearchResult
231DetectorSearcher::handleTubeGap(const V3D &detectorDir,
232 const Kernel::NearestNeighbours<3>::NearestNeighbourResults &neighbours) {
233 std::vector<double> gaps = m_compInfo.getNumberParameter(m_compInfo.root(), "tube-gap", true);
234 if (!gaps.empty()) {
235 const auto gap = static_cast<double>(gaps.front());
236 // try adding and subtracting tube-gap in 3 q dimensions to see if you can
237 // find detectors on each side of tube gap
238 for (int i = 0; i < 3; i++) {
239 auto gapDir = V3D(0., 0., 0.);
240 gapDir[i] = gap;
241
242 auto beam1 = normalize(detectorDir + gapDir);
243 const auto result1 = checkInteceptWithNeighbours(beam1, neighbours);
244 const auto hit1 = std::get<0>(result1);
245
246 const auto beam2 = normalize(detectorDir - gapDir);
247 const auto result2 = checkInteceptWithNeighbours(beam2, neighbours);
248 const auto hit2 = std::get<0>(result2);
249
250 if (hit1 && hit2) {
251 // Set the detector to one of the neighboring pixels
252 return std::make_tuple(true, m_indexMap[std::get<1>(result1)]);
253 }
254 }
255 }
256
257 return std::make_tuple(false, 0);
258}

267std::tuple<bool, size_t> DetectorSearcher::checkInteceptWithNeighbours(
268 const V3D &direction, const Kernel::NearestNeighbours<3>::NearestNeighbourResults &neighbours) const {
269 Geometry::Track track(m_detInfo.samplePosition(), direction);
270 // Find which of the neighbours we actually intersect with
271 for (const auto &neighbour : neighbours) {
272 const auto index = std::get<1>(neighbour);
273 const auto detIndex = m_indexMap[index];
274
275 Mantid::Geometry::BoundingBox bb;
276 if (!bb.doesLineIntersect(track))
277 continue;
278
279 if (!m_compInfo.hasValidShape(detIndex))
280 throw Kernel::Exception::NullPointerException("DetectorSearcher::checkInteceptWithNeighbours", "shape");
281
282 // Express the track in the detector shape's frame: remove the detector position and rotation
283 auto unRotate = m_detInfo.rotation(detIndex);
284 unRotate.inverse();
285 auto trackStart = track.startPoint() - m_detInfo.position(detIndex);
286 unRotate.rotate(trackStart);
287 auto trackDirection = track.direction();
288 unRotate.rotate(trackDirection);
289 Geometry::Track probeTrack(trackStart, trackDirection);
290
291 const auto hitDetector = m_compInfo.shape(detIndex).interceptSurface(probeTrack) > 0;
292 if (hitDetector)
293 return std::make_tuple(hitDetector, index);
294 }
295
296 return std::make_tuple(false, 0);
297}

304V3D DetectorSearcher::convertQtoDirection(const V3D &q) const {
305 const auto norm_q = q.norm();
306 const auto refFrame = m_compInfo.referenceFrame();
307 const V3D refBeamDir = refFrame->vecPointingAlongBeam();
308
309 const double qBeam = q.scalar_prod(refBeamDir) * m_crystallography_convention;
310 double one_over_wl = (norm_q * norm_q) / (2.0 * qBeam);
311
312 auto detectorDir = q * -m_crystallography_convention;
313 detectorDir[refFrame->pointingAlongBeam()] = one_over_wl - qBeam;
314 detectorDir.normalize();
315 return detectorDir;
316}
