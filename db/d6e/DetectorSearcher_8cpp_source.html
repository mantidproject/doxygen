34bool containsOnlyRectDetectors(const Geometry::ComponentInfo &compInfo, const Geometry::DetectorInfo &detInfo) {
35 using Beamline::ComponentType;
36 std::queue<size_t> compQueue;
37 for (auto const child : compInfo.children(compInfo.root()))
38 compQueue.push(child);
39
40 bool foundRect = false;
41 while (!compQueue.empty()) {
42 size_t const index = compQueue.front();
43 compQueue.pop();
44
45 if ((compInfo.hasSource() && index == compInfo.source()) || (compInfo.hasSample() && index == compInfo.sample()))
46 continue;
47 if (compInfo.isDetector(index) && detInfo.isMonitor(index))
48 continue;
49 // skip choppers, slits and supermirrors - HACK!
50 auto const &name = compInfo.name(index);
51 if (name == "chopper-position" || name.substr(0, 4) == "slit" || name == "supermirror")
52 continue;
53
54 auto const type = compInfo.componentType(index);
55 if (type == ComponentType::Rectangular) {
56 foundRect = true;
57 } else if (type == ComponentType::Unstructured || type == ComponentType::Grid ||
58 type == ComponentType::Structured || type == ComponentType::OutlineComposite) {
59 // An assembly, so check its children
60 for (auto const child : compInfo.children(index))
61 compQueue.push(child);
62 } else {
63 // Found a non-rectangular detector or component
64 return false;
65 }
66 }
67 return foundRect;
68}

84DetectorSearcher::DetectorSearcher(const Geometry::ComponentInfo &compInfo, const Geometry::DetectorInfo &detInfo)
85 : m_usingFullRayTrace(containsOnlyRectDetectors(compInfo, detInfo)), m_crystallography_convention(getQSign()),
86 m_compInfo(compInfo), m_detInfo(detInfo) {
87
88 /* Choose the search strategy to use
89 * If the instrument uses rectangular detectors (e.g. TOPAZ) then it is faster
90 * to run a full ray trace starting from the top of the instrument. This is
91 * due to the speed up of looking up a single pixel in the rectangular
92 * detector.
93 *
94 * If the instrument does not use rectangular detectors (e.g. WISH, CORELLI)
95 * then it is faster to use a nearest neighbour search to find the closest
96 * pixels, then check them for intersection.
97 * */
98 if (!m_usingFullRayTrace) {
99 createDetectorCache();
100 } else {
101 m_rayTracer = std::make_unique<InstrumentRayTracer>(compInfo, detInfo);
102 }
103}

107void DetectorSearcher::createDetectorCache() {
108 std::vector<Eigen::Vector3d> points;
109 points.reserve(m_detInfo.size());
110 m_indexMap.reserve(m_detInfo.size());
111
112 const auto frame = m_compInfo.referenceFrame();
113 auto beam = frame->vecPointingAlongBeam();
114 auto up = frame->vecPointingUp();
115
116 for (size_t pointNo = 0; pointNo < m_detInfo.size(); ++pointNo) {
117 if (m_detInfo.isMonitor(pointNo) || m_detInfo.isMasked(pointNo))
118 continue; // detector is a monitor or masked so don't use
119
120 // Calculate a unit Q vector for each detector
121 // This follows a method similar to that used in IntegrateEllipsoids
122 const auto pos = normalize(m_detInfo.position(pointNo));
123 auto E1 = (pos - beam) * -m_crystallography_convention;
124 const auto norm = E1.norm();
125 if (norm == 0.) {
126 E1 = V3D(up) * -m_crystallography_convention;
127 } else {
128 E1 /= norm;
129 }
130
131 Eigen::Vector3d point(E1[0], E1[1], E1[2]);
132
133 // Ignore nonsensical points
134 if (point.hasNaN() || up.coLinear(beam, pos))
135 continue;
136
137 points.emplace_back(point);
138 m_indexMap.emplace_back(pointNo);
139 }
140
141 // create KDtree of cached detector Q vectors
142 m_detectorCacheSearch = std::make_unique<Kernel::NearestNeighbours<3>>(points);
143}

173DetectorSearcher::DetectorSearchResult DetectorSearcher::searchUsingInstrumentRayTracing(const V3D &q) {
174 const auto direction = convertQtoDirection(q);
175 m_rayTracer->traceFromSample(direction);
176 auto const detIndex = m_rayTracer->getDetectorResult();
177
178 if (detIndex == Geometry::ComponentInfo::invalidIndex || m_detInfo.isMasked(detIndex))
179 return std::make_tuple(false, 0);
180
181 return std::make_tuple(true, detIndex);
182}

192DetectorSearcher::DetectorSearchResult DetectorSearcher::searchUsingNearestNeighbours(const V3D &q) {
193 const auto detectorDir = convertQtoDirection(q);
194 // find where this Q vector should intersect with "extended" space
195 // NOTE: increase the Neighbors from 11 to 21 to cover a wide extended space
196 const auto neighbours = m_detectorCacheSearch->findNearest(Eigen::Vector3d(q[0], q[1], q[2]), 21);
197
198 // check if neighboring is empty
199 if (neighbours.empty()) {
200 g_log.information() << "empty neighbor list for given Q\n";
201 return std::make_tuple(false, 0);
202 }
203
204 const auto result = checkInteceptWithNeighbours(detectorDir, neighbours);
205 const auto hitDetector = std::get<0>(result);
206 const auto index = std::get<1>(result);
207
208 if (hitDetector) {
209 return std::make_tuple(true, m_indexMap[index]);
210 } else if (m_compInfo.hasParameter(m_compInfo.root(), "tube-gap")) {
211 // Tube Gap Parameter specifically applies to tube instruments
212 return handleTubeGap(detectorDir, neighbours);
213 }
214
215 return std::make_tuple(false, 0);
216}

227DetectorSearcher::DetectorSearchResult
228DetectorSearcher::handleTubeGap(const V3D &detectorDir,
229 const Kernel::NearestNeighbours<3>::NearestNeighbourResults &neighbours) {
230 std::vector<double> gaps = m_compInfo.getNumberParameter(m_compInfo.root(), "tube-gap", true);
231 if (!gaps.empty()) {
232 const auto gap = static_cast<double>(gaps.front());
233 // try adding and subtracting tube-gap in 3 q dimensions to see if you can
234 // find detectors on each side of tube gap
235 for (int i = 0; i < 3; i++) {
236 auto gapDir = V3D(0., 0., 0.);
237 gapDir[i] = gap;
238
239 auto beam1 = normalize(detectorDir + gapDir);
240 const auto result1 = checkInteceptWithNeighbours(beam1, neighbours);
241 const auto hit1 = std::get<0>(result1);
242
243 const auto beam2 = normalize(detectorDir - gapDir);
244 const auto result2 = checkInteceptWithNeighbours(beam2, neighbours);
245 const auto hit2 = std::get<0>(result2);
246
247 if (hit1 && hit2) {
248 // Set the detector to one of the neighboring pixels
249 return std::make_tuple(true, m_indexMap[std::get<1>(result1)]);
250 }
251 }
252 }
253
254 return std::make_tuple(false, 0);
255}

264std::tuple<bool, size_t> DetectorSearcher::checkInteceptWithNeighbours(
265 const V3D &direction, const Kernel::NearestNeighbours<3>::NearestNeighbourResults &neighbours) const {
266 Geometry::Track track(m_detInfo.samplePosition(), direction);
267 // Find which of the neighbours we actually intersect with
268 for (const auto &neighbour : neighbours) {
269 const auto index = std::get<1>(neighbour);
270 const auto &det = m_detInfo.detector(m_indexMap[index]);
271
272 Mantid::Geometry::BoundingBox bb;
273 if (!bb.doesLineIntersect(track))
274 continue;
275
276 const auto hitDetector = det.interceptSurface(track) > 0;
277 if (hitDetector)
278 return std::make_tuple(hitDetector, index);
279
280 track.reset(m_detInfo.samplePosition(), direction);
281 }
282
283 return std::make_tuple(false, 0);
284}

291V3D DetectorSearcher::convertQtoDirection(const V3D &q) const {
292 const auto norm_q = q.norm();
293 const auto refFrame = m_compInfo.referenceFrame();
294 const V3D refBeamDir = refFrame->vecPointingAlongBeam();
295
296 const double qBeam = q.scalar_prod(refBeamDir) * m_crystallography_convention;
297 double one_over_wl = (norm_q * norm_q) / (2.0 * qBeam);
298
299 auto detectorDir = q * -m_crystallography_convention;
300 detectorDir[refFrame->pointingAlongBeam()] = one_over_wl - qBeam;
301 detectorDir.normalize();
302 return detectorDir;
303}
