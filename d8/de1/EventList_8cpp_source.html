119// comparator for pulse time with tolerance
120struct comparePulseTimeTOFDelta {
121 explicit comparePulseTimeTOFDelta(const Types::Core::DateAndTime &start, const double seconds)
122 : startNano(start.totalNanoseconds()), deltaNano(static_cast<int64_t>(seconds * SEC_TO_NANO)) {}
123
124 bool operator()(const TofEvent &e1, const TofEvent &e2) {
125 // get the pulse times converted into bin number from start time
126 const int64_t e1Pulse = (e1.pulseTime().totalNanoseconds() - startNano) / deltaNano;
127 const int64_t e2Pulse = (e2.pulseTime().totalNanoseconds() - startNano) / deltaNano;
128
129 // compare with the calculated bin information
130 if (e1Pulse < e2Pulse) {
131 return true;
132 } else if ((e1Pulse == e2Pulse) && (e1.tof() < e2.tof())) {
133 return true;
134 }
135
136 return false;
137 }
138
139 int64_t startNano;
140 int64_t deltaNano;
141};

296void EventList::createFromHistogram(const ISpectrum *inSpec, bool GenerateZeros, bool GenerateMultipleEvents,
297 int MaxEventsPerBin) {
298 // Fresh start
299 this->clear(true);
300
301 // Get the input histogram
302 Mantid::HistogramData::HistogramX const &X = inSpec->x();
303 Mantid::HistogramData::HistogramY const &Y = inSpec->y();
304 Mantid::HistogramData::HistogramE const &E = inSpec->e();
305 if (Y.size() + 1 != X.size()) {
306 throw std::runtime_error("Expected a histogram (X vector should be 1 longer than the Y vector)");
307 }
308
309 // Copy detector IDs and spectra
310 this->copyInfoFrom(*inSpec);
311 // We need weights but have no way to set the time. So use weighted, no time
312 this->switchTo(WEIGHTED_NOTIME);
313 if (GenerateZeros)
314 this->weightedEventsNoTime->reserve(Y.size());
315
316 for (size_t i = 0; i < X.size() - 1; i++) {
317 double weight = Y[i];
318 if ((weight != 0.0 || GenerateZeros) && std::isfinite(weight)) {
319 double error = E[i];
320 // Also check that the error is not a bad number
321 if (std::isfinite(error)) {
322 if (GenerateMultipleEvents) {
323 // --------- Multiple events per bin ----------
324 double errorSquared = error * error;
325 // Find how many events to fake
326 double val = weight / E[i];
327 val *= val;
328 // Convert to int with slight rounding up. This is to avoid rounding
329 // errors
330 auto numEvents = int(val + 0.2);
331 if (numEvents < 1)
332 numEvents = 1;
333 if (numEvents > MaxEventsPerBin)
334 numEvents = MaxEventsPerBin;
335 // Scale the weight and error for each
336 weight /= numEvents;
337 errorSquared /= numEvents;
338
339 // Spread the TOF. e.g. 2 events = 0.25, 0.75.
340 double tofStep = (X[i + 1] - X[i]) / (numEvents);
341 for (size_t j = 0; j < size_t(numEvents); j++) {
342 double tof = X[i] + tofStep * (0.5 + double(j));
343 // Create and add the event
344 // TODO: try emplace_back() here.
345 weightedEventsNoTime->emplace_back(tof, weight, errorSquared);
346 }
347 } else {
348 // --------- Single event per bin ----------
349 // TOF = midpoint of the bin
350 double tof = (X[i] + X[i + 1]) / 2.0;
351 // Error squared is carried in the event
352 double errorSquared = E[i];
353 errorSquared *= errorSquared;
354 // Create and add the event
355 weightedEventsNoTime->emplace_back(tof, weight, errorSquared);
356 }
357 } // error is nont NAN or infinite
358 } // weight is non-zero, not NAN, and non-infinite
359 } // (each bin)
360
361 // Set the X binning parameters
362 this->setSharedX(inSpec->sharedX());
363
364 // Manually set that this is sorted by TOF, since it is. This will make it
365 // "threadSafe" in other algos.
366 this->setSortOrder(TOF_SORT);
367}

567template <class T1, class T2> void EventList::minusHelper(std::vector<T1> &events, const std::vector<T2> &more_events) {
568 // Make the end vector big enough in one go (avoids repeated re-allocations).
569 events.reserve(events.size() + more_events.size());
570 /* In the event of subtracting in place, calling the end() vector would make
571 * it point at the wrong place
572 * Using it caused a segault, Ticket #2306.
573 * So we cache the end (this speeds up too).
574 */
575 // We call the constructor for T1. In the case of WeightedEventNoTime, the pulse time will just be ignored.
576 std::transform(more_events.cbegin(), more_events.cend(), std::back_inserter(events),
577 [](const auto &ev) { return T1(ev.tof(), ev.pulseTime(), ev.weight() * (-1.0), ev.errorSquared()); });
578}

686bool EventList::equals(const EventList &rhs, const double tolTof, const double tolWeight,
687 const int64_t tolPulse) const {
688 // generic checks
689 if (this->getNumberEvents() != rhs.getNumberEvents())
690 return false;
691 if (this->eventType != rhs.eventType)
692 return false;
693 if (this->empty())
694 return true;
695
696 // loop over the events
697 switch (this->eventType) {
698 case TOF: {
699 auto leftIter = this->events->cbegin();
700 auto leftEnd = this->events->cend();
701 auto rightIter = rhs.events->cbegin();
702 while (leftIter != leftEnd) {
703 if (!leftIter->equals(*rightIter, tolTof, tolPulse))
704 return false;
705 leftIter = std::next(leftIter);
706 rightIter = std::next(rightIter);
707 }
708 break;
709 }
710 case WEIGHTED: {
711 auto leftIter = this->weightedEvents->cbegin();
712 auto leftEnd = this->weightedEvents->cend();
713 auto rightIter = rhs.weightedEvents->cbegin();
714 while (leftIter != leftEnd) {
715 if (!leftIter->equals(*rightIter, tolTof, tolWeight, tolPulse))
716 return false;
717 leftIter = std::next(leftIter);
718 rightIter = std::next(rightIter);
719 }
720 break;
721 }
722 case WEIGHTED_NOTIME: {
723 auto leftIter = this->weightedEventsNoTime->cbegin();
724 auto leftEnd = this->weightedEventsNoTime->cend();
725 auto rightIter = rhs.weightedEventsNoTime->cbegin();
726 while (leftIter != leftEnd) {
727 if (!leftIter->equals(*rightIter, tolTof, tolWeight))
728 return false;
729 leftIter = std::next(leftIter);
730 rightIter = std::next(rightIter);
731 }
732 break;
733 }
734 default:
735 break;
736 }
737
738 // anything that gets this far is equal within tolerances
739 return true;
740}

753 switch (newType) {

776void EventList::switchToWeightedEvents() {
777 switch (eventType) {
778 case WEIGHTED:
779 // Do nothing; it already is weighted
780 return;
781
782 case WEIGHTED_NOTIME:
783 throw std::runtime_error("EventList::switchToWeightedEvents() called on an EventList with WeightedEventNoTime's. "

789 // Convert and copy all TofEvents to the weightedEvents list.

797 break;

805void EventList::switchToWeightedEventsNoTime() {
806 switch (eventType) {
807 case WEIGHTED_NOTIME:
808 // Do nothing if already there
809 return;
810
811 case TOF: {
812 if (events && !events->empty()) {
813 // Convert and copy all TofEvents to the weightedEvents list.
814 this->weightedEventsNoTime = std::make_unique<std::vector<WeightedEventNoTime>>(events->cbegin(), events->cend());
815 // Get rid of the old events
816 events.reset();
817 } else {
818 this->weightedEventsNoTime = std::make_unique<std::vector<WeightedEventNoTime>>();
819 }
820 break;
821 }
822
823 case WEIGHTED: {

1591template <class T>
1592inline void EventList::compressEventsHelper(const std::vector<T> &events, std::vector<WeightedEventNoTime> &out,
1593 double tolerance) {
1594 // Clear the output. We can't know ahead of time how much space to reserve :(
1595 out.clear();
1596 // We will make a starting guess of 1/20th of the number of input events.
1597 out.reserve(events.size() / 20);
1598
1599 // The last TOF to which we are comparing.
1600 double lastTof = events.front().m_tof;
1601 // For getting an accurate average TOF
1602 double totalTof = 0;
1603 int num = 0;
1604 // Carrying weight, error, and normalization
1605 double weight = 0;
1606 double errorSquared = 0;
1607 double normalization = 0.;
1608
1609 double bin_end = lastTof;
1610 std::function<bool(const double, const double)> compareTof;
1611 std::function<double(const double, double)> next_bin;
1612
1613 if (tolerance < 0) { // log
1614 if (lastTof < 0)
1615 throw std::runtime_error("compressEvents with log binning doesn't work with negative TOF");
1616
1617 if (lastTof == 0)
1618 bin_end = fabs(tolerance);
1619
1620 // for log we do "less than" so that is matches the log binning of the Rebin algorithm
1621 compareTof = [](const double lhs, const double rhs) { return lhs < rhs; };
1622 next_bin = [tolerance](const double lastTof, double bin_end) {
1623 // advance the bin_end until we find the one that this next event falls into
1624 while (lastTof >= bin_end)
1625 bin_end = bin_end * (1 - tolerance);
1626 return bin_end;
1627 };
1628 } else { // linear
1629 // for linear we do "less than or equals" because that is how it was originally implemented
1630 compareTof = [](const double lhs, const double rhs) { return lhs <= rhs; };
1631 next_bin = [tolerance](const double lastTof, double) { return lastTof + tolerance; };
1632 }
1633
1634 // get first bin_end
1635 bin_end = next_bin(lastTof, bin_end);
1636
1637 for (auto it = events.cbegin(); it != events.cend(); it++) {
1638 if (compareTof(it->m_tof, bin_end)) {
1639 // Carry the error and weight
1640 weight += it->weight();
1641 errorSquared += it->errorSquared();
1642 // Track the average tof
1643 num++;
1644 const double norm = calcNorm(it->errorSquared());
1645 normalization += norm;
1646 totalTof += it->m_tof * norm;
1647 } else {
1648 // We exceeded the tolerance
1649 // Create a new event with the average TOF and summed weights and
1650 // squared errors.
1651 if (num == 1) {
1652 // last time-of-flight is the only one contributing
1653 out.emplace_back(lastTof, weight, errorSquared);
1654 } else if (num > 1) {
1655 out.emplace_back(totalTof / normalization, weight, errorSquared);
1656 }
1657 // Start a new combined object
1658 num = 1;
1659 const double norm = calcNorm(it->errorSquared());
1660 normalization = norm;
1661 totalTof = it->m_tof * norm;
1662 weight = it->weight();
1663 errorSquared = it->errorSquared();
1664 lastTof = it->m_tof;
1665
1666 bin_end = next_bin(lastTof, bin_end);
1667 }
1668 }
1669
1670 // Put the last event in there too with the average TOF and summed weights and
1671 // squared errors.
1672 if (num == 1) {
1673 // last time-of-flight is the only one contributing
1674 out.emplace_back(lastTof, weight, errorSquared);
1675 } else if (num > 1) {
1676 out.emplace_back(totalTof / normalization, weight, errorSquared);
1677 }
1678
1679 // If you have over-allocated by more than 5%, reduce the size.
1680 size_t excess_limit = out.size() / 20;
1681 if ((out.capacity() - out.size()) > excess_limit) {
1682 out.shrink_to_fit();
1683 }
1684}

1686template <class T>
1687inline void EventList::compressFatEventsHelper(const std::vector<T> &events, std::vector<WeightedEvent> &out,
1688 const double tolerance, const Types::Core::DateAndTime &timeStart,
1689 const double seconds) {
1690 // Clear the output. We can't know ahead of time how much space to reserve :(
1691 out.clear();
1692 // We will make a starting guess of 1/20th of the number of input events.
1693 out.reserve(events.size() / 20);
1694
1695 // The last TOF to which we are comparing.
1696 double lastTof = events.front().m_tof;
1697 // For getting an accurate average TOF
1698 double totalTof = 0;
1699
1700 // pulsetime bin information - stored as int nanoseconds because it
1701 // is the implementation type for DateAndTime object
1702 const int64_t pulsetimeStart = timeStart.totalNanoseconds();
1703 const auto pulsetimeDelta = static_cast<int64_t>(seconds * SEC_TO_NANO);
1704
1705 // pulsetime information
1706 std::vector<DateAndTime> pulsetimes; // all the times for new event
1707 std::vector<double> pulsetimeWeights;
1708
1709 // Carrying weight and error
1710 double weight = 0.;
1711 double errorSquared = 0.;
1712 double tofNormalization = 0.;
1713
1714 // Move up to first event that has a large enough pulsetime. This is just in case someone starts from after the
1715 // starttime of the run. It is expected that users will normally use the default which means this will only check the
1716 // first event.
1717 auto it = events.cbegin();
1718 for (; it != events.cend(); ++it) {
1719 if (it->m_pulsetime >= timeStart)
1720 break;
1721 }
1722
1723 if (it == events.cend())
1724 throw std::runtime_error("failed to find first pulse time in the events");
1725
1726 // bin if the pulses are histogrammed
1727 int64_t lastPulseBin = (it->m_pulsetime.totalNanoseconds() - pulsetimeStart) / pulsetimeDelta;
1728
1729 double bin_end = lastTof;
1730 double tof_min{0};
1731 std::function<bool(const double, const double)> compareTof;
1732 std::function<double(const double, double)> next_bin;
1733
1734 if (tolerance < 0) { // log
1735 // for log we do "less than" so that is matches the log binning of the Rebin algorithm
1736 compareTof = [](const double lhs, const double rhs) { return lhs < rhs; };
1737 next_bin = [tolerance](const double lastTof, double bin_end) {
1738 // advance the bin_end until we find the one that this next event falls into
1739 while (lastTof >= bin_end)
1740 bin_end = bin_end * (1 - tolerance);
1741 return bin_end;
1742 };
1743
1744 // get minimum Tof so that binning is consistent across all pulses
1745 const auto event_min = std::min_element(
1746 events.cbegin(), events.cend(), [](const auto &left, const auto &right) { return left.tof() < right.tof(); });
1747 bin_end = tof_min = event_min->tof();
1748
1749 if (tof_min < 0)
1750 throw std::runtime_error("compressEvents with log binning doesn't work with negative TOF");
1751
1752 // can't start at 0 as this will create an infinite loop
1753 if (tof_min == 0)
1754 bin_end = tof_min = fabs(tolerance);
1755
1756 } else { // linear
1757 // for linear we do "less than or equals" because that is how it was originally implemented
1758 compareTof = [](const double lhs, const double rhs) { return lhs <= rhs; };
1759 next_bin = [tolerance](const double lastTof, double) { return lastTof + tolerance; };
1760 }
1761
1762 // get first bin_end
1763 bin_end = next_bin(lastTof, bin_end);
1764
1765 // loop through events and accumulate weight
1766 for (; it != events.cend(); ++it) {
1767 const int64_t eventPulseBin = (it->m_pulsetime.totalNanoseconds() - pulsetimeStart) / pulsetimeDelta;
1768 if ((eventPulseBin <= lastPulseBin) && compareTof(it->m_tof, bin_end)) {
1769 // Carry the error and weight
1770 weight += it->weight();
1771 errorSquared += it->errorSquared();
1772 double norm = calcNorm(it->errorSquared());
1773 tofNormalization += norm;
1774 // Track the average tof
1775 totalTof += it->m_tof * norm;
1776 // Accumulate the pulse times
1777 pulsetimes.emplace_back(it->m_pulsetime);
1778 pulsetimeWeights.emplace_back(norm);
1779 } else {
1780 // We exceeded the tolerance
1781 if (!pulsetimes.empty()) {
1782 // Create a new event with the average TOF and summed weights and
1783 // squared errors. 1 event used doesn't need to average
1784 if (pulsetimes.size() == 1) {
1785 out.emplace_back(lastTof, pulsetimes.front(), weight, errorSquared);
1786 } else {
1787 out.emplace_back(totalTof / tofNormalization,
1788 Kernel::DateAndTimeHelpers::averageSorted(pulsetimes, pulsetimeWeights), weight,
1789 errorSquared);
1790 }
1791 }
1792 if (tolerance < 0 && eventPulseBin != lastPulseBin)
1793 // reset the bin_end for the new pulse bin
1794 bin_end = tof_min;
1795
1796 // Start a new combined object
1797 double norm = calcNorm(it->errorSquared());
1798 totalTof = it->m_tof * norm;
1799 weight = it->weight();
1800 errorSquared = it->errorSquared();
1801 tofNormalization = norm;
1802 lastTof = it->m_tof;
1803 lastPulseBin = eventPulseBin;
1804 pulsetimes.clear();
1805 pulsetimes.emplace_back(it->m_pulsetime);
1806 pulsetimeWeights.clear();
1807 pulsetimeWeights.emplace_back(norm);
1808
1809 bin_end = next_bin(lastTof, bin_end);
1810 }
1811 }
1812
1813 // Put the last event in there too.
1814 if (!pulsetimes.empty()) {
1815 // Create a new event with the average TOF and summed weights and
1816 // squared errors. 1 event used doesn't need to average
1817 if (pulsetimes.size() == 1) {
1818 out.emplace_back(lastTof, pulsetimes.front(), weight, errorSquared);
1819 } else {
1820 out.emplace_back(totalTof / tofNormalization,
1821 Kernel::DateAndTimeHelpers::averageSorted(pulsetimes, pulsetimeWeights), weight, errorSquared);
1822 }
1823 }
1824
1825 // If you have over-allocated by more than 5%, reduce the size.
1826 size_t excess_limit = out.size() / 20;
1827 if ((out.capacity() - out.size()) > excess_limit) {
1828 out.shrink_to_fit();
1829 }
1830}

2135
2136 if (x_size <= 1) {
2137 // X was not set. Return an empty array.
2138 Y.resize(0, 0);
2139 return;
2140 }
2141
2142 // If the sizes are the same, then the "resize" command will NOT clear the original values.
2143 bool mustFill = (Y.size() == x_size - 1);
2144 // Clear the Y data, assign all to 0.
2145 Y.resize(x_size - 1, 0.0);
2146 // Clear the Error data, assign all to 0.
2147 // Note: Errors will be squared until the last step.
2148 E.resize(x_size - 1, 0.0);
2149
2150 if (mustFill) {
2151 // We must make sure the starting point is 0.0
2152 std::fill(Y.begin(), Y.end(), 0.0);
2153 std::fill(E.begin(), E.end(), 0.0);
2154 }
2155
2156 //---------------------- Histogram without weights
2157 //---------------------------------
2158
2159 // Do we even have any events to do?
2160 if (!events.empty()) {
2161 // Iterate through all events (sorted by tof)
2162 auto itev = findFirstEvent(events, T(X[0]));
2163 auto itev_end = events.cend();
2164 // The above can still take you to end() if no events above X[0], so check
2165 // again.
2166 if (itev == itev_end)
2167 return;
2168
2169 // Find the first bin
2170 size_t bin = 0;
2171 // The tof is greater the first bin boundary, so we need to find the first
2172 // bin
2173 double tof = itev->tof();
2174 while (bin < x_size - 1) {

2178 // precision)

2180 E[bin] += double(itev->m_errorSquared); // square of error

2182 }

2186 ++itev;
2187

2189 while ((itev != itev_end) && (bin < x_size - 1)) {
2190 tof = itev->tof();
2191 while (bin < x_size - 1) {
2192 // Within range? Since both events and X are sorted, they are going to
2193 // have
2194 // tof >= X[bin] because the previous event was.

2196 // Add up the weight (convert to double before adding, to preserve
2197 // precision)
2198 Y[bin] += double(itev->m_weight);
2199 E[bin] += double(itev->m_errorSquared); // square of error
2200 break;
2201 }
2202 ++bin;
2203 }
2204 ++itev;
2205 }
2206 } // end if (there are any events to histogram)
2207
2208 // Now do the sqrt of all errors
2209 std::transform(E.cbegin(), E.cend(), E.begin(), static_cast<double (*)(double)>(sqrt));
2210}
