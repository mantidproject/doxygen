68std::map<std::string, std::string> SetInstrumentParameter::validateInputs() {
69 std::map<std::string, std::string> errors;
70 const std::set<std::string> allowedBoolValues{"1", "0", "true", "false"};
71 const std::string type = getPropertyValue("ParameterType");
72 std::string val = getPropertyValue("Value");
73
74 if (type == "Bool") {
75 boost::algorithm::to_lower(val);
76 if (allowedBoolValues.find(val) == allowedBoolValues.end()) {
77 errors["Value"] = "Invalid value for Bool type.";
78 }
79 } else if (type == "Number") {
80 int intVal;
81 double dblVal;
82 if (!Strings::convert(val, intVal) && !Strings::convert(val, dblVal)) {
83 errors["Value"] = "Invalid value for Number type.";
84 }
85 }
86
87 return errors;
88}

93void SetInstrumentParameter::exec() {
94 // A pointer to the workspace to add a log to
95 Workspace_sptr ws = getProperty("Workspace");
96 MatrixWorkspace_sptr inputW = std::dynamic_pointer_cast<MatrixWorkspace>(ws);
97 DataObjects::PeaksWorkspace_sptr inputP = std::dynamic_pointer_cast<DataObjects::PeaksWorkspace>(ws);
98 ExperimentInfo_sptr experimentInfo;
99 if (inputW) {
100 experimentInfo = inputW;
101 } else if (inputP) {
102 experimentInfo = inputP;
103 } else {
104 throw std::runtime_error("Could not get a valid instrument from the "
105 "workspace which does not seem to be valid as "
106 "input (must be either MatrixWorkspace or "
107 "PeaksWorkspace");
108 }
109 auto &componentInfo = experimentInfo->mutableComponentInfo();
110
111 // get the data that the user wants to add
112 std::string cmptName = getProperty("ComponentName");
113 const std::vector<detid_t> detectorList = getProperty("DetectorList");
114 std::string paramName = getProperty("ParameterName");
115 std::string paramType = getProperty("ParameterType");
116 std::string paramValue = getPropertyValue("Value");
117
118 Strings::strip(cmptName);
119 Strings::strip(paramName);
120 Strings::strip(paramValue);
121
122 // set default to whole instrument
123 std::vector<size_t> componentIndices{componentInfo.root()};
124 if (!detectorList.empty()) {
125 const auto &detectorInfo = experimentInfo->detectorInfo();
126 componentIndices.clear();
127 for (const auto detID : detectorList) {
128 try {
129 // Detector indices are also component indices
130 componentIndices.emplace_back(detectorInfo.indexOf(detID));
131 } catch (const std::out_of_range &) {
132 throw Kernel::Exception::NotFoundError("Instrument: Detector with ID " + std::to_string(detID) + " not found.",
133 "");
134 }
135 }
136 } else if (cmptName.length() > 0) {
137 // get all matching cmpts
138 componentIndices = componentsWithName(componentInfo, cmptName);
139 }
140
141 if (componentIndices.empty()) {
142 g_log.warning("Could not find the component requested.");
143 }
144 for (const auto componentIndex : componentIndices) {
145 addParameter(componentInfo, componentIndex, paramName, paramType, paramValue);
146 }
147}

155std::vector<size_t> SetInstrumentParameter::componentsWithName(const ComponentInfo &componentInfo,
156 const std::string &name) {
157 std::vector<size_t> matches;
158 const auto root = componentInfo.root();
159 if (componentInfo.name(root) == name) {
160 matches.emplace_back(root);
161 }
162 std::deque<size_t> toSearch{root};
163 while (!toSearch.empty()) {
164 const auto index = toSearch.front();
165 toSearch.pop_front();
166 for (const auto child : componentInfo.children(index)) {
167 if (componentInfo.name(child) == name) {
168 matches.emplace_back(child);
169 } else {
170 toSearch.emplace_back(child);
171 }
172 }
173 }
174 return matches;
175}

184void SetInstrumentParameter::addParameter(ComponentInfo &componentInfo, const size_t componentIndex,
185 const std::string &paramName, const std::string &paramType,
186 const std::string &paramValue) const {
187
188 // remove existing parameters first
189 componentInfo.clearParameter(componentIndex, paramName);
190 // then add the new one
191 if (paramType == "String") {
192 componentInfo.addString(componentIndex, paramName, paramValue);
193 } else if (paramType == "Number") {
194 int intVal;
195 if (Strings::convert(paramValue, intVal)) {
196 componentInfo.addInt(componentIndex, paramName, intVal);
197 } else {
198 double dblVal;
199 Strings::convert(paramValue, dblVal);
200 componentInfo.addDouble(componentIndex, paramName, dblVal);
201 }
202 } else if (paramType == "Bool") {
203 std::string paramValueLower(paramValue);
204 boost::algorithm::to_lower(paramValueLower);
205 bool paramBoolValue = (paramValueLower == "true" || paramValue == "1");
206 componentInfo.addBool(componentIndex, paramName, paramBoolValue);
207 }
208}
