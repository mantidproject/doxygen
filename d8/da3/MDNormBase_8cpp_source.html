42std::string MDNormBase::inputEnergyMode() const {
43 const auto &history = m_inputWS->getHistory();
44 const size_t nalgs = history.size();
45 const auto &lastAlgorithm = history.getAlgorithmHistory(nalgs - 1);
46
47 std::string emode;
48 if (lastAlgorithm->name() == "ConvertToMD") {
49 // get dEAnalysisMode
50 emode = lastAlgorithm->getPropertyValue("dEAnalysisMode");
51 } else {
52 if ((lastAlgorithm->name() == "Load" || lastAlgorithm->name() == "LoadMD") && nalgs > 1) {
53 const auto &penultimateAlgorithm = history.getAlgorithmHistory(nalgs - 2);
54 if (penultimateAlgorithm->name() == "ConvertToMD") {
55 return penultimateAlgorithm->getPropertyValue("dEAnalysisMode");
56 }
57 }
58 throw std::invalid_argument("The last algorithm in the history of the "
59 "input workspace is not ConvertToMD");
60 }
61 return emode;
62}

111std::vector<coord_t> MDNormBase::getValuesFromOtherDimensions(bool &skipNormalization, uint16_t expInfoIndex) const {
112 const auto &currentRun = m_inputWS->getExperimentInfo(expInfoIndex)->run();
113
114 std::vector<coord_t> otherDimValues;
115 size_t lastCol = m_diffraction ? 3 : 4;
116 for (size_t i = lastCol; i < m_inputWS->getNumDims(); i++) {
117 const auto dimension = m_inputWS->getDimension(i);
118 auto dimMin = static_cast<float>(dimension->getMinimum());
119 auto dimMax = static_cast<float>(dimension->getMaximum());
120 auto *dimProp = dynamic_cast<Kernel::TimeSeriesProperty<double> *>(currentRun.getProperty(dimension->getName()));
121 if (dimProp) {
122 auto value = static_cast<coord_t>(dimProp->firstValue());
123 otherDimValues.emplace_back(value);
124 // in the original MD data no time was spent measuring between dimMin and
125 // dimMax
126 if (value < dimMin || value > dimMax) {
127 skipNormalization = true;
128 }
129 }
130 }
131 return otherDimValues;

142void MDNormBase::findIntegratedDimensions(const std::vector<coord_t> &otherDimValues, bool &skipNormalization) {
143 // Get indices of the original dimensions in the output workspace,
144 // and if not found, the corresponding dimension is integrated
145 m_transformation = m_normWS->getTransformFromOriginal(0)->makeAffineMatrix();
146
147 const size_t nrm1 = m_transformation.numRows() - 1;
148 const size_t ncm1 = m_transformation.numCols() - 1;
149 const size_t lastCol = m_diffraction ? 3 : 4;
150 for (size_t row = 0; row < nrm1; row++) // affine matrix, ignore last row
151 {
152 const auto dimen = m_normWS->getDimension(row);
153 const auto dimMin(dimen->getMinimum()), dimMax(dimen->getMaximum());
154 if (m_transformation[row][0] == 1.) {
155 m_hIntegrated = false;
156 m_hIdx = row;
157 m_hmin = std::max(m_hmin, dimMin);
158 m_hmax = std::min(m_hmax, dimMax);
159 if (m_hmin > dimMax || m_hmax < dimMin) {
160 skipNormalization = true;
161 }
162 }
163 if (m_transformation[row][1] == 1.) {
164 m_kIntegrated = false;
165 m_kIdx = row;
166 m_kmin = std::max(m_kmin, dimMin);
167 m_kmax = std::min(m_kmax, dimMax);
168 if (m_kmin > dimMax || m_kmax < dimMin) {
169 skipNormalization = true;
170 }
171 }
172 if (m_transformation[row][2] == 1.) {
173 m_lIntegrated = false;
174 m_lIdx = row;
175 m_lmin = std::max(m_lmin, dimMin);
176 m_lmax = std::min(m_lmax, dimMax);
177 if (m_lmin > dimMax || m_lmax < dimMin) {
178 skipNormalization = true;
179 }
180 }
181
182 if (!m_diffraction && m_transformation[row][3] == 1.) {
183 m_dEIntegrated = false;
184 m_eIdx = row;
185 m_dEmin = std::max(m_dEmin, dimMin);
186 m_dEmax = std::min(m_dEmax, dimMax);
187 if (m_dEmin > dimMax || m_dEmax < dimMin) {
188 skipNormalization = true;
189 }
190 }
191 for (size_t col = lastCol; col < ncm1; col++) // affine matrix, ignore last column
192 {
193 if (m_transformation[row][col] == 1.) {
194 double val = otherDimValues.at(col - lastCol);
195 if (val > dimMax || val < dimMin) {
196 skipNormalization = true;
197 }
198 }
199 }
200 }

207void MDNormBase::cacheDimensionXValues() {
208 if (!m_hIntegrated) {
209 auto &hDim = *m_normWS->getDimension(m_hIdx);
210 m_hX = std::vector<double>(hDim.getNBoundaries());
211 for (size_t i = 0; i < m_hX.size(); ++i) {
212 m_hX[i] = hDim.getX(i);
213 }
214 }
215 if (!m_kIntegrated) {
216 auto &kDim = *m_normWS->getDimension(m_kIdx);
217 m_kX = std::vector<double>(kDim.getNBoundaries());
218 for (size_t i = 0; i < m_kX.size(); ++i) {
219 m_kX[i] = kDim.getX(i);
220 }
221 }
222 if (!m_lIntegrated) {
223 auto &lDim = *m_normWS->getDimension(m_lIdx);
224 m_lX = std::vector<double>(lDim.getNBoundaries());
225 for (size_t i = 0; i < m_lX.size(); ++i) {
226 m_lX[i] = lDim.getX(i);
227 }
228 }
229 if ((!m_diffraction) && (!m_dEIntegrated)) {
230 // NOTE: store k final instead
231 auto &eDim = *m_normWS->getDimension(m_eIdx);
232 m_eX = std::vector<double>(eDim.getNBoundaries());
233 for (size_t i = 0; i < m_eX.size(); ++i) {
234 double temp = m_Ei - eDim.getX(i);
235 temp = std::max(temp, 0.);
236 m_eX[i] = std::sqrt(energyToK * temp);
237 }
238 }

248Mantid::Kernel::DblMatrix MDNormBase::calQTransform(const DblMatrix &R, const Geometry::SymmetryOperation &so,
249 bool doInvert) {
250 // Make it to a method!
251 DblMatrix soMatrix(3, 3);
252 auto v = so.transformHKL(V3D(1, 0, 0));
253 soMatrix.setColumn(0, v);
254 v = so.transformHKL(V3D(0, 1, 0));
255 soMatrix.setColumn(1, v);
256 v = so.transformHKL(V3D(0, 0, 1));
257 soMatrix.setColumn(2, v);
258 soMatrix.Invert();
259 DblMatrix Qtransform = R * m_UB * soMatrix * m_W;
260 if (doInvert)
261 Qtransform.Invert();
262
263 return Qtransform;
264}

272void MDNormBase::calculateNormalization(const std::vector<coord_t> &otherValues, uint16_t expInfoIndex) {
273 const auto &currentExptInfo = *(m_inputWS->getExperimentInfo(expInfoIndex));
274 const auto &spectrumInfo = currentExptInfo.spectrumInfo();
275 auto *rubwLog = dynamic_cast<VectorDoubleProperty *>(currentExptInfo.getLog("RUBW_MATRIX"));
276 if (!rubwLog) {
277 throw std::runtime_error("Wokspace does not contain a log entry for the RUBW matrix."
278 "Cannot continue.");
279 }
280 Kernel::DblMatrix Qtransform((*rubwLog)()); // includes the 2*pi factor but not goniometer for now :)
281 Qtransform = currentExptInfo.run().getGoniometerMatrix() * Qtransform;
282 Qtransform.Invert();
283 const double protonCharge = currentExptInfo.run().getProtonCharge();
284
285 calculateNormInner(spectrumInfo, otherValues, protonCharge, 0., Qtransform);
286}

295void MDNormBase::calculateNormalization(const std::vector<coord_t> &otherValues, const Geometry::SymmetryOperation &so,
296 uint16_t expInfoIndex) {
297 const auto &currentExptInfo = *(m_inputWS->getExperimentInfo(expInfoIndex));
298 std::vector<double> lowValues, highValues;
299 auto *lowValuesLog = dynamic_cast<VectorDoubleProperty *>(currentExptInfo.getLog("MDNorm_low"));
300 lowValues = (*lowValuesLog)();
301 auto *highValuesLog = dynamic_cast<VectorDoubleProperty *>(currentExptInfo.getLog("MDNorm_high"));
302 highValues = (*highValuesLog)();
303
304 // calculate Q transformation matrix (R * UB * SymmetryOperation * m_W)^-1
305 // in order to calculate intersections
306 Kernel::DblMatrix Qtransform = calQTransform(currentExptInfo.run().getGoniometerMatrix(), so);
307
308 // get proton charges
309 const double protonCharge = currentExptInfo.run().getProtonCharge();
310 const double protonChargeBkgd =
311 (m_backgroundWS != nullptr) ? m_backgroundWS->getExperimentInfo(0)->run().getProtonCharge() : 0;
312
313 const auto &spectrumInfo = currentExptInfo.spectrumInfo();
314
315 calculateNormInner(spectrumInfo, otherValues, protonCharge, protonChargeBkgd, Qtransform, lowValues, highValues);
316}

324void MDNormBase::calculateNormContinuous(const std::vector<coord_t> &otherValues, uint16_t expInfoIndex,
325 const Geometry::SymmetryOperation *so) {
326 const auto &currentExptInfo = *(m_inputWS->getExperimentInfo(expInfoIndex));
327 const auto &spectrumInfo = currentExptInfo.spectrumInfo();
328 bool isMDNorm = false;
329 std::vector<double> lowValues, highValues;
330 DblMatrix UBWSymm; // Symmetrised matrix (UB * symm * W) converting from lab to crystal frame
331 if (so == nullptr) {
332 auto *rubwLog = dynamic_cast<VectorDoubleProperty *>(currentExptInfo.getLog("RUBW_MATRIX"));
333 UBWSymm = (*rubwLog)(); // includes the 2*pi factor but not goniometer for now :)
334 } else {
335 if (m_backgroundWS != nullptr) {
336 throw std::runtime_error("Continuous rotation and background workspace not yet implemented.");
337 }
338 isMDNorm = true;
339 UBWSymm = calQTransform(DblMatrix(3, 3, true), *so, false);
340 auto *lowValuesLog = dynamic_cast<VectorDoubleProperty *>(currentExptInfo.getLog("MDNorm_low"));
341 lowValues = (*lowValuesLog)();
342 auto *highValuesLog = dynamic_cast<VectorDoubleProperty *>(currentExptInfo.getLog("MDNorm_high"));
343 highValues = (*highValuesLog)();
344 }
345 // MDEventWS was created with the "useLogTimes" option: should be only a single expInfo, but
346 // gonios vary with time - we now coarse-bin it to compute the normalisation.
347 const Run &run = currentExptInfo.run();
348 if (!run.hasProperty(LOG_CHARGE_NAME)) {
349 throw std::runtime_error("Wokspace does not contain the proton charge log. Cannot continue.");
350 }
351
352 double progressStart = 0.3 + 0.7 * expInfoIndex / m_numExptInfos;
353 double progressEnd = 0.3 + 0.7 * (expInfoIndex + 1) / m_numExptInfos;
354 double normfac = run.hasProperty("NormalizationFactor")
355 ? (*dynamic_cast<Kernel::PropertyWithValue<double> *>(run.getProperty("NormalizationFactor")))()
356 : 1.0;
357 std::istringstream tosplit;
358 tosplit.str((*dynamic_cast<PropertyWithValue<std::string> *>(run.getProperty("useLogTimes")))());
359 std::vector<TimeSeriesProperty<double> *> logs;
360 std::vector<size_t> movingGonioIndex;
361 const TimeSeriesProperty<double> *protonlog = run.getTimeSeriesProperty<double>(LOG_CHARGE_NAME);
362 std::vector<double> protonCharge = protonlog->valuesAsVector();
363 std::vector<Types::Core::DateAndTime> protonTimes = protonlog->timesAsVector();
364 Geometry::Goniometer gonio(run.getGoniometer());
365
366 for (std::string name; std::getline(tosplit, name, ',');) {
367 auto *log = run.getTimeSeriesProperty<double>(name);
368 logs.push_back(log);
369 if ((log->maxValue() - log->minValue()) > STATIONARYANGLIM) { // Assume gonio logs in degrees
370 movingGonioIndex.push_back(logs.size() - 1);
371 }
372 }
373
374 // Convert from picoCoulomb to uA.hr for SNS data
375 if (protonlog->units().find("picoCoulomb") != std::string::npos) {
376 normfac *= 3600.e6;
377 }
378
379 if (movingGonioIndex.size() == 1) {
380 // If we only have a single moving gonio, bin all its values to GONIOBINSTEP degree bins and
381 // run inner loop on each binned angle
382 const TimeROI &timeroi = run.getTimeROI();
383 const auto &gonioAxLog = logs[movingGonioIndex[0]];
384 std::vector<double> filteredVals = gonioAxLog->filteredValuesAsVector(&timeroi);
385 const auto &[min, max] = std::minmax_element(filteredVals.begin(), filteredVals.end());
386 std::vector<double> gonioCharge(static_cast<int>((*max - *min) / GONIOBINSTEP) + 1, 0.0);
387 for (size_t n = 0; n < protonCharge.size(); n++) {
388 double logval = gonioAxLog->getSingleValue(protonTimes[n]);
389 if (std::isnan(logval) || logval > *max || logval < *min) {
390 continue;
391 }
392 auto idx = static_cast<size_t>(floor((logval - *min) / GONIOBINSTEP));
393 gonioCharge[idx] += protonCharge[n];
394 }
395 m_progress->resetNumSteps(static_cast<int64_t>(gonioCharge.size()), progressStart, progressEnd);
396 for (size_t n = 0; n < gonioCharge.size(); n++) {
397 if (gonioCharge[n] < MINPROTONCHARGE) {
398 continue;
399 }
400 auto nn = static_cast<double>(n);
401 gonio.setRotationAngle(movingGonioIndex[0], nn * GONIOBINSTEP + *min);
402 DblMatrix Qtransform = gonio.getR() * UBWSymm;
403 Qtransform.Invert();
404 if (isMDNorm) {
405 calculateNormInner(spectrumInfo, otherValues, gonioCharge[n] / normfac, 0., Qtransform, lowValues, highValues);
406 } else {
407 calculateNormInner(spectrumInfo, otherValues, gonioCharge[n] / normfac, 0., Qtransform);
408 }
409 m_progress->report();
410 }
411 } else {
412 // Otherwise run inner loop over small bins of proton charge in time
413 double chargeSum = 0.0;
414 size_t i0 = 0;
415 bool skipIter = false;
416 m_progress->resetNumSteps(static_cast<int64_t>(protonCharge.size()), progressStart, progressEnd);
417 for (size_t n = 0; n < protonCharge.size(); n++) {
418 chargeSum += protonCharge[n];
419 if (chargeSum > CHARGEBINSIZE) {
420 size_t index = static_cast<int>(floor(static_cast<double>(n - i0) / 2.)) + i0;
421 skipIter = false;
422 for (size_t gAx = 0; gAx < gonio.getNumberAxes(); gAx++) {
423 double logval = logs[gAx]->getSingleValue(protonTimes[index]);
424 if (std::isnan(logval)) {
425 skipIter = true;
426 continue;
427 }
428 gonio.setRotationAngle(gAx, logval);
429 }
430 if (!skipIter) {
431 DblMatrix Qtrans = gonio.getR() * UBWSymm;
432 Qtrans.Invert();
433 if (isMDNorm) {
434 calculateNormInner(spectrumInfo, otherValues, chargeSum / normfac, 0., Qtrans, lowValues, highValues);
435 } else {
436 calculateNormInner(spectrumInfo, otherValues, chargeSum / normfac, 0., Qtrans);
437 }
438 }
439 chargeSum = 0;
440 i0 = n;
441 }
442 m_progress->report();
443 }
444 }
445 if (m_numExptInfos > 1) {
446 m_progress->resetNumSteps(m_numExptInfos - expInfoIndex, progressStart, 1.0);
447 }
448}

450void MDNormBase::calculateNormInner(const API::SpectrumInfo &spectrumInfo, const std::vector<coord_t> &otherValues,
451 const double protonCharge, const double protonChargeBkgd,
452 const Kernel::DblMatrix &Qtransform, const std::vector<double> &lowValues,
453 const std::vector<double> &highValues) {
454 // Mapping
455 const auto ndets = static_cast<int64_t>(spectrumInfo.size());
456 bool haveSA = false;
457 API::MatrixWorkspace_const_sptr integrFlux, solidAngleWS = getProperty("SolidAngleWorkspace");
458 detid2index_map fluxDetToIdx, solidAngDetToIdx;
459 bool thread_safe = true;
460 if (m_diffraction) {
461 integrFlux = getProperty("FluxWorkspace"); // FluxWorkspace is mandatory for diffraction
462 integrFlux->getXMinMax(m_kfmin, m_kfmax);
463 fluxDetToIdx = integrFlux->getDetectorIDToWorkspaceIndexMap();
464 thread_safe = Kernel::threadSafe(*integrFlux);
465 }
466 if (solidAngleWS != nullptr) {
467 haveSA = true;
468 solidAngDetToIdx = solidAngleWS->getDetectorIDToWorkspaceIndexMap();
469 }
470
471 const size_t vmdDims = m_diffraction ? 3 : 4;
472 size_t numNPoints = (m_backgroundWS) ? m_bkgdNormWS->getNPoints() : 0;
473 if (m_backgroundWS && numNPoints != m_normWS->getNPoints()) {
474 throw std::runtime_error("N points are different");
475 }
476
477 bool isMDNorm = !lowValues.empty();
478 if (isMDNorm) {
479 m_hmin = static_cast<coord_t>(m_hX[0]);
480 m_kmin = static_cast<coord_t>(m_kX[0]);
481 m_lmin = static_cast<coord_t>(m_lX[0]);
482 m_hmax = static_cast<coord_t>(m_hX.back());
483 m_kmax = static_cast<coord_t>(m_kX.back());
484 m_lmax = static_cast<coord_t>(m_lX.back());
485 }
486
487 PRAGMA_OMP(parallel for if(thread_safe))
488 for (int64_t i = 0; i < ndets; i++) {
489 PARALLEL_START_INTERRUPT_REGION
490
491 if (!spectrumInfo.hasDetectors(i) || spectrumInfo.isMonitor(i) || spectrumInfo.isMasked(i)) {
492 continue;
493 }
494 const auto &detector = spectrumInfo.detector(i);
495 double theta = detector.getTwoTheta(m_samplePos, m_beamDir);
496 double phi = detector.getPhi();
497 // If the detector is a group, this should be the ID of the first detector
498 const auto detID = detector.getID();
499
500 // Get the flux spectrum spectrum number for diffraction
501 size_t wsIdx = 0;
502 if (m_diffraction) {
503 if (auto index = fluxDetToIdx.find(detID); index != fluxDetToIdx.end()) {
504 wsIdx = index->second;
505 } else { // masked detector in flux, but not in input workspace
506 continue;
507 }
508 }
509
510 // Intersections
511 std::vector<std::array<double, 4>> intersections;
512 std::vector<coord_t> pos, posNew;
513 if (isMDNorm) {
514 this->calculateIntersections(intersections, theta, phi, Qtransform, lowValues[i], highValues[i]);
515 } else {
516 this->calculateIntersections(intersections, theta, phi, Qtransform);
517 }
518 // No need to do normalization calculation if there is no intersection
519 if (intersections.empty())
520 continue;
521
522 // Get solid angle for this contribution
523 double solid = protonCharge;
524 double bkgdSolid = protonChargeBkgd;
525 if (haveSA) {
526 double solid_angle_factor = solidAngleWS->y(solidAngDetToIdx.find(detID)->second)[0];
527 solid = solid_angle_factor * protonCharge;
528 bkgdSolid = solid_angle_factor * protonChargeBkgd;
529 }
530
531 // -- calculate integrals for the intersection --
532 // momentum values at intersections
533 std::vector<double> yValues;
534 if (m_diffraction) {
535 // copy momenta to xValues
536 std::vector<double> xValues(intersections.size());
537 yValues.resize(intersections.size());
538 auto x = xValues.begin();
539 for (auto it = intersections.begin(); it != intersections.end(); ++it, ++x) {
540 *x = (*it)[3];
541 }
542 // calculate integrals at momenta from xValues by interpolating between
543 // points in spectrum sp
544 // of workspace integrFlux. The result is stored in yValues
545 calcIntegralsForIntersections(xValues, *integrFlux, wsIdx, yValues);
546 }
547
548 // Compute final position in HKL
549 // pre-allocate for efficiency and copy non-hkl dim values into place
550 pos.resize(vmdDims + otherValues.size() + 1);
551 std::copy(otherValues.begin(), otherValues.end(), pos.begin() + vmdDims);
552 pos.emplace_back(1.f);
553 for (auto it = intersections.begin() + 1; it != intersections.end(); ++it) {
554 const auto &curIntSec = *it;
555 const auto &prevIntSec = *(it - 1);
556 // the full vector isn't used so compute only what is necessary
557 double delta = m_diffraction ? (curIntSec[3] - prevIntSec[3]) / 1000. // tolerance is 1e-7
558 : (curIntSec[3] * curIntSec[3] - prevIntSec[3] * prevIntSec[3]) / energyToK;
559 if (delta < 1e-10)
560 continue; // Assume zero contribution if difference is small
561
562 // Average between two intersections for final position
563 std::transform(curIntSec.data(), curIntSec.data() + vmdDims, prevIntSec.data(), pos.begin(),
564 [](const double rhs, const double lhs) { return static_cast<coord_t>(0.5 * (rhs + lhs)); });
565
566 // transform kf to energy transfer
567 if (!m_diffraction) {
568 pos[3] = static_cast<coord_t>(m_Ei - pos[3] * pos[3] / energyToK);
569 }
570 m_transformation.multiplyPoint(pos, posNew);
571 size_t linIndex = m_normWS->getLinearIndexAtCoord(posNew.data());
572 if (linIndex == static_cast<size_t>(-1))
573 continue;
574
575 if (m_diffraction) {
576 // index of the current intersection
577 auto k = static_cast<size_t>(std::distance(intersections.begin(), it));
578 delta = (yValues[k] - yValues[k - 1]);
579 }
580 // signal = delta * solid = integral between two consecutive intersections * solid angle
581 Mantid::Kernel::AtomicOp(m_signalArray[linIndex], delta * solid, std::plus<signal_t>());
582 if (m_backgroundWS) {
583 Mantid::Kernel::AtomicOp(m_bkgdSignalArray[linIndex], delta * bkgdSolid, std::plus<signal_t>());
584 }
585 }
586 PARALLEL_END_INTERRUPT_REGION
587 }
588 PARALLEL_CHECK_INTERRUPT_REGION

599void MDNormBase::calcIntegralsForIntersections(const std::vector<double> &xValues,
600 const API::MatrixWorkspace &integrFlux, size_t sp,
601 std::vector<double> &yValues) const {
602 assert(xValues.size() == yValues.size());
603
604 // the x-data from the workspace
605 const auto &xData = integrFlux.x(sp);
606 const double xStart = xData.front();
607 const double xEnd = xData.back();
608
609 // the values in integrFlux are expected to be integrals of a non-negative
610 // function
611 // ie they must make a non-decreasing function
612 const auto &yData = integrFlux.y(sp);
613 size_t spSize = yData.size();
614
615 const double yMin = 0.0;
616 const double yMax = yData.back();
617
618 size_t nData = xValues.size();
619 // all integrals below xStart must be 0
620 if (xValues[nData - 1] < xStart) {
621 std::fill(yValues.begin(), yValues.end(), yMin);
622 return;
623 }
624
625 // all integrals above xEnd must be equal tp yMax
626 if (xValues[0] > xEnd) {
627 std::fill(yValues.begin(), yValues.end(), yMax);
628 return;
629 }
630
631 size_t i = 0;
632 // integrals below xStart must be 0
633 while (i < nData - 1 && xValues[i] < xStart) {
634 yValues[i] = yMin;
635 i++;
636 }
637 size_t j = 0;
638 for (; i < nData; i++) {
639 // integrals above xEnd must be equal tp yMax
640 if (j >= spSize - 1) {
641 yValues[i] = yMax;
642 } else {
643 double xi = xValues[i];
644 while (j < spSize - 1 && xi > xData[j])
645 j++;
646 // if x falls onto an interpolation point return the corresponding y
647 if (xi == xData[j]) {
648 yValues[i] = yData[j];
649 } else if (j == spSize - 1) {
650 // if we get above xEnd it's yMax
651 yValues[i] = yMax;
652 } else if (j > 0) {
653 // interpolate between the consecutive points
654 double x0 = xData[j - 1];
655 double x1 = xData[j];
656 double y0 = yData[j - 1];
657 double y1 = yData[j];
658 yValues[i] = y0 + (y1 - y0) * (xi - x0) / (x1 - x0);
659 } else // j == 0
660 {
661 yValues[i] = yMin;
662 }
663 }
664 }

678void MDNormBase::calculateIntersections(std::vector<std::array<double, 4>> &intersections, const double theta,
679 const double phi, const Kernel::DblMatrix &transform, double lowvalue,
680 double highvalue) {
681 V3D qin, qout;
682 double kfmin, kfmax, kimin, kimax;
683 bool isMDNorm = !std::isnan(lowvalue);
684 if (isMDNorm) {
685 qout = V3D(sin(theta) * cos(phi), sin(theta) * sin(phi), cos(theta));
686 qin = V3D(0., 0., 1.);
687 if (m_diffraction) {
688 kimin = kfmin = lowvalue;
689 kimax = kfmax = highvalue;
690 } else {
691 kimin = kimax = std::sqrt(energyToK * m_Ei);
692 kfmin = std::sqrt(energyToK * (m_Ei - highvalue));
693 kfmax = std::sqrt(energyToK * (m_Ei - lowvalue));
694 }
695 } else {
696 if (m_diffraction) {
697 qin = qout = V3D(-sin(theta) * cos(phi), -sin(theta) * sin(phi), 1. - cos(theta));
698 kimin = kfmin = m_kfmin;
699 kimax = kfmax = m_kfmax;
700 } else {
701 qout = V3D(sin(theta) * cos(phi), sin(theta) * sin(phi), cos(theta));
702 qin = V3D(0., 0., 1.);
703 kimin = kimax = m_ki;
704 kfmin = m_kfmin;
705 kfmax = m_kfmax;
706 }
707 }
708
709 qout = transform * qout;
710 qin = transform * qin;
711 if (m_convention == "Crystallography") {
712 qout *= -1;
713 qin *= -1;
714 }
715 double hStart = qin.X() * kimin - qout.X() * kfmin, hEnd = qin.X() * kimax - qout.X() * kfmax;
716 double kStart = qin.Y() * kimin - qout.Y() * kfmin, kEnd = qin.Y() * kimax - qout.Y() * kfmax;
717 double lStart = qin.Z() * kimin - qout.Z() * kfmin, lEnd = qin.Z() * kimax - qout.Z() * kfmax;
718 double eps = 1e-10;
719 auto hNBins = m_hX.size();
720 auto kNBins = m_kX.size();
721 auto lNBins = m_lX.size();
722 auto eNBins = m_eX.size();
723 intersections.clear();
724 intersections.reserve(hNBins + kNBins + lNBins + eNBins + 8); // 8 is 3*(min,max for each Q component)+kfmin+kfmax
725
726 // calculate intersections with planes perpendicular to h
727 if (fabs(hStart - hEnd) > eps) {
728 double fmom = (kfmax - kfmin) / (hEnd - hStart);
729 double fk = (kEnd - kStart) / (hEnd - hStart);
730 double fl = (lEnd - lStart) / (hEnd - hStart);
731 if (!m_hIntegrated) {
732 for (size_t i = 0; i < hNBins; i++) {
733 double hi = m_hX[i];
734 if ((hi >= m_hmin) && (hi <= m_hmax) && ((hStart - hi) * (hEnd - hi) < 0)) {
735 // if hi is between hStart and hEnd, then ki and li will be between
736 // kStart, kEnd and lStart, lEnd and momi will be between kfmin and kfmax
737 double ki = fk * (hi - hStart) + kStart;
738 double li = fl * (hi - hStart) + lStart;
739 if ((ki >= m_kmin) && (ki <= m_kmax) && (li >= m_lmin) && (li <= m_lmax)) {
740 double momi = fmom * (hi - hStart) + kfmin;
741 intersections.push_back({{hi, ki, li, momi}});
742 }
743 }
744 }
745 }
746 if (!isMDNorm) {
747 double momhMin = fmom * (m_hmin - hStart) + kfmin;
748 if ((momhMin - kfmin) * (momhMin - kfmax) < 0) // kfmin>kfmax
749 {
750 // khmin and lhmin
751 double khmin = fk * (m_hmin - hStart) + kStart;
752 double lhmin = fl * (m_hmin - hStart) + lStart;
753 if ((khmin >= m_kmin) && (khmin <= m_kmax) && (lhmin >= m_lmin) && (lhmin <= m_lmax)) {
754 intersections.push_back({{m_hmin, khmin, lhmin, momhMin}});
755 }
756 }
757 double momhMax = fmom * (m_hmax - hStart) + kfmin;
758 if ((momhMax - kfmin) * (momhMax - kfmax) <= 0) {
759 // khmax and lhmax
760 double khmax = fk * (m_hmax - hStart) + kStart;
761 double lhmax = fl * (m_hmax - hStart) + lStart;
762 if ((khmax >= m_kmin) && (khmax <= m_kmax) && (lhmax >= m_lmin) && (lhmax <= m_lmax)) {
763 intersections.push_back({{m_hmax, khmax, lhmax, momhMax}});
764 }
765 }
766 }
767 }
768
769 // calculate intersections with planes perpendicular to k
770 if (fabs(kStart - kEnd) > eps) {
771 double fmom = (kfmax - kfmin) / (kEnd - kStart);
772 double fh = (hEnd - hStart) / (kEnd - kStart);
773 double fl = (lEnd - lStart) / (kEnd - kStart);
774 if (!m_kIntegrated) {
775 for (size_t i = 0; i < kNBins; i++) {
776 double ki = m_kX[i];
777 if ((ki >= m_kmin) && (ki <= m_kmax) && ((kStart - ki) * (kEnd - ki) < 0)) {
778 // if ki is between kStart and kEnd, then hi and li will be between
779 // hStart, hEnd and lStart, lEnd and momi will be between kfmin and kfmax
780 double hi = fh * (ki - kStart) + hStart;
781 double li = fl * (ki - kStart) + lStart;
782 if ((hi >= m_hmin) && (hi <= m_hmax) && (li >= m_lmin) && (li <= m_lmax)) {
783 double momi = fmom * (ki - kStart) + kfmin;
784 intersections.push_back({{hi, ki, li, momi}});
785 }
786 }
787 }
788 }
789 if (!isMDNorm) {
790 double momkMin = fmom * (m_kmin - kStart) + kfmin;
791 if ((momkMin - kfmin) * (momkMin - kfmax) < 0) {
792 // hkmin and lkmin
793 double hkmin = fh * (m_kmin - kStart) + hStart;
794 double lkmin = fl * (m_kmin - kStart) + lStart;
795 if ((hkmin >= m_hmin) && (hkmin <= m_hmax) && (lkmin >= m_lmin) && (lkmin <= m_lmax)) {
796 intersections.push_back({{hkmin, m_kmin, lkmin, momkMin}});
797 }
798 }
799 double momkMax = fmom * (m_kmax - kStart) + kfmin;
800 if ((momkMax - kfmin) * (momkMax - kfmax) <= 0) {
801 // hkmax and lkmax
802 double hkmax = fh * (m_kmax - kStart) + hStart;
803 double lkmax = fl * (m_kmax - kStart) + lStart;
804 if ((hkmax >= m_hmin) && (hkmax <= m_hmax) && (lkmax >= m_lmin) && (lkmax <= m_lmax)) {
805 intersections.push_back({{hkmax, m_kmax, lkmax, momkMax}});
806 }
807 }
808 }
809 }
810
811 // calculate intersections with planes perpendicular to l
812 if (fabs(lStart - lEnd) > eps) {
813 double fmom = (kfmax - kfmin) / (lEnd - lStart);
814 double fh = (hEnd - hStart) / (lEnd - lStart);
815 double fk = (kEnd - kStart) / (lEnd - lStart);
816 if (!m_lIntegrated) {
817 for (size_t i = 0; i < lNBins; i++) {
818 double li = m_lX[i];
819 if ((li >= m_lmin) && (li <= m_lmax) && ((lStart - li) * (lEnd - li) < 0)) {
820 // if li is between lStart and lEnd, then hi and ki will be between
821 // hStart, hEnd and kStart, kEnd
822 double hi = fh * (li - lStart) + hStart;
823 double ki = fk * (li - lStart) + kStart;
824 if ((hi >= m_hmin) && (hi <= m_hmax) && (ki >= m_kmin) && (ki <= m_kmax)) {
825 double momi = fmom * (li - lStart) + kfmin;
826 intersections.push_back({{hi, ki, li, momi}});
827 }
828 }
829 }
830 }
831 if (!isMDNorm) {
832 double momlMin = fmom * (m_lmin - lStart) + kfmin;
833 if ((momlMin - kfmin) * (momlMin - kfmax) <= 0) {
834 // hlmin and klmin
835 double hlmin = fh * (m_lmin - lStart) + hStart;
836 double klmin = fk * (m_lmin - lStart) + kStart;
837 if ((hlmin >= m_hmin) && (hlmin <= m_hmax) && (klmin >= m_kmin) && (klmin <= m_kmax)) {
838 intersections.push_back({{hlmin, klmin, m_lmin, momlMin}});
839 }
840 }
841 double momlMax = fmom * (m_lmax - lStart) + kfmin;
842 if ((momlMax - kfmin) * (momlMax - kfmax) < 0) {
843 // hlmax and klmax
844 double hlmax = fh * (m_lmax - lStart) + hStart;
845 double klmax = fk * (m_lmax - lStart) + kStart;
846 if ((hlmax >= m_hmin) && (hlmax <= m_hmax) && (klmax >= m_kmin) && (klmax <= m_kmax)) {
847 intersections.push_back({{hlmax, klmax, m_lmax, momlMax}});
848 }
849 }
850 }
851 }
852
853 // intersections with dE
854 if (!m_dEIntegrated) {
855 for (size_t i = 0; i < eNBins; i++) {
856 double kfi = m_eX[i];
857 if ((kfi - kfmin) * (kfi - kfmax) <= 0) {
858 double h = qin.X() * kimin - qout.X() * kfi;
859 double k = qin.Y() * kimin - qout.Y() * kfi;
860 double l = qin.Z() * kimin - qout.Z() * kfi;
861 if ((h >= m_hmin) && (h <= m_hmax) && (k >= m_kmin) && (k <= m_kmax) && (l >= m_lmin) && (l <= m_lmax)) {
862 intersections.push_back({{h, k, l, kfi}});
863 }
864 }
865 }
866 }
867
868 // endpoints
869 if ((hStart >= m_hmin) && (hStart <= m_hmax) && (kStart >= m_kmin) && (kStart <= m_kmax) && (lStart >= m_lmin) &&
870 (lStart <= m_lmax)) {
871 intersections.push_back({{hStart, kStart, lStart, kfmin}});
872 }
873 if ((hEnd >= m_hmin) && (hEnd <= m_hmax) && (kEnd >= m_kmin) && (kEnd <= m_kmax) && (lEnd >= m_lmin) &&
874 (lEnd <= m_lmax)) {
875 intersections.push_back({{hEnd, kEnd, lEnd, kfmax}});
876 }
877
878 // sort intersections by final momentum
879 std::stable_sort(intersections.begin(), intersections.end(), compareMomentum);
880}
