207 template <class T> std::vector<T> getType(const std::string &compName, const std::string &name) const {
208 std::vector<T> retval;
209
210 pmap_cit it;
211 for (it = m_map.begin(); it != m_map.end(); ++it) {
212 if (compName == it->first->getName()) {
213 std::shared_ptr<Parameter> param = get(it->first, name);
214 if (param)
215 retval.emplace_back(param->value<T>());
216 }
217 }
218 return retval;
219 }
