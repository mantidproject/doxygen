221 template <class T> std::vector<T> getType(const std::string &compName, const std::string &name) const {
222 std::vector<T> retval;
223
224 pmap_cit it;
225 for (it = m_map.begin(); it != m_map.end(); ++it) {
226 if (compName == it->first->getName()) {
227 std::shared_ptr<Parameter> param = get(it->first, name);
228 if (param)
229 retval.emplace_back(param->value<T>());
230 }
231 }
232 return retval;
233 }
