36detid_t gridDetectorId(Beamline::PixelGridComponent const &grid, int const x, int const y, int const z) {
37 auto const coordinate = [x, y, z](char const axis) { return axis == 'x' ? x : axis == 'y' ? y : z; };
38 auto const pixels = [&grid](char const axis) { return axis == 'x' ? grid.nX : axis == 'y' ? grid.nY : grid.nZ; };
39 auto const &order = grid.idFillOrder;
40 return grid.idStart + coordinate(order[0]) * grid.idStep + coordinate(order[1]) * grid.idStepByRow +
41 coordinate(order[2]) * pixels(order[1]) * grid.idStepByRow;
42}

46class BeamlineBuilder::Layout {
47public:
48 using Rotations = std::vector<Eigen::Quaterniond, Eigen::aligned_allocator<Eigen::Quaterniond>>;

50 explicit Layout(BeamlineBuilder const &builder) : m_builder(builder), m_nodeIndices(builder.m_nodes.size()) {
51 auto const &nodes = builder.m_nodes;
52 // Detectors take the leading indices, in detector ID order
53 std::vector<Node> detectors;
54 std::ranges::copy_if(std::views::iota(Node{0}, nodes.size()), std::back_inserter(detectors),
55 [&nodes](Node const node) { return nodes[node].isDetector; });
56 std::sort(detectors.begin(), detectors.end(),
57 [&nodes](Node const a, Node const b) { return nodes[a].id < nodes[b].id; });
58 auto const duplicate =
59 std::adjacent_find(detectors.cbegin(), detectors.cend(),
60 [&nodes](Node const a, Node const b) { return nodes[a].id == nodes[b].id; });
61 if (duplicate != detectors.cend()) {
62 throw Kernel::Exception::InstrumentDefinitionError(
63 "BeamlineBuilder: detector ID " + std::to_string(nodes[*duplicate].id) + " is used more than once");
64 }
65 auto const nDetectors = detectors.size();
66 m_detectorIds = std::make_shared<std::vector<detid_t>>(nDetectors);
67 auto detectorIdToIndex = std::make_shared<std::unordered_map<detid_t, size_t>>();
68 detectorIdToIndex->reserve(nDetectors);
69 for (size_t index = 0; index < nDetectors; ++index) {
70 auto const id = nodes[detectors[index]].id;
71 (*m_detectorIds)[index] = id;
72 (*detectorIdToIndex)[id] = index;
73 m_nodeIndices[detectors[index]] = index;
74 }
75 m_detectorIdToIndex = std::move(detectorIdToIndex);
76
77 m_parentIndices = std::make_shared<std::vector<size_t>>(nDetectors, 0);
78 m_detectorPositions.resize(nDetectors);
79 m_detectorRotations.resize(nDetectors);
80 m_shapes = std::make_shared<std::vector<std::shared_ptr<const IObject>>>(nDetectors, m_nullShape);
81 m_scaleFactors = std::make_shared<std::vector<Eigen::Vector3d>>(nDetectors, Eigen::Vector3d{1, 1, 1});
82 m_names = std::make_shared<std::vector<std::string>>(nDetectors);
83 m_assemblySortedDetectorIndices->reserve(nDetectors);
84
85 placeNodes();
86 visit(0);
87 }

115 void placeNodes() {
116 auto const &nodes = m_builder.m_nodes;
117 m_absolutePositions.resize(nodes.size());
118 m_absoluteRotations.resize(nodes.size());
119 m_absoluteSideBySide.resize(nodes.size());
120 for (Node node = 0; node < nodes.size(); ++node) {
121 auto const &record = nodes[node];
122 if (record.parent == noParent) {
123 m_absolutePositions[node] = record.position;
124 m_absoluteRotations[node] = record.rotation;
125 m_absoluteSideBySide[node] = record.sideBySideViewPosition;
126 continue;
127 }
128 V3D position(record.position);
129 m_absoluteRotations[record.parent].rotate(position);
130 m_absolutePositions[node] = position + m_absolutePositions[record.parent];
131 m_absoluteRotations[node] = m_absoluteRotations[record.parent] * record.rotation;
132 auto const &parentSideBySide = m_absoluteSideBySide[record.parent];
133 if (record.sideBySideViewPosition || parentSideBySide) {
134 m_absoluteSideBySide[node] =
135 record.sideBySideViewPosition.value_or(V2D(0., 0.)) + parentSideBySide.value_or(V2D(0., 0.));
136 }
137 }
138 }

141 size_t visit(Node const node) {
142 auto const &record = m_builder.m_nodes[node];
143 if (record.isDetector) {
144 return visitDetector(node);
145 }
146 if (!record.isAssembly) {
147 m_detectorRanges->emplace_back(0, 0);
148 auto const componentIndex = commonRegistration(node);
149 auto const componentStart = m_assemblySortedComponentIndices->size();
150 m_componentRanges->emplace_back(componentStart, componentStart + 1);
151 m_assemblySortedComponentIndices->emplace_back(componentIndex);
152 m_parentIndices->emplace_back(componentIndex);
153 m_children->emplace_back();
154 return componentIndex;
155 }
156 auto const detectorStart = m_assemblySortedDetectorIndices->size();
157 auto const componentStart = m_assemblySortedComponentIndices->size();
158 auto const &childNodes = m_builder.m_children[node];
159 std::vector<size_t> children(childNodes.size());
160 for (size_t i = 0; i < childNodes.size(); ++i) {
161 children[i] = visit(childNodes[i]);
162 }
163 auto const detectorStop = m_assemblySortedDetectorIndices->size();
164 auto const componentIndex = commonRegistration(node);
165 m_assemblySortedComponentIndices->emplace_back(componentIndex);
166 // The root is its own parent; any other parent is set when its parent is registered
167 m_parentIndices->emplace_back(componentIndex);
168 auto const componentStop = m_assemblySortedComponentIndices->size();
169 m_detectorRanges->emplace_back(detectorStart, detectorStop);
170 m_componentRanges->emplace_back(componentStart, componentStop);
171 for (auto const child : children) {
172 (*m_parentIndices)[child] = componentIndex;
173 }
174 m_children->emplace_back(std::move(children));
175 if (record.pixelGrid) {
176 (*m_pixelGridComponents)[componentIndex] = *record.pixelGrid;
177 }
178 if (record.structuredGrid) {
179 (*m_structuredGridComponents)[componentIndex] = *record.structuredGrid;
180 }
181 return componentIndex;
182 }
183
184 size_t visitDetector(Node const node) {
185 auto const &record = m_builder.m_nodes[node];
186 auto const detectorIndex = m_nodeIndices[node];
187 m_assemblySortedDetectorIndices->emplace_back(detectorIndex);
188 m_detectorPositions[detectorIndex] = Kernel::toVector3d(m_absolutePositions[node]);
189 m_detectorRotations[detectorIndex] = Kernel::toQuaterniond(m_absoluteRotations[node]);
190 (*m_shapes)[detectorIndex] = record.shape ? *record.shape : m_nullShape;
191 (*m_scaleFactors)[detectorIndex] = Kernel::toVector3d(record.scaleFactor);
192 if (record.isMonitor) {
193 m_monitorIndices.emplace_back(detectorIndex);
194 }
195 (*m_names)[detectorIndex] = record.name;
196 if (auto const &sideBySide = m_absoluteSideBySide[node]) {
197 (*m_sideBySideViewPositions)[detectorIndex] = Kernel::toVector2d(*sideBySide);
198 }
199 return detectorIndex;
200 }
201
203 size_t commonRegistration(Node const node) {
204 auto const &record = m_builder.m_nodes[node];
205 auto const componentIndex = m_names->size();
206 m_nodeIndices[node] = componentIndex;
207 m_positions->emplace_back(Kernel::toVector3d(m_absolutePositions[node]));
208 m_rotations->emplace_back(Kernel::toQuaterniond(m_absoluteRotations[node]));
209 m_shapes->emplace_back(record.shape ? *record.shape : m_nullShape);
210 m_scaleFactors->emplace_back(Kernel::toVector3d(record.scaleFactor));
211 m_names->emplace_back(record.name);
212 m_componentTypes->emplace_back(record.type);
213 if (auto const &sideBySide = m_absoluteSideBySide[node]) {
214 (*m_sideBySideViewPositions)[componentIndex] = Kernel::toVector2d(*sideBySide);
215 }
216 return componentIndex;
217 }

303void BeamlineBuilder::addGridPixels(Node const bank, std::shared_ptr<const IObject> const &pixelShape) {
304 // Copied, because adding nodes reallocates m_nodes
305 auto const name = m_nodes[bank].name;
306 auto grid = *m_nodes[bank].pixelGrid;
307 if (grid.nX <= 0 || grid.nY <= 0 || grid.nZ < 0) {
308 throw std::invalid_argument("BeamlineBuilder: grid bank " + name + " needs at least one pixel in x and y");
309 }
310 if (!std::ranges::all_of(grid.idFillOrder, [](char const axis) { return axis >= 'x' && axis <= 'z'; })) {
311 throw std::invalid_argument("BeamlineBuilder: grid bank " + name + " has an ID fill order other than x, y and z");
312 }
313 auto const layered = grid.nZ > 0;
314 grid.minDetectorID = grid.idStart;
315 grid.maxDetectorID = grid.idStart;
316 for (int iz = 0; iz < std::max(grid.nZ, 1); ++iz) {
317 auto const layer = layered ? addAssembly(bank, name + "(z=" + std::to_string(iz) + ")") : bank;
318 auto const z = grid.zStart + iz * grid.zStep;
319 auto const columnPrefix = layered ? name + "(z=" + std::to_string(iz) + ",x=" : name + "(x=";
320 auto const pixelSuffix = layered ? "," + std::to_string(iz) + ")" : std::string(")");
321 for (int ix = 0; ix < grid.nX; ++ix) {
322 auto const column = addAssembly(layer, columnPrefix + std::to_string(ix) + ")");
323 auto const x = grid.xStart + ix * grid.xStep;
324 for (int iy = 0; iy < grid.nY; ++iy) {
325 auto const id = gridDetectorId(grid, ix, iy, iz);
326 grid.minDetectorID = std::min(grid.minDetectorID, id);
327 grid.maxDetectorID = std::max(grid.maxDetectorID, id);
328 auto const y = grid.yStart + iy * grid.yStep;
329 addDetector(column, name + "(" + std::to_string(ix) + "," + std::to_string(iy) + pixelSuffix, id, V3D(x, y, z),
330 Quat(), pixelShape);
331 }
332 }
333 }
334 m_nodes[bank].pixelGrid = grid;
335}

370BeamlineBuilder::Node BeamlineBuilder::addCopy(Node const parent, ComponentInfo const &componentInfo,
371 DetectorInfo const &detectorInfo, size_t const index) {
372 auto const &name = componentInfo.name(index);
373 auto const hasParent = componentInfo.hasParent(index);
374 auto const position = hasParent ? componentInfo.relativePosition(index) : componentInfo.position(index);
375 auto const rotation = hasParent ? componentInfo.relativeRotation(index) : componentInfo.rotation(index);
376 auto const &shape = (*componentInfo.m_shapes)[index];
377 Node node;
378 if (componentInfo.isDetector(index)) {
379 node = addDetector(parent, name, detectorInfo.detectorIDs()[index], position, rotation, shape,
380 detectorInfo.isMonitor(index));
381 } else {
382 switch (componentInfo.componentType(index)) {
383 case Beamline::ComponentType::Unstructured:
384 // An outline assembly without an outline holds no shape at all
385 node = shape ? addAssembly(parent, name, position, rotation)
386 : addOutlineAssembly(parent, name, position, rotation, nullptr);
387 break;
388 case Beamline::ComponentType::OutlineComposite:
389 node = addOutlineAssembly(parent, name, position, rotation, shape);
390 break;
391 case Beamline::ComponentType::Rectangular:
392 node = addRectangularBank(parent, name, position, rotation, componentInfo.pixelGridComponent(index));
393 break;
394 case Beamline::ComponentType::Grid:
395 node = addGridBank(parent, name, position, rotation, componentInfo.pixelGridComponent(index));
396 break;
397 case Beamline::ComponentType::Structured:
398 node = addStructuredBank(parent, name, position, rotation, componentInfo.structuredGridComponent(index));
399 break;
400 case Beamline::ComponentType::Generic:
401 node = shape == ComponentInfo::noShape() ? addComponent(parent, name, position, rotation)
402 : addObjComponent(parent, name, position, rotation, shape);
403 break;
404 case Beamline::ComponentType::Infinite:
405 node = addObjComponent(parent, name, position, rotation, shape);
406 break;
407 default:
408 throw std::invalid_argument("BeamlineBuilder: cannot copy " + name + ", which has an unexpected type");
409 }
410 }
411 // Side-by-side view positions are held absolute; a component holds its own relative to its parent's, and one equal
412 // to its parent's is taken to be inherited
413 auto const sideBySide = componentInfo.sideBySideViewPosition(index);
414 if (sideBySide.X() != EMPTY_DBL()) {
415 auto const parentSideBySide =
416 hasParent ? componentInfo.sideBySideViewPosition(componentInfo.parent(index)) : V2D(EMPTY_DBL(), EMPTY_DBL());
417 if (parentSideBySide.X() == EMPTY_DBL()) {
418 setSideBySideViewPosition(node, sideBySide);
419 } else if (sideBySide != parentSideBySide) {
420 setSideBySideViewPosition(node, sideBySide - parentSideBySide);
421 }
422 }
423 setScaleFactor(node, componentInfo.scaleFactor(index));
424 for (auto const child : componentInfo.children(index)) {
425 addCopy(node, componentInfo, detectorInfo, child);
426 }
427 return node;
428}

430BeamlineBuilder::Node BeamlineBuilder::add(NodeRecord record) {
431 if (m_nodes.empty()) {
432 if (record.parent != noParent) {
433 throw std::invalid_argument("BeamlineBuilder: the first node is the root and has no parent");
434 }
435 if (!record.isAssembly) {
436 throw std::invalid_argument("BeamlineBuilder: the root must be an assembly");
437 }
438 } else {
439 if (record.parent == noParent) {
440 throw std::invalid_argument("BeamlineBuilder: only the first node can be the root");
441 }
442 checkNode(record.parent);
443 if (!m_nodes[record.parent].isAssembly) {
444 throw std::invalid_argument("BeamlineBuilder: " + m_nodes[record.parent].name +
445 " is not an assembly, so cannot have children");
446 }
447 }
448 Node const node = m_nodes.size();
449 if (record.parent != noParent) {
450 m_children[record.parent].emplace_back(node);
451 }
452 m_nodes.emplace_back(std::move(record));
453 m_children.emplace_back();
454 return node;
455}

519bool BeamlineBuilder::isPixelGrid(Node const node, int const columns, int const pixels) const {
520 auto const &columnNodes = m_children[node];
521 if (columns <= 0 || pixels <= 0 || columnNodes.size() != static_cast<size_t>(columns)) {
522 return false;
523 }
524 return std::ranges::all_of(columnNodes, [this, pixels](Node const column) {
525 auto const &pixelNodes = m_children[column];
526 return m_nodes[column].isAssembly && pixelNodes.size() == static_cast<size_t>(pixels) &&
527 std::ranges::all_of(pixelNodes, [this](Node const pixel) { return m_nodes[pixel].isDetector; });
528 });
529}

532void BeamlineBuilder::checkBanks() const {
533 for (Node node = 0; node < m_nodes.size(); ++node) {
534 auto const &record = m_nodes[node];
535 bool valid = true;
536 if (record.structuredGrid) {
537 valid = isPixelGrid(node, record.structuredGrid->nX, record.structuredGrid->nY);
538 } else if (record.pixelGrid) {
539 auto const &grid = *record.pixelGrid;
540 if (record.type == Beamline::ComponentType::Grid && grid.nZ > 0) {
541 auto const &layers = m_children[node];
542 valid = layers.size() == static_cast<size_t>(grid.nZ) &&
543 std::ranges::all_of(layers, [this, &grid](Node const layer) {
544 return m_nodes[layer].isAssembly && isPixelGrid(layer, grid.nX, grid.nY);
545 });
546 } else {
547 // A rectangular bank is flat
548 valid = (record.type != Beamline::ComponentType::Rectangular || grid.nZ == 0) &&
549 isPixelGrid(node, grid.nX, grid.nY);
550 }
551 }
552 if (!valid) {
553 throw std::invalid_argument("BeamlineBuilder: bank " + record.name +
554 " does not hold the grid of pixels it describes");
555 }
556 }
557}

559BeamlineRef BeamlineBuilder::build() const {
560 if (m_nodes.empty()) {
561 throw std::logic_error("BeamlineBuilder: there is no root node to build");
562 }
563 checkBanks();
564 Layout layout(*this);
565
566 auto parameterInfo = std::make_shared<ParameterInfo>();
567 for (auto const &[node, parameter] : m_parameters) {
568 parameterInfo->add(layout.index(node), parameter);
569 }
570 for (auto const &[node, cacheKey, parameter] : m_logfileParameters) {
571 parameterInfo->addLogfileParameter(layout.index(node), cacheKey, parameter);
572 }
573
574 auto beamlineComponentInfo = layout.componentInfo();
575 auto beamlineDetectorInfo = layout.detectorInfo();
576 beamlineComponentInfo->setDetectorInfo(beamlineDetectorInfo.get());
577 auto componentInfo = std::make_unique<ComponentInfo>(std::move(beamlineComponentInfo), layout.shapes(), parameterInfo,
578 m_referenceFrame);
579 auto detectorInfo = std::make_unique<DetectorInfo>(std::move(beamlineDetectorInfo), layout.detectorIds(),
580 layout.detectorIdToIndex(), parameterInfo, m_referenceFrame);
581 return makeBeamlineRef(std::move(componentInfo), std::move(detectorInfo), m_metadata);
582}

584BeamlineRef makeBeamlineRef(std::unique_ptr<ComponentInfo> componentInfo, std::unique_ptr<DetectorInfo> detectorInfo,
585 std::shared_ptr<const InstrumentMetadata> metadata) {
586 // The ComponentInfo points at its DetectorInfo, so one owner keeps both
587 struct Infos {
588 std::unique_ptr<ComponentInfo> componentInfo;
589 std::unique_ptr<DetectorInfo> detectorInfo;
590 };
591 auto const infos = std::make_shared<Infos>(std::move(componentInfo), std::move(detectorInfo));
592 return {std::shared_ptr<const ComponentInfo>(infos, infos->componentInfo.get()),
593 std::shared_ptr<const DetectorInfo>(infos, infos->detectorInfo.get()), std::move(metadata)};
594}
