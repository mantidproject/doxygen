59template <typename T>
60std::optional<std::string> distinctValuesInTimeROI(const Kernel::Property *log, const Kernel::TimeROI &roi) {
61 const auto *timeSeries = dynamic_cast<const Kernel::TimeSeriesProperty<T> *>(log);
62 if (!timeSeries) {
63 return std::nullopt;
64 }
65 if (timeSeries->size() == 0) {
66 return std::string();
67 }
68 std::vector<std::string> values;
69 for (const auto &rawValue : timeSeries->filteredValuesAsVector(&roi)) {
70 const auto value = sanitiseHeaderValue(Kernel::Strings::toString(rawValue));
71 if (!value.empty() && (values.empty() || values.back() != value)) {
72 values.emplace_back(value);
73 }
74 }
75 return Kernel::Strings::join(values.cbegin(), values.cend(), ", ");
76}

94std::map<std::string, std::string> SaveSESANS::validateInputs() {
95 std::map<std::string, std::string> invalidInputs;
96
97 for (const auto &propertyName : mandatoryDoubleProperties) {
98 double value = getProperty(propertyName);
99 if (value == EMPTY_DBL()) {
100 invalidInputs[propertyName] = propertyName + " must be set";
101 }
102 }
103
104 if (getPropertyValue("Sample").empty()) {
105 invalidInputs["Sample"] = "Sample must be set";
106 }
107
108 m_sampleThickness = static_cast<double>(getProperty("OverrideSampleThickness"));
109 if (m_sampleThickness <= SaveSESANS::TOLERANCE) {
110 invalidInputs["OverrideSampleThickness"] = "OverrideSampleThickness value must be greater than 0";
111 }
112
113 return invalidInputs;
114}

231Kernel::TimeROI SaveSESANS::getSamplePositionTimeROI(const API::Run &run) const {
232 const auto &roi = run.getTimeROI();
233 if (!roi.useAll()) {
234 return roi;
235 }
236 try {
237 const auto startTime = run.startTime();
238 const auto endTime = run.endTime();
239 if (startTime < endTime) {
240 return Kernel::TimeROI(startTime, endTime);
241 }
242 } catch (const std::runtime_error &) {
243 g_log.debug("The run has no valid start and end times, so the whole SamplePos log is used");
244 }

256std::string SaveSESANS::getSamplePosition(const API::MatrixWorkspace_const_sptr &ws) const {
257 const std::string samplePosition = getPropertyValue("SamplePosition");
258 if (!samplePosition.empty()) {
259 return samplePosition;
260 }
261 const auto &run = ws->run();
262 if (!run.hasProperty("SamplePos")) {
263 return "";
264 }
265 const auto *log = run.getLogData("SamplePos");
266 const auto roi = getSamplePositionTimeROI(run);
267 // IBEX records the position as a string or a number, depending on how the block is configured
268 auto positions = distinctValuesInTimeROI<std::string>(log, roi);
269 if (!positions) {
270 positions = distinctValuesInTimeROI<double>(log, roi);
271 }
272 if (!positions) {
273 positions = distinctValuesInTimeROI<float>(log, roi);
274 }
275 if (!positions) {
276 positions = distinctValuesInTimeROI<int>(log, roi);
277 }
278 if (positions) {
279 return *positions;
280 }
281 if (dynamic_cast<const Kernel::ITimeSeriesProperty *>(log)) {
282 g_log.warning("The SamplePos log has an unsupported type and is not written to the file. Use the SamplePosition "
283 "property to provide the sample position.");
284 return "";
285 }
286 return log->value();
287}

292void SaveSESANS::exec() {
293 API::MatrixWorkspace_const_sptr ws = getProperty("InputWorkspace");
294
295 // Check workspace has only one spectrum
296 if (ws->getNumberHistograms() != 1) {
297 g_log.error("This algorithm expects a workspace with exactly 1 spectrum");
298 throw std::runtime_error("SaveSESANS passed workspace with incorrect "
299 "number of spectra, expected 1");
300 }
301
302 if (m_sampleThickness == EMPTY_DBL()) {
303 m_sampleThickness = ws->sample().getThickness();
304 if (m_sampleThickness <= SaveSESANS::TOLERANCE) {
305 g_log.error("The workspace passed in does not provide the sample thickness. Please use the "
306 "OverrideSampleThickness property to "
307 "provide this value instead");
308 throw std::runtime_error("SaveSESANS passed workspace with sample thickness less than "
309 "or equal to 0 and no value provided for OverrideSampleThickness property");
310 }
311 }
312
313 auto filename = getPropertyValue("Filename");
314 std::ofstream outfile(filename, std::ofstream::trunc);
315 if (outfile.fail()) {
316 const std::string error = strerror(errno);
317 g_log.error("Failed to open file. Error was: " + error);
318 throw std::runtime_error("Could not open file at the following path: " + filename);
319 }
320
321 writeHeaders(outfile, ws);
322 outfile << "\n"
323 << "BEGIN_DATA"
324 << "\n";
325
326 const auto &wavelength = ws->points(0);
327 const auto &yValues = ws->y(0);
328 const auto &eValues = ws->e(0);
329
330 const auto spinEchoLength = calculateSpinEchoLength(wavelength);
331 const auto depolarisation = calculateDepolarisation(yValues, wavelength);
332 const auto error = calculateError(eValues, yValues, wavelength);
333
334 outfile << "SpinEchoLength Depolarisation Depolarisation_error Wavelength\n";
335
336 for (size_t i = 0; i < spinEchoLength.size(); ++i) {
337 outfile << formatDouble(spinEchoLength[i]) << " ";
338 outfile << formatDouble(depolarisation[i]) << " ";
339 outfile << formatDouble(error[i]) << " ";
340 outfile << formatDouble(wavelength[i]) << "\n";
341 }
342
343 outfile.close();
344}

358 writeHeader(outfile, "Theta_zmax_unit", getPropertyValue("ThetaZMaxUnit"));
359 writeHeader(outfile, "Theta_ymax", getPropertyValue("ThetaYMax"));
360 writeHeader(outfile, "Theta_ymax_unit", getPropertyValue("ThetaYMaxUnit"));
361 writeHeader(outfile, "Orientation", getPropertyValue("Orientation"));
362 writeHeader(outfile, "SpinEchoLength_unit", "A");
363 writeHeader(outfile, "Depolarisation_unit", "A-2 cm-1");
364 writeHeader(outfile, "Wavelength_unit", "A");
365 writeHeader(outfile, "Echo_constant", getPropertyValue("EchoConstant"));
366
367 // Optional metadata identifying the measurement. These are only written if

371 writeOptionalHeader(outfile, "Instrument", instrumentName);

418std::vector<double> SaveSESANS::calculateSpinEchoLength(const HistogramData::Points &wavelength) const {
419 std::vector<double> spinEchoLength;
420 const double echoConstant = getProperty("EchoConstant");
421
422 // SEL is calculated as wavelength^2 * echoConstant
423 transform(wavelength.begin(), wavelength.end(), back_inserter(spinEchoLength),
424 [&](double w) { return w * w * echoConstant; });
425 return spinEchoLength;
426}

436std::vector<double> SaveSESANS::calculateDepolarisation(const HistogramData::HistogramY &yValues,
437 const HistogramData::Points &wavelength) const {
438 Mantid::MantidVec depolarisation;
439 // The sample thickness is provided in mm but we need to use cm here
440 double thickness = m_sampleThickness * 0.1;
441
442 // Depol is calculated as ln(y) / wavelength^2 / sample thickness in cm
443 transform(yValues.begin(), yValues.end(), wavelength.begin(), back_inserter(depolarisation),
444 [&](double y, double w) { return log(y) / (w * w) / thickness; });
445 return depolarisation;
446}

456Mantid::MantidVec SaveSESANS::calculateError(const HistogramData::HistogramE &eValues,
457 const HistogramData::HistogramY &yValues,
458 const HistogramData::Points &wavelength) const {
459 Mantid::MantidVec error;
460
461 // The sample thickness is provided in mm but we need to use cm here
462 double thickness = m_sampleThickness * 0.1;
463
464 // Error is calculated as e / (y * wavelength^2) / sample thickness in cm
465 for (size_t i = 0; i < eValues.size(); i++) {
466 error.emplace_back(eValues[i] / (yValues[i] * wavelength[i] * wavelength[i]) / thickness);
467 }
468 return error;
469}
