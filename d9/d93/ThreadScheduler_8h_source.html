148 ~ThreadSchedulerFIFO() override { clear(); }

150 //-------------------------------------------------------------------------------
152 bool empty() override {
153 std::lock_guard<std::mutex> _lock(m_queueLock);
154 return m_queue.empty();
155 }
156
157 //-------------------------------------------------------------------------------
158 void push(std::shared_ptr<Task> newTask) override {
159 // Cache the total cost
160 m_queueLock.lock();
161 m_cost += newTask->cost();
162 m_queue.emplace_back(newTask);
163 m_queueLock.unlock();
164 }
165
166 //-------------------------------------------------------------------------------
167 std::shared_ptr<Task> pop(size_t threadnum) override {
168 UNUSED_ARG(threadnum);
169 std::shared_ptr<Task> temp = nullptr;
170 m_queueLock.lock();
171 // Check the size within the same locking block; otherwise the size may
172 // change before you get the next item.
173 if (!m_queue.empty()) {
174 // TODO: Would a try/catch block be smart here?
175 temp = m_queue.front();
176 m_queue.pop_front();
177 }
178 m_queueLock.unlock();
179 return temp;
180 }
181
182 //-------------------------------------------------------------------------------
183 size_t size() override {
184 m_queueLock.lock();
185 size_t temp = m_queue.size();
186 m_queueLock.unlock();
187 return temp;
188 }
189
190 //-------------------------------------------------------------------------------
191 void clear() override final {
192 m_queueLock.lock();
193 // Empty out the queue
194 m_queue.clear();
195 m_cost = 0;
196 m_costExecuted = 0;
197 m_queueLock.unlock();
198 }
199
200protected:
202 std::deque<std::shared_ptr<Task>> m_queue;

245class MANTID_KERNEL_DLL ThreadSchedulerLargestCost : public ThreadScheduler {
246public:
247 ThreadSchedulerLargestCost() : ThreadScheduler() {}
248
250 ~ThreadSchedulerLargestCost() override { clear(); }
251
252 //-------------------------------------------------------------------------------
254 bool empty() override {
255 std::lock_guard<std::mutex> _lock(m_queueLock);
256 return m_map.empty();
257 }
258
259 //-------------------------------------------------------------------------------
260 void push(std::shared_ptr<Task> newTask) override {
261 // Cache the total cost
262 m_queueLock.lock();
263 m_cost += newTask->cost();
264 m_map.emplace(newTask->cost(), newTask);
265 m_queueLock.unlock();
266 }
267
268 //-------------------------------------------------------------------------------
269 std::shared_ptr<Task> pop(size_t threadnum) override {
270 UNUSED_ARG(threadnum);
271 std::shared_ptr<Task> temp = nullptr;
272 m_queueLock.lock();
273 // Check the size within the same locking block; otherwise the size may
274 // change before you get the next item.
275 if (!m_map.empty()) {
276 // Since the map is sorted by cost, we want the LAST item.
277 auto it = m_map.end();
278 it--;
279 temp = it->second;
280 m_map.erase(it);
281 }
282 m_queueLock.unlock();
283 return temp;
284 }
285
286 //-------------------------------------------------------------------------------
287 size_t size() override {
288 m_queueLock.lock();
289 size_t temp = m_map.size();
290 m_queueLock.unlock();
291 return temp;
292 }
293
294 //-------------------------------------------------------------------------------
295 void clear() override final {
296 m_queueLock.lock();
297 // Empty out the queue and delete the pointers!
298 m_map.clear();
299 m_cost = 0;
300 m_costExecuted = 0;
301 m_queueLock.unlock();
302 }
303
304protected:
306 std::multimap<double, std::shared_ptr<Task>> m_map;
307};
