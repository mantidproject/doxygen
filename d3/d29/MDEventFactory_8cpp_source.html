204API::IMDNode *MDEventFactory::createBox(size_t nDimensions, MDEventFactory::BoxType Type,
205 API::BoxController_sptr &splitter,
206 const std::vector<Mantid::Geometry::MDDimensionExtents<coord_t>> &extentsVector,
207 const uint32_t depth, const size_t nBoxEvents, const size_t boxID) {
208
209 if (nDimensions > MAX_MD_DIMENSIONS_NUM)
210 throw std::invalid_argument(" there are more dimensions requested then instantiated");
211
212 size_t id = nDimensions * MDEventFactory::NumBoxTypes + Type;
213 if (extentsVector.size() != nDimensions) // set defaults
214 {
215 std::vector<Mantid::Geometry::MDDimensionExtents<coord_t>> defaultExtents(nDimensions);
216 for (size_t i = 0; i < nDimensions; i++) {
217 // set to smaller than float max, so the entire range fits in a float.
218 defaultExtents[i].setExtents(-1e30f, 1e30f);
219 }
220 return (*(boxCreatorFP[id]))(splitter.get(), defaultExtents, depth, nBoxEvents, boxID);
221 }
222
223 return (*(boxCreatorFP[id]))(splitter.get(), extentsVector, depth, nBoxEvents, boxID);
224}
