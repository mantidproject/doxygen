41V3D observerOrSample(const SpectrumInfo &self, const object &observer) {
42 return observer.is_none() ? self.samplePosition() : extract<V3D>(observer)();
43}
44
46V3D axisOrBeam(const SpectrumInfo &self, const object &axis) {
47 if (!axis.is_none())
48 return extract<V3D>(axis)();
49 const auto beamLine = self.samplePosition() - self.sourcePosition();
50 if (beamLine.nullVector())
51 throw Mantid::Kernel::Exception::InstrumentDefinitionError("Source and sample are at same position!");
52 return beamLine;
53}

55double twoTheta(const SpectrumInfo &self, const size_t index, const object &observer, const object &axis) {
56 if (observer.is_none() && axis.is_none())
57 return self.twoTheta(index);
58 return self.twoTheta(index, observerOrSample(self, observer), axisOrBeam(self, axis));
59}

61double signedTwoTheta(const SpectrumInfo &self, const size_t index, const object &observer, const object &axis,
62 const object &instrumentUp) {
63 if (observer.is_none() && axis.is_none() && instrumentUp.is_none())
64 return self.signedTwoTheta(index);
65 if (instrumentUp.is_none())
66 return self.signedTwoTheta(index, observerOrSample(self, observer), axisOrBeam(self, axis));
67 return self.signedTwoTheta(index, observerOrSample(self, observer), axisOrBeam(self, axis),
68 extract<V3D>(instrumentUp)());
69}
