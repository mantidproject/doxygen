122void InstrumentRayTracer::fireRay(Track &testRay) const {
123 auto const &compInfo = m_componentInfo;
124 // Go through the instrument tree and see if we get any hits by
125 // (a) first testing the bounding box and if we're inside that then
126 // (b) test the lower components.
127 std::deque<size_t> nodeQueue;
128
129 // Start at the root of the tree
130 nodeQueue.emplace_back(compInfo.root());
131
132 while (!nodeQueue.empty()) {
133 size_t const node = nodeQueue.front();
134 nodeQueue.pop_front();
135 BoundingBox bbox;
136 auto it = m_boxCache.find(node);
137 if (it != m_boxCache.end()) {
138 bbox = it->second;
139 } else {
140 bbox = compInfo.boundingBox(node);
141 std::lock_guard<std::mutex> lock(m_mutex);
142 m_boxCache[node] = bbox;
143 }
144
145 // Quick test. If this suceeds moved on to test the children
146 if (!bbox.doesLineIntersect(testRay))
147 continue;
148 if (compInfo.isGridDetector(node)) {
149 testIntersectionWithGrid(node, testRay);
150 continue;
151 }
152 for (auto const child : compInfo.children(node)) {
153 if (!compInfo.children(child).empty())
154 nodeQueue.emplace_back(child);
155 else if (compInfo.hasValidShape(child))
156 testIntersectionWithShape(child, testRay);
157 }
158 }
159}

167void InstrumentRayTracer::testIntersectionWithShape(size_t const index, Track &testRay) const {
168 auto const &compInfo = m_componentInfo;
169 auto const &shape = compInfo.shape(index);
170 V3D const pos = compInfo.position(index);
171 Kernel::Quat const rot = compInfo.rotation(index);
172 Kernel::Quat unRotate = rot;
173 unRotate.inverse();
174 V3D const scale = compInfo.scaleFactor(index);
175
176 // Transform the track into the frame of the shape. As in
177 // ObjComponent::interceptSurface, scaling is not factored out here.
178 V3D trkStart = testRay.startPoint() - pos;
179 unRotate.rotate(trkStart);
180 V3D trkDirection = testRay.direction();
181 unRotate.rotate(trkDirection);
182
183 Track probeTrack(trkStart, trkDirection);
184 shape.interceptSurface(probeTrack);
185
186 for (auto const &link : probeTrack) {
187 V3D in = link.entryPoint;
188 rot.rotate(in);
189 in *= scale;
190 in += pos;
191 V3D out = link.exitPoint;
192 rot.rotate(out);
193 out *= scale;
194 out += pos;
195 // ComponentID is only used as a handle, so this is never written through
196 testRay.addLink(in, out, out.distance(testRay.startPoint()), shape,
197 const_cast<IComponent *>(compInfo.componentID(index)));
198 }
199}

207void InstrumentRayTracer::testIntersectionWithGrid(size_t const index, Track &testRay) const {
208 auto const &compInfo = m_componentInfo;
209 auto const grid = compInfo.pixelGridComponent(index);
210
212 V3D const basePoint = compInfo.position(compInfo.detectorIndexAtXYZ(index, 0, 0, 0));
213
215 V3D const vertical = compInfo.position(compInfo.detectorIndexAtXYZ(index, 0, grid.nY - 1, 0)) - basePoint;
216
218 V3D const horizontal = compInfo.position(compInfo.detectorIndexAtXYZ(index, grid.nX - 1, 0, 0)) - basePoint;
219
220 // The beam direction
221 V3D const beam = testRay.direction();
222
223 // From: http://en.wikipedia.org/wiki/Line-plane_intersection (taken on May 4,
224 // 2011),
225 // We build a matrix to solve the linear equation:
226 Kernel::Matrix<double> mat(3, 3);
227 mat.setColumn(0, beam * -1.0);
228 mat.setColumn(1, horizontal);
229 mat.setColumn(2, vertical);
230 mat.Invert();
231
232 // Multiply by the inverted matrix to find t,u,v
233 V3D const tuv = mat * (testRay.startPoint() - basePoint);
234
235 // Intersection point
236 V3D const intersec = beam * tuv[0];
237
238 // t = coordinate along the line
239 // u,v = coordinates along horizontal, vertical
240 // (and correct for it being between 0, nX-1). The +0.5 is because the
241 // base point is at the CENTER of pixel 0,0.
242 auto const xIndex = int(double(grid.nX - 1) * tuv[1] + 0.5);
243 auto const yIndex = int(double(grid.nY - 1) * tuv[2] + 0.5);
244
245 // Out of range?
246 if (xIndex < 0 || yIndex < 0 || xIndex >= grid.nX || yIndex >= grid.nY)
247 return;
248
249 size_t const pixel = compInfo.detectorIndexAtXYZ(index, xIndex, yIndex, 0);
250 // ComponentID is only used as a handle, so this is never written through
251 testRay.addLink(intersec, intersec, 0.0, compInfo.shape(pixel),
252 const_cast<IComponent *>(compInfo.componentID(pixel)));
253}
