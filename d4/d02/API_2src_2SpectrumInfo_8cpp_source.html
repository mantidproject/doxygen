173double SpectrumInfo::phi(const size_t index) const {
174 const auto &spectrumDef = checkAndGetSpectrumDefinition(index);
175 if (spectrumDef.size() == 1)
176 return m_detectorInfo.phi(spectrumDef[0]);
177 auto sumPos = std::accumulate(spectrumDef.cbegin(), spectrumDef.cend(), Kernel::V3D(),
178 [this](const auto &x, const std::pair<size_t, size_t> &detIndex) {
179 return x + m_detectorInfo.position(detIndex);
180 });
181 // Match DetectorGroup::getPos: tiny components should be zero, otherwise atan2 can flip between +pi and -pi
182 if (std::abs(sumPos[0]) < Kernel::Tolerance)
183 sumPos[0] = 0.0;
184 if (std::abs(sumPos[1]) < Kernel::Tolerance)
185 sumPos[1] = 0.0;
186 // Dividing by the detector count does not change the angle, so atan2 can use the sum directly
187 return std::atan2(sumPos[1], sumPos[0]);
188}

233UnitParametersMap SpectrumInfo::diffractometerConstants(const size_t index, std::vector<detid_t> &warningDets) const {
234 if (m_detectorInfo.isScanning()) {
235 throw std::runtime_error("Retrieval of diffractometer constants not "
236 "implemented for scanning instrument");
237 }
238 const auto &spectrumDef = checkAndGetSpectrumDefinition(index);
239 std::vector<size_t> detectorIndicesOnly;
240 std::vector<detid_t> calibratedDets;
241 std::vector<detid_t> uncalibratedDets;
242 std::transform(spectrumDef.cbegin(), spectrumDef.cend(), std::back_inserter(detectorIndicesOnly),
243 [](auto const &pair) { return pair.first; });
244 double difa{0.}, difc{0.}, tzero{0.};
245 for (const auto &detIndex : detectorIndicesOnly) {
246 const auto newDiffConstants = m_detectorInfo.diffractometerConstants(detIndex, calibratedDets, uncalibratedDets);
247 difa += std::get<0>(newDiffConstants);
248 difc += std::get<1>(newDiffConstants);
249 tzero += std::get<2>(newDiffConstants);
250 }
251
252 if (calibratedDets.size() > 0 && uncalibratedDets.size() > 0) {
253 warningDets.insert(warningDets.end(), uncalibratedDets.begin(), uncalibratedDets.end());
254 };
255 // if no calibration is found then return difc only based on the average
256 // of the detector L2 and twoThetas.
257 if (calibratedDets.size() == 0) {
258 return {{UnitParams::difc, difcUncalibrated(index)}};
259 }
260 const auto specDefSize = static_cast<double>(spectrumDefinition(index).size());
261 return {{UnitParams::difa, difa / specDefSize},
262 {UnitParams::difc, difc / specDefSize},
263 {UnitParams::tzero, tzero / specDefSize}};
264}

282double SpectrumInfo::difcUncalibrated(const size_t index) const {
283 // calculate difc based on the average of the detector L2 and twoThetas.
284 // This will be different to the average of the per detector difcs. This is
285 // for backwards compatibility because Mantid always used to calculate
286 // spectrum level difc's this way
287 return 1. / Kernel::Units::tofToDSpacingFactor(l1(), l2(index), twoTheta(index), 0.);
288}

301void SpectrumInfo::getDetectorValues(const Kernel::Unit &inputUnit, const Kernel::Unit &outputUnit,
302 const Kernel::DeltaEMode::Type emode, const bool signedTheta, int64_t wsIndex,
303 UnitParametersMap &pmap) const {
304 if (!hasDetectors(wsIndex))
305 return;
306 pmap[UnitParams::l2] = l2(wsIndex);
307
308 if (!isMonitor(wsIndex)) {
309 // The scattering angle for this detector (in radians).
310 try {
311 if (signedTheta)
312 pmap[UnitParams::twoTheta] = signedTwoTheta(wsIndex);
313 else
314 pmap[UnitParams::twoTheta] = twoTheta(wsIndex);
315 } catch (const std::runtime_error &e) {
316 g_log.warning(e.what());
317 }
318 if (emode != Kernel::DeltaEMode::Elastic && pmap.find(UnitParams::efixed) == pmap.end()) {
319 std::shared_ptr<const Geometry::IDetector> det(&detector(wsIndex), Mantid::NoDeleting());
320 try {
321 pmap[UnitParams::efixed] = m_experimentInfo.getEFixedGivenEMode(det, emode);
322 g_log.debug() << "Detector: " << det->getID() << " EFixed: " << pmap[UnitParams::efixed] << "\n";
323 } catch (std::runtime_error &) {
324 // let the unit classes work out if this is a problem
325 }
326 }
327
328 try {
329 std::set<std::string> diffConstUnits = {"dSpacing", "MomentumTransfer", "Empty"};
330 if ((emode == Kernel::DeltaEMode::Elastic) &&
331 (diffConstUnits.count(inputUnit.unitID()) || diffConstUnits.count(outputUnit.unitID()))) {
332 std::vector<detid_t> warnDetIds;
333 auto diffConstsMap = diffractometerConstants(wsIndex, warnDetIds);
334 pmap.insert(diffConstsMap.begin(), diffConstsMap.end());
335 if (warnDetIds.size() > 0) {
336 createDetectorIdLogMessages(warnDetIds, wsIndex);
337 }
338 } else {
339 pmap[UnitParams::difc] = difcUncalibrated(wsIndex);
340 }
341 } catch (const std::runtime_error &e) {
342 g_log.warning(e.what());
343 }
344 } else {
345 pmap[UnitParams::twoTheta] = 0.0;
346 pmap[UnitParams::efixed] = std::numeric_limits<double>::min();
347 // Energy transfer is meaningless for a monitor, so set l2 to 0.
348 if (outputUnit.unitID().find("DeltaE") != std::string::npos) {
349 pmap[UnitParams::l2] = 0.0;
350 }
351 pmap[UnitParams::difc] = 0;
352 }
353}

355void SpectrumInfo::createDetectorIdLogMessages(const std::vector<detid_t> &detids, int64_t wsIndex) const {
356 std::string detIDstring;
357 auto iter = detids.begin();
358 auto itEnd = detids.end();
359 for (; iter != itEnd; ++iter) {

413const Geometry::IDetector &SpectrumInfo::getDetector(const size_t index) const {
414 auto thread = static_cast<size_t>(PARALLEL_THREAD_NUMBER);
415 if (m_lastIndex[thread] == index)
416 return *m_lastDetector[thread];
417
418 // Note: This function body has big overlap with the method
419 // MatrixWorkspace::getDetector(). The plan is to eventually remove the
420 // latter, once SpectrumInfo is in widespread use.
421 const auto &specDef = spectrumDefinition(index);
422 const size_t ndets = specDef.size();
423 if (ndets == 1) {
424 // If only 1 detector for the spectrum number, just return it
425 const auto detIndex = specDef[0].first;
426 GNU_DIAG_OFF("deprecated-declarations")
427 MSVC_DIAG_OFF(4996)
428 m_lastDetector[thread] = m_detectorInfo.getDetectorPtr(detIndex);
429 MSVC_DIAG_ON(4996)
430 GNU_DIAG_ON("deprecated-declarations")
431 } else if (ndets == 0) {
432 throw Kernel::Exception::NotFoundError("MatrixWorkspace::getDetector(): No "
433 "detectors for this workspace "
434 "index.",
435 std::to_string(index));
436 } else {
437 // Else need to construct a DetectorGroup and use that
438 std::vector<std::shared_ptr<const Geometry::IDetector>> det_ptrs;
439 for (const auto &specDefIndex : specDef) {
440 const auto detIndex = specDefIndex.first;
441 GNU_DIAG_OFF("deprecated-declarations")
442 MSVC_DIAG_OFF(4996)
443 det_ptrs.emplace_back(m_detectorInfo.getDetectorPtr(detIndex));
444 MSVC_DIAG_ON(4996)
445 GNU_DIAG_ON("deprecated-declarations")
446 }
447 m_lastDetector[thread] = std::make_shared<Geometry::DetectorGroup>(det_ptrs);
448 }
449 m_lastIndex[thread] = index;
450 return *m_lastDetector[thread];
451}
