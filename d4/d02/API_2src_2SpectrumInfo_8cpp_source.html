170double SpectrumInfo::phi(const size_t index) const {
171 const auto &spectrumDef = checkAndGetSpectrumDefinition(index);
172 if (spectrumDef.size() == 1)
173 return m_detectorInfo.phi(spectrumDef[0]);
174 auto sumPos = std::accumulate(spectrumDef.cbegin(), spectrumDef.cend(), Kernel::V3D(),
175 [this](const auto &x, const std::pair<size_t, size_t> &detIndex) {
176 return x + m_detectorInfo.position(detIndex);
177 });
178 // Match DetectorGroup::getPos: tiny components should be zero, otherwise atan2 can flip between +pi and -pi
179 if (std::abs(sumPos[0]) < Kernel::Tolerance)
180 sumPos[0] = 0.0;
181 if (std::abs(sumPos[1]) < Kernel::Tolerance)
182 sumPos[1] = 0.0;
183 // Dividing by the detector count does not change the angle, so atan2 can use the sum directly
184 return std::atan2(sumPos[1], sumPos[0]);
185}

230UnitParametersMap SpectrumInfo::diffractometerConstants(const size_t index, std::vector<detid_t> &warningDets) const {
231 if (m_detectorInfo.isScanning()) {
232 throw std::runtime_error("Retrieval of diffractometer constants not "
233 "implemented for scanning instrument");
234 }
235 const auto &spectrumDef = checkAndGetSpectrumDefinition(index);
236 std::vector<size_t> detectorIndicesOnly;
237 std::vector<detid_t> calibratedDets;
238 std::vector<detid_t> uncalibratedDets;
239 std::transform(spectrumDef.cbegin(), spectrumDef.cend(), std::back_inserter(detectorIndicesOnly),
240 [](auto const &pair) { return pair.first; });
241 double difa{0.}, difc{0.}, tzero{0.};
242 for (const auto &detIndex : detectorIndicesOnly) {
243 const auto newDiffConstants = m_detectorInfo.diffractometerConstants(detIndex, calibratedDets, uncalibratedDets);
244 difa += std::get<0>(newDiffConstants);
245 difc += std::get<1>(newDiffConstants);
246 tzero += std::get<2>(newDiffConstants);
247 }
248
249 if (calibratedDets.size() > 0 && uncalibratedDets.size() > 0) {
250 warningDets.insert(warningDets.end(), uncalibratedDets.begin(), uncalibratedDets.end());
251 };
252 // if no calibration is found then return difc only based on the average
253 // of the detector L2 and twoThetas.
254 if (calibratedDets.size() == 0) {
255 return {{UnitParams::difc, difcUncalibrated(index)}};
256 }
257 const auto specDefSize = static_cast<double>(spectrumDefinition(index).size());
258 return {{UnitParams::difa, difa / specDefSize},
259 {UnitParams::difc, difc / specDefSize},
260 {UnitParams::tzero, tzero / specDefSize}};
261}

279double SpectrumInfo::difcUncalibrated(const size_t index) const {
280 // calculate difc based on the average of the detector L2 and twoThetas.
281 // This will be different to the average of the per detector difcs. This is
282 // for backwards compatibility because Mantid always used to calculate
283 // spectrum level difc's this way
284 return 1. / Kernel::Units::tofToDSpacingFactor(l1(), l2(index), twoTheta(index), 0.);
285}

298void SpectrumInfo::getDetectorValues(const Kernel::Unit &inputUnit, const Kernel::Unit &outputUnit,
299 const Kernel::DeltaEMode::Type emode, const bool signedTheta, int64_t wsIndex,
300 UnitParametersMap &pmap) const {
301 if (!hasDetectors(wsIndex))
302 return;
303 pmap[UnitParams::l2] = l2(wsIndex);
304
305 if (!isMonitor(wsIndex)) {
306 // The scattering angle for this detector (in radians).
307 try {
308 if (signedTheta)
309 pmap[UnitParams::twoTheta] = signedTwoTheta(wsIndex);
310 else
311 pmap[UnitParams::twoTheta] = twoTheta(wsIndex);
312 } catch (const std::runtime_error &e) {
313 g_log.warning(e.what());
314 }
315 if (emode != Kernel::DeltaEMode::Elastic && pmap.find(UnitParams::efixed) == pmap.end()) {
316 std::shared_ptr<const Geometry::IDetector> det(&detector(wsIndex), Mantid::NoDeleting());
317 try {
318 pmap[UnitParams::efixed] = m_experimentInfo.getEFixedGivenEMode(det, emode);
319 g_log.debug() << "Detector: " << det->getID() << " EFixed: " << pmap[UnitParams::efixed] << "\n";
320 } catch (std::runtime_error &) {
321 // let the unit classes work out if this is a problem
322 }
323 }
324
325 try {
326 std::set<std::string> diffConstUnits = {"dSpacing", "MomentumTransfer", "Empty"};
327 if ((emode == Kernel::DeltaEMode::Elastic) &&
328 (diffConstUnits.count(inputUnit.unitID()) || diffConstUnits.count(outputUnit.unitID()))) {
329 std::vector<detid_t> warnDetIds;
330 auto diffConstsMap = diffractometerConstants(wsIndex, warnDetIds);
331 pmap.insert(diffConstsMap.begin(), diffConstsMap.end());
332 if (warnDetIds.size() > 0) {
333 createDetectorIdLogMessages(warnDetIds, wsIndex);
334 }
335 } else {
336 pmap[UnitParams::difc] = difcUncalibrated(wsIndex);
337 }
338 } catch (const std::runtime_error &e) {
339 g_log.warning(e.what());
340 }
341 } else {
342 pmap[UnitParams::twoTheta] = 0.0;
343 pmap[UnitParams::efixed] = std::numeric_limits<double>::min();
344 // Energy transfer is meaningless for a monitor, so set l2 to 0.
345 if (outputUnit.unitID().find("DeltaE") != std::string::npos) {
346 pmap[UnitParams::l2] = 0.0;
347 }
348 pmap[UnitParams::difc] = 0;
349 }
350}

352void SpectrumInfo::createDetectorIdLogMessages(const std::vector<detid_t> &detids, int64_t wsIndex) const {
353 std::string detIDstring;
354 auto iter = detids.begin();
355 auto itEnd = detids.end();
356 for (; iter != itEnd; ++iter) {
357 detIDstring += std::to_string(*iter) + ",";

361 detIDstring.pop_back(); // Drop last comma

404const Geometry::IDetector &SpectrumInfo::getDetector(const size_t index) const {
405 auto thread = static_cast<size_t>(PARALLEL_THREAD_NUMBER);
406 if (m_lastIndex[thread] == index)
407 return *m_lastDetector[thread];
408
409 // Note: This function body has big overlap with the method
410 // MatrixWorkspace::getDetector(). The plan is to eventually remove the
411 // latter, once SpectrumInfo is in widespread use.
412 const auto &specDef = spectrumDefinition(index);
413 const size_t ndets = specDef.size();
414 if (ndets == 1) {
415 // If only 1 detector for the spectrum number, just return it
416 const auto detIndex = specDef[0].first;
417 m_lastDetector[thread] = m_detectorInfo.getDetectorPtr(detIndex);
418 } else if (ndets == 0) {
419 throw Kernel::Exception::NotFoundError("MatrixWorkspace::getDetector(): No "
420 "detectors for this workspace "
421 "index.",
422 std::to_string(index));
423 } else {
424 // Else need to construct a DetectorGroup and use that
425 std::vector<std::shared_ptr<const Geometry::IDetector>> det_ptrs;
426 for (const auto &specDefIndex : specDef) {
427 const auto detIndex = specDefIndex.first;
428 det_ptrs.emplace_back(m_detectorInfo.getDetectorPtr(detIndex));
429 }
430 m_lastDetector[thread] = std::make_shared<Geometry::DetectorGroup>(det_ptrs);
431 }
432 m_lastIndex[thread] = index;
433 return *m_lastDetector[thread];
434}
