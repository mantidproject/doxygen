96std::string suggestWorkspaceName(const std::vector<std::string> &fileNames) {
97 Parser parser;
98 RunRangeList runs;
99
100 // For each file name, parse the run number out of it, and add it to a
101 // RunRangeList.
102 for (const auto &fileName : fileNames) {
103 parser.parse(fileName);
104 runs.addRun(parser.runs()[0][0]);
105 }
106
107 // Return the suggested ws name.
108 return parser.instString() + parser.underscoreString() + toString(runs);
109}

121bool ReverseCaselessCompare::operator()(const std::string &a, const std::string &b) const {
122 std::string lowerA;
123 lowerA.resize(a.size());
124 std::string lowerB;
125 lowerB.resize(b.size());
126
127 std::transform(a.cbegin(), a.cend(), lowerA.begin(), tolower);
128 std::transform(b.cbegin(), b.cend(), lowerB.begin(), tolower);
129
130 return lowerA > lowerB;

138Parser::Parser()
139 : m_runs(), m_fileNames(), m_multiFileName(), m_dirString(), m_instString(), m_underscoreString(), m_runString(),
140 m_extString(), m_validInstNames(), m_compoundLeadExtensions(), m_trimWhiteSpaces(true) {
141 const ConfigServiceImpl &config = ConfigService::Instance();
142
143 const auto facilities = config.getFacilities();
144 for (const auto facility : facilities) {
145 for (const auto &instrument : facility->instruments()) {
146 m_validInstNames.insert(instrument.name());
147 m_validInstNames.insert(instrument.shortName());
148 }
149
150 // Derive compound-extension leads from FileExtensions. For a two-component
151 // entry like ".nxs.h5" the lead is ".nxs"; the parser uses this set in
152 // split() to keep the inner extension out of the file stem. Entries with
153 // a single dot (e.g. ".nxs", "_event.nxs") or more than two dots are
154 // skipped — the parser only models two-component compounds.
155 for (const auto &ext : facility->extensions()) {
156 if (std::count(ext.cbegin(), ext.cend(), '.') != 2)
157 continue;
158 auto lead = ext.substr(0, ext.find_last_of('.'));
159 if (!lead.empty() && lead.front() == '.')
160 m_compoundLeadExtensions.insert(std::move(lead));
161 }
162 }
163}

172void Parser::parse(const std::string &multiFileName) {
173 // Clear any contents of the member variables.
174 clear();
175
176 // Set the string to parse.
177 m_multiFileName = multiFileName;
178
179 // Split the string to be parsed into sections, and do some validation.
180 split();
181
182 // Parse the run section into unsigned ints we can use.
183 m_runs = parseMultiRunString(m_runString);
184
185 // Set up helper functor.
186 GenerateFileName generateFileName(m_dirString, m_extString, m_instString);
187
188 // Generate complete file names for each run using helper functor.
189 std::transform(m_runs.begin(), m_runs.end(), std::back_inserter(m_fileNames), generateFileName);
190}

203std::vector<std::vector<unsigned int>> Parser::parseMultiRunString(std::string runString) {
204 // If the run string is empty, return no runs.
205 if (runString.empty())
206 return std::vector<std::vector<unsigned int>>();
207
208 // Remove whitespaces if requested.
209 if (trimWhiteSpaces()) {
210 runString.erase(std::remove_if( // ("Erase-remove" idiom.)
211 runString.begin(), runString.end(), isspace),
212 runString.end());
213 }
214 // Only numeric characters, or occurances of plus, minus, comma and colon are
215 // allowed.
216 if (!matchesFully(runString, "([0-9]|\\+|\\-|,|:)+")) {
217 throw std::runtime_error("Non-numeric or otherwise unaccetable character(s) detected.");
218 }
219
220 // Tokenize on commas.
221 std::vector<std::string> tokens;
222 tokens = boost::split(tokens, runString, boost::is_any_of(","));
223
224 // Validate each token.
225 std::for_each(tokens.begin(), tokens.end(), validateToken);
226
227 // Parse each token, accumulate the results, and return them.
228 std::vector<std::vector<unsigned int>> runGroups;
229 for (auto const &token : tokens) {
230 parseToken(runGroups, token);
231 }
232 return runGroups;
233}

276void Parser::split() {
277 if (m_multiFileName.empty())
278 throw std::runtime_error("No file name to parse.");
279
280 // Clear whitespace before getting extentions and directories, if requested.
281 if (trimWhiteSpaces()) {
282 m_multiFileName.erase(std::remove_if( // ("Erase-remove" idiom.)
283 m_multiFileName.begin(), m_multiFileName.end(), isspace),
284 m_multiFileName.end());
285 }
286 // Get the directory, if there is one.
287 const size_t lastSeparator = m_multiFileName.find_last_of("/\\");
288 if (lastSeparator != std::string::npos)
289 m_dirString = m_multiFileName.substr(0, lastSeparator + 1);
290
291 // Get the extension. Compound extensions like ".nxs.h5" are kept whole when
292 // the leading half (".nxs") matches a known compound lead derived from the
293 // facility FileExtensions in Facilities.xml. Otherwise just the final dot's
294 // suffix is taken. This avoids misidentifying stems that contain dots
295 // (e.g. "run.v2.nxs") as compound-extension files.
296 const size_t lastDot = m_multiFileName.find_last_of('.');
297 if (lastDot != std::string::npos && lastDot > m_dirString.size()) {
298 const size_t prevDot = m_multiFileName.find_last_of('.', lastDot - 1);
299 if (prevDot != std::string::npos && prevDot >= m_dirString.size()) {
300 const auto midExt = m_multiFileName.substr(prevDot, lastDot - prevDot);
301 if (m_compoundLeadExtensions.contains(midExt))
302 m_extString = m_multiFileName.substr(prevDot);
303 }
304 if (m_extString.empty())
305 m_extString = m_multiFileName.substr(lastDot);
306 }
307
308 // If the directory contains an instance of a comma, then the string is
309 // a comma separated list of single *full* file names to load.
310 if (std::string::npos != m_dirString.find(','))
311 throw std::runtime_error("Unable to parse.");
312
313 // Slice off the directory and extension.
314 std::string base =
315 m_multiFileName.substr(m_dirString.size(), m_multiFileName.size() - (m_dirString.size() + m_extString.size()));
316
317 if (base.empty())
318 throw std::runtime_error("There does not appear to be any runs present.");
319
320 auto instrumentNameIt = std::find_if(m_validInstNames.cbegin(), m_validInstNames.cend(), [&base](const auto &name) {
321 return matchesFully(base, name + ".*", true);
322 }); // USE CASELESS MATCHES HERE.
323
324 // See if the user has typed in one of the available instrument names.
325 if (instrumentNameIt != m_validInstNames.cend()) {
326 m_instString = getMatchingString("^" + *instrumentNameIt, base, true);
327 }
328
329 // If not, use the default, or throw if we encounter an unrecognisable
330 // non-numeric string.
331 if (m_instString.empty()) {
332 if (base.empty())
333 throw std::runtime_error("There does not appear to be any runs present.");
334
335 if (isdigit(base[0]))
336 m_instString = ConfigService::Instance().getString("default.instrument");
337 else
338 throw std::runtime_error("There does not appear to be a valid instrument name present.");
339 } else {
340 // Chop off instrument name.
341 base = base.substr(m_instString.size(), base.size());
342 }
343
344 if (base.empty())
345 throw std::runtime_error("There does not appear to be any runs present.");
346
347 const auto &instInfo = ConfigService::Instance().getInstrument(m_instString);
348 // why?
349 // m_instString = instInfo.shortName(); // Make sure we're using the shortened
350 // form of the isntrument name.
351
352 if (base.starts_with(instInfo.delimiter())) {
353 // Store the instrument delimiter, and strip it off the start of the string.
354 m_underscoreString = instInfo.delimiter();
355 base = base.substr(m_underscoreString.size(), base.size());
356 }
357
358 m_runString = getMatchingString("^" + Regexs::LIST, base);
359
360 if (m_runString.size() != base.size()) {
361 throw std::runtime_error("There is an unparsable token present.");
362 }
363}

426void RunRangeList::addRun(const unsigned int run) {
427 // If the run is inside one of the ranges, do nothing.
428 if (std::binary_search(m_rangeList.begin(), m_rangeList.end(), run, RangeContainsRun()))
429 return;
430
431 // Else create a new range, containing a single run, and add it to the list.
432 m_rangeList.emplace(run, run);
433
434 // Now merge any ranges that are adjacent.
435 m_rangeList = std::accumulate(m_rangeList.begin(), m_rangeList.end(),
436 std::set<std::pair<unsigned int, unsigned int>>(), mergeAdjacentRanges);
437}

472void parseToken(std::vector<std::vector<unsigned int>> &parsedRuns, const std::string &token) {
473 std::vector<std::vector<unsigned int>> runs;
474 // Tokenise further on plus.
475 std::vector<std::string> subTokens;
476 boost::split(subTokens, token, boost::is_any_of("+"));
477 std::vector<unsigned int> runsToAdd;
478 for (auto const &subToken : subTokens) {
479 // E.g. "2012".
480 if (matchesFully(subToken, Regexs::SINGLE)) {
481 runsToAdd.emplace_back(std::stoi(subToken));
482 }
483 // E.g. "2012:2020".
484 else if (matchesFully(subToken, Regexs::RANGE)) {
485 // Fill in runs directly.
486 constexpr bool addRuns{false};
487 std::vector<std::string> rangeDetails;
488 rangeDetails.reserve(2);
489 boost::split(rangeDetails, subToken, boost::is_any_of(":"));
490 runs = generateRange(std::stoi(rangeDetails.front()), std::stoi(rangeDetails.back()), 1, addRuns);
491 }
492 // E.g. "2012:2020:4".
493 else if (matchesFully(subToken, Regexs::STEP_RANGE)) {
494 // Fill in runs directly.
495 constexpr bool addRuns{false};
496 std::vector<std::string> rangeDetails;
497 rangeDetails.reserve(3);
498 boost::split(rangeDetails, subToken, boost::is_any_of(":"));
499 runs = generateRange(std::stoi(rangeDetails[0]), std::stoi(rangeDetails[1]), std::stoi(rangeDetails[2]), addRuns);
500 }
501 // E.g. "2012-2020".
502 else if (matchesFully(subToken, Regexs::ADD_RANGE)) {
503 constexpr bool addRuns{true};
504 std::vector<std::string> rangeDetails;
505 rangeDetails.reserve(2);
506 boost::split(rangeDetails, subToken, boost::is_any_of("-"));
507 const auto generated = generateRange(std::stoi(rangeDetails.front()), std::stoi(rangeDetails.back()), 1, addRuns);
508 std::copy(generated.front().cbegin(), generated.front().cend(), back_inserter(runsToAdd));
509 }
510 // E.g. "2012-2020:4".
511 else if (matchesFully(subToken, Regexs::ADD_STEP_RANGE)) {
512 constexpr bool addRuns{true};
513 std::vector<std::string> rangeDetails;
514 rangeDetails.reserve(3);
515 boost::split(rangeDetails, subToken, boost::is_any_of("-:"));
516 const auto generated =
517 generateRange(std::stoi(rangeDetails[0]), std::stoi(rangeDetails[1]), std::stoi(rangeDetails[2]), addRuns);
518 std::copy(generated.front().cbegin(), generated.front().cend(), back_inserter(runsToAdd));
519 } else {
520 // We should never reach here - the validation done on the token
521 // previously should prevent any other possible scenario.
522 assert(false);
523 }
524 }
525 if (!runsToAdd.empty()) {
526 if (!runs.empty()) {
527 // We have either add ranges or step ranges. Never both.
528 throw std::runtime_error("Unable to handle a mixture of add ranges and step ranges");
529 }
530 runs.emplace_back(runsToAdd);
531 }
532 // Add the runs on to the end of parsedRuns, and return it.
533 std::copy(runs.begin(), runs.end(), std::back_inserter(parsedRuns));
534}

551std::vector<std::vector<unsigned int>> generateRange(unsigned int const from, unsigned int const to,
552 unsigned int const stepSize, bool const addRuns) {
553 if (stepSize == 0)
554 throw std::runtime_error("Unable to generate a range with a step size of zero.");
555
556 size_t limit;
557 auto limitStr = ConfigService::Instance().getValue<std::string>("loading.multifilelimit");
558 if (!limitStr.has_value() || !Strings::convert(limitStr.value(), limit)) {
559 limit = ConfigService::Instance().getFacility().multiFileLimit();
560 }
561
562 unsigned int const orderedTo = from > to ? from : to;
563 unsigned int const orderedFrom = from > to ? to : from;
564 unsigned int const numberOfFiles = (orderedTo - orderedFrom) / stepSize;
565 if (numberOfFiles > limit) {
566 std::stringstream sstream;
567 sstream << "The range from " << orderedFrom << " to " << orderedTo << " with step " << stepSize
568 << " would generate " << numberOfFiles << " files. "
569 << "This is greater than the current limit of " << limit << ". "
570 << "This limit can be configured in the Mantid.user.properties "
571 "file using the key loading.multifilelimit=200.";
572 throw std::range_error(sstream.str());
573 }
574
575 unsigned int currentRun = from;
576 std::vector<std::vector<unsigned int>> runs;
577
578 // If ascending range
579 if (from <= to) {
580 while (currentRun <= to) {
581 if (addRuns) {
582 if (runs.empty())
583 runs.emplace_back(1, currentRun);
584 else
585 runs.front().emplace_back(currentRun);
586 } else {
587 runs.emplace_back(1, currentRun);
588 }
589
590 currentRun += stepSize;
591 }
592 }
593 // Else descending range
594 else {
595 while (currentRun >= to) {
596 if (addRuns) {
597 if (runs.empty())
598 runs.emplace_back(1, currentRun);
599 else
600 runs.front().emplace_back(currentRun);
601 } else {
602 runs.emplace_back(1, currentRun);
603 }
604
605 // Guard against case where stepSize would take us into negative
606 // numbers (not supported by unsigned ints ...).
607 if (static_cast<int>(currentRun) - static_cast<int>(stepSize) < 0)
608 break;
609
610 currentRun -= stepSize;
611 }
612 }
613
614 return runs;
615}

624void validateToken(const std::string &token) {
625 // Each token must be non-empty.
626 if (token.empty())
627 throw std::runtime_error("A comma-separated token is empty.");
628
629 // Each token must begin and end with a numeric character.
630 if (!matchesFully(token, "[0-9].+[0-9]|[0-9]"))
631 throw std::runtime_error("The token \"" + token +
632 "\" is of an incorrect form. Does it begin or "
633 "end with a plus, minus or colon?");
634
635 // Each token must be one of the acceptable forms, i.e. a single run, an added
636 // range of runs, etc.
637 if (!matchesFully(token, Regexs::ANY))
638 throw std::runtime_error("The token \"" + token + "\" is of an incorrect form.");
639}

650bool matchesFully(const std::string &stringToMatch, const std::string &regexString, const bool caseless) {
651 const auto flags = caseless ? std::regex::ECMAScript | std::regex::icase : std::regex::ECMAScript;
652 const std::regex regex("^(" + regexString + "$)", flags);
653 return std::regex_match(stringToMatch, regex);

665std::string getMatchingString(const std::string &regexString, const std::string &toParse, const bool caseless) {
666 const auto flags = caseless ? std::regex::ECMAScript | std::regex::icase : std::regex::ECMAScript;
667 const std::regex regex(regexString, flags);
668
669 std::sregex_iterator it(toParse.begin(), toParse.end(), regex);
670 if (it == std::sregex_iterator())
671 return "";
672
673 return it->str();

685std::string pad(const unsigned int run, const std::string &instString) {
686 InstrumentInfo const instInfo = ConfigService::Instance().getInstrument(instString);
687 std::string prefix;
688 if (!instInfo.facility().noFilePrefix())
689 prefix = instInfo.filePrefix(run) + instInfo.delimiter();
690 unsigned int padLength = instInfo.zeroPadding(run);
691 std::string runStr = std::to_string(run);
692 if (runStr.size() < padLength)
693 runStr.insert(0, padLength - runStr.size(), '0');
694 else if (padLength > 0 && runStr.size() > padLength)
695 throw std::runtime_error("Could not parse run number \"" + runStr +
696 "\" since the instrument run number length required is " + std::to_string(padLength));
697 runStr.insert(0, prefix);
698 return runStr;
699}

724std::set<std::pair<unsigned int, unsigned int>>
725mergeAdjacentRanges(std::set<std::pair<unsigned int, unsigned int>> ranges,
726 const std::pair<unsigned int, unsigned int> &range) {
727 // If ranges is empty, just insert the new range.
728 if (ranges.empty()) {
729 ranges.insert(range);
730 }
731 // Else there are already some ranges present ...
732 else {
733 // ... if the last one is adjacent to the new range, merge the two.
734 if (ranges.rbegin()->second + 1 == range.first) {
735 unsigned int from = ranges.rbegin()->first;
736 unsigned int to = range.second;
737 std::pair<unsigned int, unsigned int> temp(from, to);
738
739 ranges.erase(--ranges.end(), ranges.end());
740 ranges.insert(temp);
741 }
742 // ... else just insert it.
743 else {
744 ranges.insert(range);
745 }
746 }
747 return ranges;
748}
