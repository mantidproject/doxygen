58template <typename T>
59std::map<std::string, T> getSettingsAsMap(QSettings const &settings, std::string const &settingGroup) {
60 std::map<std::string, T> settingsMap;
61 auto groupPrefix = QString::fromStdString(settingGroup);
62 if (!groupPrefix.isEmpty() && !groupPrefix.endsWith('/'))
63 groupPrefix.append('/');
64 std::string templateTypeName = typeid(T).name();
65 for (auto const &qualifiedName : settings.allKeys()) {
66 if (qualifiedName.startsWith(groupPrefix) && qualifiedName.endsWith("/type")) {
67 std::string settingTypeName = settings.value(qualifiedName).toString().toStdString();
68 if (settingTypeName == templateTypeName) {
69 auto settingName = qualifiedName.mid(groupPrefix.size());
70 settingName.chop(5);
71 auto setting = settings.value(groupPrefix + settingName + "/value");
72 settingsMap[settingName.toStdString()] = setting.template value<T>();
73 }
74 }
75 }
76 return settingsMap;
77}
