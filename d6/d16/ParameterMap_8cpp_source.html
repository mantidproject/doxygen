197bool ParameterMap::relErr(double x1, double x2, double errorVal) const {
198 double num = std::fabs(x1 - x2);
199 // how to treat x1<0 and x2 > 0 ? probably this way
200 double den = 0.5 * (std::fabs(x1) + std::fabs(x2));
201 if (den < errorVal)
202 return (num > errorVal);
203
204 return (num / den > errorVal);
205}

221const std::string ParameterMap::diff(const ParameterMap &rhs, const bool &firstDiffOnly, const bool relative,
222 const double doubleTolerance) const {
223 if (this == &rhs)
224 return std::string(""); // True for the same object
225
226 // Quick size check
227 if (this->size() != rhs.size()) {
228 return std::string("Number of parameters does not match: ") + std::to_string(this->size()) + " not equal to " +
229 std::to_string(rhs.size());
230 }
231
232 // The map is unordered and the key is only valid at runtime. The
233 // asString method turns the ComponentIDs to full-qualified name identifiers
234 // so we will use the same approach to compare them
235
236 std::unordered_multimap<std::string, Parameter_sptr> thisMap, rhsMap;
237 for (auto &mappair : this->m_map) {
238 thisMap.emplace(mappair.first->getFullName(), mappair.second);
239 }
240 for (auto &mappair : rhs.m_map) {
241 rhsMap.emplace(mappair.first->getFullName(), mappair.second);
242 }
243
244 std::stringstream strOutput;
245 for (auto thisIt = thisMap.cbegin(); thisIt != thisMap.cend(); ++thisIt) {
246 const std::string fullName = thisIt->first;
247 const auto &param = thisIt->second;
248 bool match(false);
249 for (auto rhsIt = rhsMap.cbegin(); rhsIt != rhsMap.cend(); ++rhsIt) {
250 const std::string rhsFullName = rhsIt->first;
251 const auto &rhsParam = rhsIt->second;
252 if ((fullName == rhsFullName) && (param->name() == (rhsParam->name()))) {
253 if ((param->type() == rhsParam->type()) && (rhsParam->type() == "double")) {
254 if (relative) {
255 if (!relErr(param->value<double>(), rhsParam->value<double>(), doubleTolerance))
256 match = true;
257 } else if (std::abs(param->value<double>() - rhsParam->value<double>()) <= doubleTolerance)
258 match = true;
259 } else if (param->asString() == rhsParam->asString()) {
260 match = true;
261 }
262 if (match)
263 break;
264 }
265 }
266
267 if (!match) {
268 // output some information that helps with understanding the mismatch
269 strOutput << "Parameter mismatch LHS=RHS for LHS parameter in component "
270 "with name: "
271 << fullName << ". Parameter name is: " << (*param).name() << " and value: " << (*param).asString()
272 << '\n';
273 bool componentWithSameNameRHS = false;
274 bool parameterWithSameNameRHS = false;
275 for (auto rhsIt = rhsMap.cbegin(); rhsIt != rhsMap.cend(); ++rhsIt) {
276 const std::string rhsFullName = rhsIt->first;
277 if (fullName == rhsFullName) {
278 componentWithSameNameRHS = true;
279 if ((*param).name() == (*rhsIt->second).name()) {
280 parameterWithSameNameRHS = true;
281 strOutput << "RHS param with same name has value: " << (*rhsIt->second).asString() << '\n';
282 }
283 }
284 }
285 if (!componentWithSameNameRHS) {
286 strOutput << "No matching RHS component name\n";
287 }
288 if (componentWithSameNameRHS && !parameterWithSameNameRHS) {
289 strOutput << "Found matching RHS component name but not parameter name\n";
290 }
291 if (firstDiffOnly)
292 return strOutput.str();
293 }
294 }
295 return strOutput.str();
296}

365void ParameterMap::add(const IComponent *comp, const std::shared_ptr<Parameter> &par,
366 const std::string *const pDescription) {
367 if (!par)
368 return;
369 checkIsNotMaskingParameter(par->name());
370 if (pDescription)
371 par->setDescription(*pDescription);
372
373 auto existing_par = positionOf(comp, par->name().c_str(), "");
374 // As this is only an add method it should really throw if it already
375 // exists.
376 // However, this is old behavior and many things rely on this actually be
377 // an
378 // add/replace-style function
379 if (existing_par != m_map.end()) {
380 std::atomic_store(&(existing_par->second), par);
381 } else {
382// When using Clang & Linux, TBB 4.4 doesn't detect C++11 features.
383// https://software.intel.com/en-us/forums/intel-threading-building-blocks/topic/641658
384#if defined(__clang__) && !defined(__APPLE__)
385#define CLANG_ON_LINUX true
386#else
387#define CLANG_ON_LINUX false
388#endif
389#if TBB_VERSION_MAJOR >= 4 && TBB_VERSION_MINOR >= 4 && !CLANG_ON_LINUX
390 m_map.emplace(comp->getComponentID(), par);
391#else
392 m_map.insert(std::make_pair(comp->getComponentID(), par));
393#endif
394 }

411void ParameterMap::addFittingParameter(const IComponent *comp, const std::string &name,
412 const std::string &fittingFunction, const std::string &value,
413 const std::string *const pDescription, const std::string &pVisible) {
414 checkIsNotMaskingParameter(name);
415 auto param = ParameterFactory::create("fitting", name, pVisible);
416 param->fromString(value);
417 if (pDescription)
418 param->setDescription(*pDescription);
419
420 // Look for an existing fitting parameter on this component with the same name AND the same
421 // embedded function; only that one should be replaced.
422 if (!m_map.empty()) {
423 const ComponentID id = comp->getComponentID();
424 auto it_found = m_map.find(id);
425 if (it_found != m_map.end()) {
426 auto itrs = m_map.equal_range(id);
427 for (auto itr = itrs.first; itr != itrs.second; ++itr) {
428 const auto &existing = itr->second;
429 if (existing->type() == "fitting" && strcasecmp(existing->nameAsCString(), name.c_str()) == 0) {
430 try {
431 if (existing->value<FitParameter>().getFunction() == fittingFunction) {
432 g_log.debug() << "addFittingParameter: [replace] matched existing (name='" << name << "', function='"
433 << fittingFunction << "') on component '" << comp->getName() << "'; overwriting in place\n";
434 std::atomic_store(&(itr->second), param);
435 return;
436 }
437 } catch (...) {
438 // Not a FitParameter value despite the type tag
439 g_log.debug() << "addFittingParameter: [skip-match] existing 'fitting' entry (name='" << name
440 << "') on component '" << comp->getName()
441 << "' did not hold a FitParameter value; ignoring it and continuing search\n";
442 }
443 }
444 }
445 }
446 }
447
448 g_log.debug() << "addFittingParameter: [insert] no existing match for (name='" << name << "', function='"
449 << fittingFunction << "') on component '" << comp->getName() << "'; adding new entry\n";
450 m_map.insert({comp->getComponentID(), param});
451}

462void ParameterMap::addPositionCoordinate(const IComponent *comp, const std::string &name, const double value,
463 const std::string *const pDescription) {
464 Parameter_sptr param = get(comp, pos());
465 V3D position;
466 if (param) {
467 // so "pos" already defined
468 position = param->value<V3D>();
469 } else {
470 // so "pos" is not defined - therefore get position from component
471 position = comp->getPos();
472 }
473
474 // adjust position
475
476 if (name == posx())
477 position.setX(value);
478 else if (name == posy())
479 position.setY(value);
480 else if (name == posz())
481 position.setZ(value);
482 else {
483 g_log.warning() << "addPositionCoordinate() called with unrecognized "
484 "coordinate symbol: "
485 << name;
486 // set description if one is provided
487 if (pDescription) {
488 param->setDescription(*pDescription);
489 }
490 return;
491 }
492
493 // clear the position cache
494 clearPositionSensitiveCaches();
495 // finally add or update "pos" parameter
496 addV3D(comp, pos(), position, pDescription);
497}

508void ParameterMap::addRotationParam(const IComponent *comp, const std::string &name, const double deg,
509 const std::string *const pDescription) {
510 Parameter_sptr paramRotX = get(comp, rotx());
511 Parameter_sptr paramRotY = get(comp, roty());
512 Parameter_sptr paramRotZ = get(comp, rotz());
513 double rotX, rotY, rotZ;
514
515 if (paramRotX)
516 rotX = paramRotX->value<double>();
517 else
518 rotX = 0.0;
519
520 if (paramRotY)
521 rotY = paramRotY->value<double>();
522 else
523 rotY = 0.0;
524
525 if (paramRotZ)
526 rotZ = paramRotZ->value<double>();
527 else
528 rotZ = 0.0;
529
530 // adjust rotation
531 Quat quat;
532 if (name == rotx()) {
533 addDouble(comp, rotx(), deg);
534 quat = Quat(deg, V3D(1, 0, 0)) * Quat(rotY, V3D(0, 1, 0)) * Quat(rotZ, V3D(0, 0, 1));
535 } else if (name == roty()) {
536 addDouble(comp, roty(), deg);
537 quat = Quat(rotX, V3D(1, 0, 0)) * Quat(deg, V3D(0, 1, 0)) * Quat(rotZ, V3D(0, 0, 1));
538 } else if (name == rotz()) {
539 addDouble(comp, rotz(), deg);
540 quat = Quat(rotX, V3D(1, 0, 0)) * Quat(rotY, V3D(0, 1, 0)) * Quat(deg, V3D(0, 0, 1));
541 } else {
542 g_log.warning() << "addRotationParam() called with unrecognized coordinate symbol: " << name;
543 return;
544 }
545
546 // clear the position cache
547 clearPositionSensitiveCaches();
548
549 // finally add or update "pos" parameter
550 addQuat(comp, rot(), quat, pDescription);
551}

738bool ParameterMap::contains(const IComponent *comp, const char *name, const char *type) const {
739 checkIsNotMaskingParameter(name);
740 if (m_map.empty())
741 return false;
742 const ComponentID id = comp->getComponentID();
743 std::pair<pmap_cit, pmap_cit> components = m_map.equal_range(id);
744 bool anytype = (strlen(type) == 0);
745 for (auto itr = components.first; itr != components.second; ++itr) {
746 const auto &param = itr->second;
747 if (strcasecmp(param->nameAsCString(), name) == 0 && (anytype || param->type() == type)) {
748 return true;
749 }
750 }
751 return false;
752}

773 return false;
774 } else
775 return false;

870Parameter_sptr ParameterMap::getByType(const IComponent *comp, const std::string &type) const {
871 Parameter_sptr result;
872 if (!m_map.empty()) {
873 const ComponentID id = comp->getComponentID();
874 auto it_found = m_map.find(id);
875 if (it_found != m_map.end() && it_found->first) {
876 auto itrs = m_map.equal_range(id);
877 for (auto itr = itrs.first; itr != itrs.second; ++itr) {
878 const auto &param = itr->second;

958Parameter_sptr ParameterMap::getRecursiveFittingParameter(const IComponent *comp, const std::string &name,
959 const std::string &fittingFunction) const {
960 checkIsNotMaskingParameter(name);
961 if (!comp || m_map.empty())
962 return Parameter_sptr();
963
964 // Walk up the component tree using a raw pointer. Wrapping the externally-owned
965 // component in a shared_ptr (even with NoDeleting) makes Coverity flag it as being
966 // managed by two smart pointers; the parent shared_ptr keeps each level alive.
967 const IComponent *current = comp;
968 std::shared_ptr<const IComponent> parent;
969 while (current != nullptr) {
970 const ComponentID id = current->getComponentID();
971 auto it_found = m_map.find(id);
972 if (it_found != m_map.end()) {
973 auto itrs = m_map.equal_range(id);
974 for (auto itr = itrs.first; itr != itrs.second; ++itr) {
975 const auto &param = itr->second;
976 if (param->type() == "fitting" && strcasecmp(param->nameAsCString(), name.c_str()) == 0) {
977 try {
978 if (param->value<FitParameter>().getFunction() == fittingFunction) {
979 return std::atomic_load(&itr->second);
980 }
981 } catch (...) {
982 // Not a FitParameter value despite the type tag, keep looking.
983 }
984 }
985 }
986 }
987 parent = current->getParent();
988 current = parent.get();
989 }
990 return Parameter_sptr();
991}

1267size_t ParameterMap::getMemorySize() const {
1268 // m_map: tbb::concurrent_unordered_multimap<ComponentID, shared_ptr<Parameter>>
1269 // Each node: key + value + ~2 pointers for chaining and bucket slot
1270 const size_t n = m_map.size();
1271 const size_t mapMem = n * (sizeof(pmap::value_type) + 2 * sizeof(void *));
1272 // Count the heap-allocated Parameter objects the shared_ptrs point to.
1273 // ParameterMap is a friend of Parameter so private string members are accessible.
1274 size_t parameterObjectsMem = 0;
1275 for (const auto &entry : m_map) {
1276 const auto &param = entry.second;
1277 if (param) {
1278 parameterObjectsMem += sizeof(Parameter);
1279 parameterObjectsMem += param->m_name.capacity();
1280 parameterObjectsMem += param->m_type.capacity();
1281 parameterObjectsMem += param->m_str_value.capacity();
1282 parameterObjectsMem += param->m_description.capacity();
1283 }
1284 }
1285 // m_parameterFileNames: vector of strings; sizeof(*this) covers the vector object, count the heap buffer
1286 const size_t fileNamesMem = m_parameterFileNames.size() * sizeof(std::string) +
1287 std::accumulate(m_parameterFileNames.cbegin(), m_parameterFileNames.cend(), size_t{0},
1288 [](size_t acc, const auto &s) { return acc + s.capacity(); });
1289 // m_cacheLocMap and m_cacheRotMap: count the heap Cache objects; the internal std::map entries are not
1290 // counted because Kernel::Cache::size() is non-const and cannot be called here.
1291 const size_t cacheLocMem = m_cacheLocMap ? sizeof(*m_cacheLocMap) : 0;
1292 const size_t cacheRotMem = m_cacheRotMap ? sizeof(*m_cacheRotMap) : 0;
1293 // m_detectorInfo and m_componentInfo: owned by ParameterMap for parametrized instruments
1294 const size_t detectorInfoMem = m_detectorInfo ? m_detectorInfo->getMemorySize() : 0;
1295 const size_t componentInfoMem = m_componentInfo ? m_componentInfo->getMemorySize() : 0;
1296 // m_instrument is a non-owning raw pointer to the base Instrument; its memory is counted in
1297 // Instrument::getMemorySize() and excluded here to avoid double-counting.
1298 return sizeof(*this) + mapMem + parameterObjectsMem + fileNamesMem + cacheLocMem + cacheRotMem + detectorInfoMem +
1299 componentInfoMem;
1300}
