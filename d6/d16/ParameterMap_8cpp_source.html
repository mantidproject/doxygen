92void ParameterMap::rebuildBeamlineFrom(const ParameterMap &source) {
93 if (!m_instrument) {
94 return;
95 }
96 auto [newComponentInfo, newDetectorInfo] = m_instrument->makeBeamline(*this, &source);
97 m_componentInfo = std::move(newComponentInfo);
98 m_detectorInfo = std::move(newDetectorInfo);
99 // Deliberately does NOT adopt m_componentInfo's store here, unlike setInstrument().
100 // This map was copy-constructed and therefore already holds its own deep copy of the parameters
101 // The metadata is read-only and refers to nothing in the source map, so it can be shared rather
102 // than rebuilt, which would walk the physical instrument again on every workspace copy.
103 if (source.m_instrumentMetadata) {
104 m_instrumentMetadata = source.m_instrumentMetadata;
105 } else {
106 buildInstrumentMetadata();
107 }
108}

122size_t ParameterMap::indexForWrite(const IComponent *comp) {
123 size_t result = indexOf(comp);
124 if (comp && result == ComponentInfo::invalidIndex) {
125 if (!m_domain) {
126 throw std::logic_error("ParameterMap: cannot add a parameter to component '" + comp->getFullName() +
127 "' before the map is attached to an instrument");
128 }
129 throw std::invalid_argument("ParameterMap: cannot add a parameter to component '" + comp->getFullName() +
130 "', which is not part of the instrument this map belongs to");
131 }
132 return result;

276bool ParameterMap::relErr(double x1, double x2, double errorVal) const {
277 double num = std::fabs(x1 - x2);
278 // how to treat x1<0 and x2 > 0 ? probably this way
279 double den = 0.5 * (std::fabs(x1) + std::fabs(x2));
280 if (den < errorVal)
281 return (num > errorVal);
282
283 return (num / den > errorVal);
284}

300const std::string ParameterMap::diff(const ParameterMap &rhs, const bool &firstDiffOnly, const bool relative,
301 const double doubleTolerance) const {
302 if (this == &rhs)
303 return std::string(""); // True for the same object
304
305 // Quick size check
306 if (this->size() != rhs.size()) {
307 return std::string("Number of parameters does not match: ") + std::to_string(this->size()) + " not equal to " +
308 std::to_string(rhs.size());
309 }
310
311 // The map is unordered and the key is only valid at runtime. The
312 // asString method turns the components to full-qualified name identifiers
313 // so we will use the same approach to compare them
314
315 std::unordered_multimap<std::string, Parameter_sptr> thisMap, rhsMap;
316 auto const byFullName = [](ParameterMap const &map, std::unordered_multimap<std::string, Parameter_sptr> &result) {
317 for (auto const &[index, parameters] : *map.m_parameterInfo) {
318 auto const fullName = map.componentFullName(index);
319 for (auto const &[parameterName, parameter] : parameters) {
320 static_cast<void>(parameterName);
321 result.emplace(fullName, parameter);
322 }
323 }
324 };
325 byFullName(*this, thisMap);
326 byFullName(rhs, rhsMap);
327
328 std::stringstream strOutput;
329 for (auto thisIt = thisMap.cbegin(); thisIt != thisMap.cend(); ++thisIt) {
330 const std::string fullName = thisIt->first;
331 const auto &param = thisIt->second;
332 bool match(false);
333 for (auto rhsIt = rhsMap.cbegin(); rhsIt != rhsMap.cend(); ++rhsIt) {
334 const std::string rhsFullName = rhsIt->first;
335 const auto &rhsParam = rhsIt->second;
336 if ((fullName == rhsFullName) && (param->name() == (rhsParam->name()))) {
337 if ((param->type() == rhsParam->type()) && (rhsParam->type() == "double")) {
338 if (relative) {
339 if (!relErr(param->value<double>(), rhsParam->value<double>(), doubleTolerance))
340 match = true;
341 } else if (std::abs(param->value<double>() - rhsParam->value<double>()) <= doubleTolerance)
342 match = true;
343 } else if (param->asString() == rhsParam->asString()) {
344 match = true;
345 }
346 if (match)
347 break;
348 }
349 }
350
351 if (!match) {
352 // output some information that helps with understanding the mismatch
353 strOutput << "Parameter mismatch LHS=RHS for LHS parameter in component "
354 "with name: "
355 << fullName << ". Parameter name is: " << (*param).name() << " and value: " << (*param).asString()
356 << '\n';
357 bool componentWithSameNameRHS = false;
358 bool parameterWithSameNameRHS = false;
359 for (auto rhsIt = rhsMap.cbegin(); rhsIt != rhsMap.cend(); ++rhsIt) {
360 const std::string rhsFullName = rhsIt->first;
361 if (fullName == rhsFullName) {
362 componentWithSameNameRHS = true;
363 if ((*param).name() == (*rhsIt->second).name()) {
364 parameterWithSameNameRHS = true;
365 strOutput << "RHS param with same name has value: " << (*rhsIt->second).asString() << '\n';
366 }
367 }
368 }
369 if (!componentWithSameNameRHS) {
370 strOutput << "No matching RHS component name\n";
371 }
372 if (componentWithSameNameRHS && !parameterWithSameNameRHS) {
373 strOutput << "Found matching RHS component name but not parameter name\n";
374 }
375 if (firstDiffOnly)
376 return strOutput.str();
377 }
378 }
379 return strOutput.str();
380}

436void ParameterMap::add(const size_t componentIndex, const std::shared_ptr<Parameter> &par,
437 const std::string *const pDescription) {
438 checkIsAttached(m_domain != nullptr);
439 if (!par || componentIndex == ComponentInfo::invalidIndex)
440 return;
441 checkIsNotMaskingParameter(par->name());
442 if (pDescription)
443 par->setDescription(*pDescription);
444
445 // As this is only an add method it should really throw if it already exists. However, this
446 // is old behavior and many things rely on this actually being an add/replace-style function
447 // -- ParameterInfo::add() implements exactly that, including the case-insensitive name match.
448 m_parameterInfo->add(componentIndex, par);
449}

479 const std::string &fittingFunction, const std::string &value,

495 << "', function='" << fittingFunction << "') on component index " << componentIndex << '\n';
496}

507void ParameterMap::addPositionCoordinate(const IComponent *comp, const std::string &name, const double value,
508 const std::string *const pDescription) {
509 Parameter_sptr param = get(comp, pos());
510 V3D position;
511 if (param) {
512 // so "pos" already defined
513 position = param->value<V3D>();
514 } else {
515 // so "pos" is not defined - therefore get position from component
516 position = comp->getPos();
517 }
518
519 // adjust position
520
521 if (name == posx())
522 position.setX(value);
523 else if (name == posy())
524 position.setY(value);
525 else if (name == posz())
526 position.setZ(value);
527 else {
528 g_log.warning() << "addPositionCoordinate() called with unrecognized "
529 "coordinate symbol: "
530 << name;
531 // set description if one is provided
532 if (pDescription) {
533 param->setDescription(*pDescription);
534 }
535 return;
536 }
537
538 // finally add or update "pos" parameter
539 addV3D(comp, pos(), position, pDescription);
540}

551void ParameterMap::addRotationParam(const IComponent *comp, const std::string &name, const double deg,
552 const std::string *const pDescription) {
553 Parameter_sptr paramRotX = get(comp, rotx());
554 Parameter_sptr paramRotY = get(comp, roty());
555 Parameter_sptr paramRotZ = get(comp, rotz());
556 double rotX, rotY, rotZ;
557
558 if (paramRotX)
559 rotX = paramRotX->value<double>();
560 else
561 rotX = 0.0;
562
563 if (paramRotY)
564 rotY = paramRotY->value<double>();
565 else
566 rotY = 0.0;
567
568 if (paramRotZ)
569 rotZ = paramRotZ->value<double>();
570 else
571 rotZ = 0.0;
572
573 // adjust rotation
574 Quat quat;
575 if (name == rotx()) {
576 addDouble(comp, rotx(), deg);
577 quat = Quat(deg, V3D(1, 0, 0)) * Quat(rotY, V3D(0, 1, 0)) * Quat(rotZ, V3D(0, 0, 1));
578 } else if (name == roty()) {
579 addDouble(comp, roty(), deg);
580 quat = Quat(rotX, V3D(1, 0, 0)) * Quat(deg, V3D(0, 1, 0)) * Quat(rotZ, V3D(0, 0, 1));
581 } else if (name == rotz()) {
582 addDouble(comp, rotz(), deg);
583 quat = Quat(rotX, V3D(1, 0, 0)) * Quat(rotY, V3D(0, 1, 0)) * Quat(deg, V3D(0, 0, 1));
584 } else {
585 g_log.warning() << "addRotationParam() called with unrecognized coordinate symbol: " << name;
586 return;
587 }
588
589 // finally add or update "pos" parameter
590 addQuat(comp, rot(), quat, pDescription);
591}

777bool ParameterMap::contains(const IComponent *comp, const Parameter &parameter) const {
778 checkIsNotMaskingParameter(parameter.name());
779 const size_t index = indexOf(comp);
780 if (index == ComponentInfo::invalidIndex)
781 return false;
782
783 // Compares whole Parameter objects (name, type and value), not just the name, so every
784 // parameter on the component has to be examined rather than looked up by name.
785 for (auto const &[storedName, param] : m_parameterInfo->parameters(index)) {
786 static_cast<void>(storedName);
787 if (param && *param == parameter)
788 return true;
789 }
790 return false;
791}

832Parameter_sptr ParameterMap::getRecursiveByType(const IComponent *comp, const std::string &type) const {
833 std::shared_ptr<const IComponent> compInFocus(comp, NoDeleting());
834 while (compInFocus != nullptr) {
835 Parameter_sptr param = getByType(compInFocus.get(), type);
836 if (param) {
837 return param;
838 }
839 compInFocus = compInFocus->getParent();
840 }
841 // Nothing was found!
842 return Parameter_sptr();
843}

866Parameter_sptr ParameterMap::getRecursive(const IComponent *comp, const char *name, const char *type) const {
867 checkIsNotMaskingParameter(name);
868 Parameter_sptr result = this->get(comp->getComponentID(), name, type);
869 if (result)
870 return result;
871
872 auto parent = comp->getParent();
873 while (parent) {
874 result = this->get(parent->getComponentID(), name, type);
875 if (result)
876 return result;
877 parent = parent->getParent();
878 }

895Parameter_sptr ParameterMap::getRecursiveFittingParameter(const IComponent *comp, const std::string &name,
896 const std::string &fittingFunction) const {
897 checkIsNotMaskingParameter(name);
898 if (!comp || m_parameterInfo->empty())
899 return Parameter_sptr();
900
901 // Walk up the component tree using a raw pointer. Wrapping the externally-owned
902 // component in a shared_ptr (even with NoDeleting) makes Coverity flag it as being
903 // managed by two smart pointers; the parent shared_ptr keeps each level alive.
904 const IComponent *current = comp;
905 std::shared_ptr<const IComponent> parent;
906 while (current != nullptr) {
907 const size_t index = indexOf(current);
908 if (index != ComponentInfo::invalidIndex) {
909 for (auto const &[storedName, param] : m_parameterInfo->parameters(index)) {
910 static_cast<void>(storedName);
911 if (param->type() == "fitting" && strcasecmp(param->nameAsCString(), name.c_str()) == 0) {
912 try {
913 if (param->value<FitParameter>().getFunction() == fittingFunction) {
914 return param;
915 }
916 } catch (...) {
917 // Not a FitParameter value despite the type tag, keep looking.
918 }
919 }
920 }
921 }
922 parent = current->getParent();
923 current = parent.get();
924 }
925 return Parameter_sptr();
926}

1123void ParameterMap::setInstrument(const Instrument *instrument) {
1124 if (instrument == m_instrument)
1125 return;
1126 if (!instrument) {
1127 m_componentInfo = nullptr;
1128 m_detectorInfo = nullptr;
1129 m_instrumentMetadata = nullptr;
1130 return;
1131 }
1132 if (m_instrument)
1133 throw std::logic_error("ParameterMap::setInstrument: Cannot change "
1134 "instrument once it has been set.");
1135 if (instrument->isParametrized())
1136 throw std::logic_error("ParameterMap::setInstrument must be called with "
1137 "base instrument, not a parametrized instrument");
1138 m_instrument = instrument;
1139 // Assigned one at a time rather than through std::tie: the members are shared_ptr while
1140 // makeBeamline returns unique_ptr, and the conversion is clearer spelled out.
1141 auto [newComponentInfo, newDetectorInfo] = m_instrument->makeBeamline(*this);
1142 m_componentInfo = std::move(newComponentInfo);
1143 m_detectorInfo = std::move(newDetectorInfo);
1144 // makeBeamline() called rekey() while building the beamline, so this map and the new
1145 // ComponentInfo now share one store. Re-point at it to be certain of that.
1146 if (m_componentInfo) {
1147 m_parameterInfo = m_componentInfo->sharedParameterInfo();
1148 seedLogfileParameters();
1149 }
1150 buildInstrumentMetadata();
1151}

1185size_t ParameterMap::getMemorySize() const {
1186 // The parameters themselves, including the Parameter objects on the heap. Note that the
1187 // store may be shared with the owning ComponentInfo, which reports it too, so summing this
1188 // with ComponentInfo::getMemorySize() double-counts it.
1189 const size_t mapMem = m_parameterInfo->getMemorySize();
1190 // m_parameterFileNames: vector of strings; sizeof(*this) covers the vector object, count the heap buffer
1191 const size_t fileNamesMem = m_parameterFileNames.size() * sizeof(std::string) +
1192 std::accumulate(m_parameterFileNames.cbegin(), m_parameterFileNames.cend(), size_t{0},
1193 [](size_t acc, const auto &s) { return acc + s.capacity(); });
1194 // m_detectorInfo and m_componentInfo: owned by ParameterMap for parametrized instruments
1195 const size_t detectorInfoMem = m_detectorInfo ? m_detectorInfo->getMemorySize() : 0;
1196 const size_t componentInfoMem = m_componentInfo ? m_componentInfo->getMemorySize() : 0;
1197 // m_instrument is a non-owning raw pointer to the base Instrument; its memory is counted in
1198 // Instrument::getMemorySize() and excluded here to avoid double-counting.
1199 return sizeof(*this) + mapMem + fileNamesMem + detectorInfoMem + componentInfoMem;
1200}
