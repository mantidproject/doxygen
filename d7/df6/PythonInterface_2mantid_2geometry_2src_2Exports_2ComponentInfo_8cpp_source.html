66// The C++ add*() methods take the description as an optional pointer. Python passes a
67// string, with the empty default standing in for "no description", since a parameter
68// description is only ever set to something non-empty.
69const std::string *descriptionOrNull(const std::string &description) {
70 return description.empty() ? nullptr : &description;
71}

73// Named parameters are usually the instrument-level ones held by the root component, so
74// 'index' is optional and None stands in for root(). A boost::python default cannot be
75// computed from 'self', so it is resolved here rather than in the keyword list.
76size_t indexOrRoot(const ComponentInfo &self, const object &componentIndex) {
77 if (componentIndex.is_none()) {
78 return self.root();
79 }
80 const extract<size_t> index(componentIndex);
81 if (!index.check()) {
82 PyErr_SetString(PyExc_TypeError, "'index' must be a component index, or None for the root component.");
83 throw_error_already_set();
84 }
85 return index();
86}

127dict shapeToComponentIndices(const ComponentInfo &componentInfo) {
128 dict result;
129 const auto shapeMap = componentInfo.shapeToComponentIndices();
130 for (const auto &shape : shapeMap) {
131 const auto csgObject = std::dynamic_pointer_cast<const Mantid::Geometry::CSGObject>(shape.first);
132 if (csgObject != nullptr) {
133 result[csgObject->getShapeXML()] = shape.second;
134 }
135 }
136 return result;
137}
