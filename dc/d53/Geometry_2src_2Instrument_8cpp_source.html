74Instrument::Instrument(const Instrument &instr)
75 : CompAssembly(instr), m_sourceCache(nullptr), m_sampleCache(nullptr), /* Should only be temporarily null */
76 m_logfileCache(instr.m_logfileCache), m_logfileUnit(instr.m_logfileUnit), m_defaultView(instr.m_defaultView),
77 m_defaultViewAxis(instr.m_defaultViewAxis), m_instr(), m_map_nonconst(), /* Should not be parameterized */
78 m_ValidFrom(instr.m_ValidFrom), m_ValidTo(instr.m_ValidTo), m_referenceFrame(instr.m_referenceFrame) {
79 // Note that we do not copy m_detectorInfo and m_componentInfo into the new instrument since they are only non-NULL
80 // for the base instrument, which should usually not be copied.
81
82 // Now we need to fill the detector, source and sample caches with pointers into the new instrument
83 std::vector<IComponent_const_sptr> children;
84 getChildren(children, true);
85 std::vector<IComponent_const_sptr>::const_iterator it;
86 for (it = children.begin(); it != children.end(); ++it) {
87 // First check if the current component is a detector and add to cache if it is
88 if (const IDetector *det = dynamic_cast<const Detector *>(it->get())) {
89 if (instr.isMonitor(det->getID()))
90 markAsMonitor(det);
91 else
92 markAsDetector(det);
93 continue;
94 }
95 // Now check whether the current component is the source or sample.
96 // As the majority of components will be detectors, we will rarely get to here
97 if (const auto *obj = dynamic_cast<const Component *>(it->get())) {
98 const std::string objName = obj->getName();
99 // This relies on the source and sample having a unique name. I think the way our instrument definition files work
100 // ensures this is the case.
101 if (objName == instr.m_sourceCache->getName()) {
102 markAsSource(obj);
103 continue;
104 }
105 if (objName == instr.m_sampleCache->getName()) {
106 markAsSamplePos(obj);
107 continue;
108 }
109 }
110 }

402std::vector<std::shared_ptr<const IComponent>> Instrument::getAllComponentsWithName(const std::string &cname) const {
403 std::shared_ptr<const IComponent> node = std::shared_ptr<const IComponent>(this, NoDeleting());
404 std::vector<std::shared_ptr<const IComponent>> retVec;
405 // Check the instrument name first
406 if (this->getName() == cname) {
407 retVec.emplace_back(node);
408 }
409 // Same algorithm as used in getComponentByName() but searching the full tree
410 std::deque<std::shared_ptr<const IComponent>> nodeQueue;
411 // Need to be able to enter the while loop
412 nodeQueue.emplace_back(node);
413 while (!nodeQueue.empty()) {
414 node = nodeQueue.front();
415 nodeQueue.pop_front();
416 int nchildren(0);
417 std::shared_ptr<const ICompAssembly> asmb = std::dynamic_pointer_cast<const ICompAssembly>(node);
418 if (asmb) {
419 nchildren = asmb->nelements();
420 }
421 for (int i = 0; i < nchildren; ++i) {
422 std::shared_ptr<const IComponent> comp = (*asmb)[i];
423 if (comp->getName() == cname) {
424 retVec.emplace_back(comp);
425 } else {
426 nodeQueue.emplace_back(comp);
427 }
428 }
429 } // while-end
430
431 // If we have reached here then the search failed
432 return retVec;
433}

690void Instrument::markAsDetector(const IDetector *det) {
691 if (m_map)
692 throw std::runtime_error("Instrument::markAsDetector() called on a "
693 "parametrized Instrument object.");
694 // ensure instrument is finalized
695 if (!isFinalized())
696 throw std::runtime_error("Instrument definition is not finalized. Add detector with markAsDetectorIncomplete, "
697 "then call markAsDetectorFinalized when finished.");
698
699 // Duplicate detector ids are forbidden
700 auto it = m_detectorCache.lower_bound(det->getID());
701 if ((it != m_detectorCache.end()) && (it->id() == det->getID())) {
702 raiseDuplicateDetectorError(det->getID());
703 }
704 // Create a (non-deleting) shared pointer to it
705 IDetector_const_sptr det_sptr = IDetector_const_sptr(det, NoDeleting());
706 m_detectorCache.emplace(it, det->getID(), det_sptr, false);
707}

724void Instrument::markAsDetectorFinalize() {
725 // We're already finalized! We can't finalize any further!
726 if (isFinalized())
727 return;
728
729 // Detectors (even when different objects) are NOT allowed to have duplicate
730 // ids. This method establishes the presence of duplicates.
731 std::sort(m_detectorCache.begin(), m_detectorCache.end(),
732 [](DetectorCacheEntry const &a, DetectorCacheEntry const &b) -> bool { return a.id() < b.id(); });
733
734 auto resultIt = std::adjacent_find(
735 m_detectorCache.begin(), m_detectorCache.end(),
736 [](DetectorCacheEntry const &a, DetectorCacheEntry const &b) -> bool { return a.id() == b.id(); });
737 if (resultIt != m_detectorCache.end()) {
738 raiseDuplicateDetectorError(resultIt->id());
739 } else {
740 m_detectorCache.setFinalized();
741 }

1234std::shared_ptr<ParameterMap> Instrument::makeLegacyParameterMap() const {
1235 if (!isFinalized())
1236 throw std::runtime_error("Instrument definition is not finalized. Can't create legacy ParameterMap.");
1237
1238 auto pmap = std::make_shared<ParameterMap>(*getParameterMap());
1239 // Instrument is only needed for DetectorInfo access so it is not needed. This
1240 // also clears DetectorInfo and ComponentInfo (information will be stored
1241 // directly in pmap so we do not need them).
1242 pmap->setInstrument(nullptr);
1243
1244 const auto &baseInstr = m_map ? *m_instr : *this;
1245
1246 if (!getParameterMap()->hasComponentInfo(&baseInstr))
1247 return pmap;
1248
1249 // Tolerance 1e-9 m with rotation center at a distance of L = 1000 m as in
1250 // Beamline::DetectorInfo::isEquivalent.
1251 constexpr double d_max = 1e-9;
1252 constexpr double L = 1000.0;
1253 constexpr double safety_factor = 2.0;
1254 const double imag_norm_max = sin(d_max / (2.0 * L * safety_factor));
1255
1256 auto transformation = Eigen::Affine3d::Identity();
1257 int64_t oldParentIndex = -1;
1258
1259 const auto &componentInfo = getParameterMap()->componentInfo();
1260 const auto &detectorInfo = getParameterMap()->detectorInfo();
1261 for (size_t i = 0; i < componentInfo.size(); ++i) {
1262
1263 const int64_t parentIndex = componentInfo.parent(i);
1264 const bool makeTransform = parentIndex != oldParentIndex;
1265 bool isDetFixedInBank = false;
1266
1267 if (makeTransform) {
1268 oldParentIndex = parentIndex;
1269 const auto parentPos = toVector3d(componentInfo.position(parentIndex));
1270 const auto invParentRot = toQuaterniond(componentInfo.rotation(parentIndex)).conjugate();
1271
1272 transformation = invParentRot;
1273 transformation.translate(-parentPos);
1274 }
1275
1276 if (componentInfo.isDetector(i)) {
1277 isDetFixedInBank = ComponentInfoBankHelpers::isDetectorFixedInBank(componentInfo, i);
1278 if (detectorInfo.isMasked(i)) {
1279 auto const &baseDet = baseInstr.m_detectorCache[i].detector();
1280 pmap->forceUnsafeSetMasked(baseDet.get(), true);
1281 }
1282
1283 if (makeTransform) {
1284 // Special case: scaling for GridDetectorPixel.
1285 if (isDetFixedInBank) {
1286
1287 size_t panelIndex = componentInfo.parent(parentIndex);
1288 const auto panelID = componentInfo.componentID(panelIndex);
1289
1290 Eigen::Vector3d scale(1, 1, 1);
1291 if (auto scalex = pmap->get(panelID, "scalex"))
1292 scale[0] = 1.0 / scalex->value<double>();
1293 if (auto scaley = pmap->get(panelID, "scaley"))
1294 scale[1] = 1.0 / scaley->value<double>();
1295 transformation.prescale(scale);
1296 }
1297 }
1298 }
1299
1300 const auto componentId = componentInfo.componentID(i);
1301 const IComponent *baseComponent = componentId->getBaseComponent();
1302 // Generic sca scale factors
1303 const auto newScaleFactor = Kernel::toVector3d(componentInfo.scaleFactor(i));
1304 if ((newScaleFactor - toVector3d(baseComponent->getScaleFactor())).norm() >= 1e-9) {
1305 pmap->addV3D(componentId, ParameterMap::scale(), componentInfo.scaleFactor(i));
1306 }
1307
1308 // Undo parent transformation to obtain relative position/rotation.
1309 Eigen::Vector3d relPos = transformation * toVector3d(componentInfo.position(i));
1310 Eigen::Quaterniond relRot = toQuaterniond(componentInfo.relativeRotation(i));
1311
1312 // Tolerance 1e-9 m as in Beamline::DetectorInfo::isEquivalent.
1313 if ((relPos - toVector3d(baseComponent->getRelativePos())).norm() >= 1e-9) {
1314 if (isDetFixedInBank) {
1315 throw std::runtime_error(
1316 "Cannot create legacy ParameterMap: Position parameters for GridDetectorPixel are not supported");
1317 }
1318 pmap->addV3D(componentId, ParameterMap::pos(), Kernel::toV3D(relPos));
1319 }
1320 if ((relRot * toQuaterniond(baseComponent->getRelativeRot()).conjugate()).vec().norm() >= imag_norm_max) {
1321 pmap->addQuat(componentId, ParameterMap::rot(), Kernel::toQuat(relRot));
1322 }
1323 }
1324
1325 return pmap;
1326}
