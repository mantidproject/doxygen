31size_t indexOfPath(ComponentInfo const &componentInfo, std::string const &path) {
32 if (componentInfo.size() == 0) {
33 return ComponentInfo::invalidIndex;
34 }
35 size_t current = componentInfo.root();
36 size_t segmentStart = 0;
37 while (segmentStart <= path.size()) {
38 auto const separator = path.find('/', segmentStart);
39 auto const segment =
40 path.substr(segmentStart, separator == std::string::npos ? separator : separator - segmentStart);
41 std::deque<size_t> queue{current};
42 size_t found = ComponentInfo::invalidIndex;
43 while (!queue.empty() && found == ComponentInfo::invalidIndex) {
44 auto const index = queue.front();
45 queue.pop_front();
46 if (componentInfo.name(index) == segment) {
47 found = index;
48 } else {
49 auto const &children = componentInfo.children(index);
50 queue.insert(queue.end(), children.cbegin(), children.cend());
51 }
52 }
53 if (found == ComponentInfo::invalidIndex || separator == std::string::npos) {
54 return found;
55 }
56 current = found;
57 segmentStart = separator + 1;
58 }
59 return ComponentInfo::invalidIndex;
60}

100void MoveInstrumentComponent::exec() {
101 // Get the input workspace
102 Workspace_sptr ws = getProperty("Workspace");
103 MatrixWorkspace_sptr inputW = std::dynamic_pointer_cast<MatrixWorkspace>(ws);
104 DataObjects::PeaksWorkspace_sptr inputP = std::dynamic_pointer_cast<DataObjects::PeaksWorkspace>(ws);
105
106 if (!inputW && !inputP) {
107 throw std::runtime_error("The workspace does not seem to be valid as input (must be either MatrixWorkspace or "
108 "PeaksWorkspace");
109 }
110 const std::string ComponentName = getProperty("ComponentName");
111 const int DetID = getProperty("DetectorID");
112 const double X = getProperty("X");
113 const double Y = getProperty("Y");
114 const double Z = getProperty("Z");
115 const bool relativePosition = getProperty("RelativePosition");
116 const bool moveFixedDetectors = getProperty("MoveFixedDetectors");
117
118 auto &componentInfo = inputW ? inputW->mutableComponentInfo() : inputP->mutableComponentInfo();
119 size_t compIndex = ComponentInfo::invalidIndex;
120 // Find the component to move
121 if (DetID != -1) {
122 auto const &detectorInfo = inputW ? inputW->detectorInfo() : inputP->detectorInfo();
123 try {
124 compIndex = detectorInfo.indexOf(DetID);
125 } catch (std::out_of_range const &) {
126 std::ostringstream mess;
127 mess << "Detector with ID " << DetID << " was not found.";
128 g_log.error(mess.str());
129 throw std::runtime_error(mess.str());
130 }
131 } else if (!ComponentName.empty()) {
132 compIndex = indexOfPath(componentInfo, ComponentName);
133 if (compIndex == ComponentInfo::invalidIndex) {
134 std::ostringstream mess;
135 mess << "Component with name " << ComponentName << " was not found.";
136 g_log.error(mess.str());
137 throw std::runtime_error(mess.str());
138 }
139 } else {
140 g_log.error("DetectorID or ComponentName must be given.");
141 throw std::invalid_argument("DetectorID or ComponentName must be given.");
142 }
143
144 if (!moveFixedDetectors && ComponentInfoBankHelpers::isDetectorFixedInBank(componentInfo, compIndex)) {
145 // DetectorInfo makes changing positions possible but we keep the old
146 // behavior of ignoring position changes for Structured banks.
147 g_log.warning("Component is fixed within a structured bank, moving is not "
148 "possible, doing nothing.");
149 return;
150 }
151
152 // Do the move
153 V3D position(X, Y, Z);
154 if (relativePosition)
155 position += componentInfo.position(compIndex);
156
157 componentInfo.setPosition(compIndex, position);
158}
