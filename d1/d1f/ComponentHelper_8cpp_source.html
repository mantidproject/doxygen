24Geometry::Instrument_sptr makeVirtualInstrument(Kernel::V3D const &sourcePos, Kernel::V3D const &samplePos,
25 std::vector<Kernel::V3D> const &vecdetpos,
26 std::vector<detid_t> const &vecdetid,
27 std::shared_ptr<const ReferenceFrame> referenceFrame) {
28 BeamlineBuilder builder;
29 auto const root = builder.addAssembly(BeamlineBuilder::noParent, "");
30 builder.setReferenceFrame(std::move(referenceFrame));
31 builder.setSource(builder.addObjComponent(root, "source", sourcePos, Kernel::Quat(),
32 createSphere(0.01 /*1cm*/, V3D(0, 0, 0), "1")));
33 builder.setSample(builder.addComponent(root, "some-surface-holder", samplePos));
34 for (size_t i = 0; i < vecdetpos.size(); ++i) {
35 // FIXME - should be cubi... pixel
36 builder.addDetector(root, "point-detector", vecdetid[i] /*detector id*/, vecdetpos[i], Kernel::Quat(),
37 createSphere(0.01 /*1cm*/, V3D(0, 0, 0), "1"));
38 }
39 return Instrument::fromBeamline(builder.build());
40}
