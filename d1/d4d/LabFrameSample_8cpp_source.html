20Geometry::IObject_sptr labFrameSampleShape(const ExperimentInfo &experiment) {
21 const auto &sample = experiment.sample();
22 const auto &goniometerR = experiment.run().getGoniometer().getR();
23 if (Geometry::outstandingGoniometerRotation(sample.getShape(), goniometerR) == Kernel::Matrix<double>(3, 3, true)) {
24 return sample.getShapePtr();
25 }
26 if (sample.hasEnvironment()) {
27 g_log.warning("The sample environment is not currently rotated by the goniometer, so the goniometer "
28 "rotation is not applied to the sample in it either. The sample is used as defined.");
29 return sample.getShapePtr();
30 }
31 return Geometry::getLabFrameShape(sample.getShape(), goniometerR);
32}

34Sample labFrameSample(const ExperimentInfo &experiment) {
