24void addFullInstrumentToWorkspace(MatrixWorkspace &workspace, bool includeMonitors, bool startYNegative,
25 const std::string &instrumentName) {
26 BeamlineBuilder builder;
27 auto const root = builder.addAssembly(BeamlineBuilder::noParent, instrumentName);
28 builder.setReferenceFrame(std::make_shared<ReferenceFrame>(Y, Z, Right, ""));
29
30 const double pixelRadius(0.05);
31 auto pixelShape =
32 ComponentCreationHelper::createCappedCylinder(pixelRadius, 0.02, V3D(0.0, 0.0, 0.0), V3D(0., 1.0, 0.), "tube");
33
34 const double detZPos(5.0);
35 // Careful! Do not use size_t or auto, the unsigned will break the -=2 below.
36 auto ndets = static_cast<int>(workspace.getNumberHistograms());
37 if (includeMonitors)
38 ndets -= 2;
39 for (int i = 0; i < ndets; ++i) {
40 std::ostringstream lexer;
41 lexer << "pixel-" << i << ")";
42 int ycount(i);
43 if (startYNegative)
44 ycount -= 1;
45 const double ypos = ycount * 2.0 * pixelRadius;
46 const detid_t id = workspace.getAxis(1)->spectraNo(i);
47 builder.addDetector(root, lexer.str(), id, V3D(0.0, ypos, detZPos), Quat(), pixelShape);
48 workspace.getSpectrum(i).setDetectorID(id);
49 }
50 if (ndets > 0) {
51 // Only the XML text differs from a new instrument's defaults
52 builder.setMetadata(std::make_shared<InstrumentMetadata>(Types::Core::DateAndTime(), Types::Core::DateAndTime(), "",
53 "Fake XML", "3D", "Z+"));
54 }
55
56 // Monitors last
57 if (includeMonitors) // These occupy the last 2 spectra
58 {
59 const detid_t monitorId = workspace.getAxis(1)->spectraNo(ndets);
60 builder.addDetector(root, "mon1", monitorId, V3D(0.0, 0.0, -9.0), Quat(), nullptr, /*isMonitor*/ true);
61 workspace.getSpectrum(ndets).setDetectorID(ndets + 1);
62 builder.addDetector(root, "mon2", monitorId + 1, V3D(0.0, 0.0, -2.0), Quat(), nullptr, /*isMonitor*/ true);
63 workspace.getSpectrum(ndets + 1).setDetectorID(ndets + 2);
64 }
65
66 // Define a source component
67 builder.setSource(builder.addObjComponent(root, "moderator", V3D(0.0, 0.0, -20.0), Quat(),
68 ComponentCreationHelper::createSphere(0.1, V3D(0, 0, 0), "1")));
69 // Define a sample position
70 builder.setSample(builder.addComponent(root, "samplePos"));
71 // chopper position
72 builder.addComponent(root, "chopper-position", V3D(0, 0, -10));
73 workspace.setInstrument(Instrument::fromBeamline(builder.build()));
74}

76void addInstrumentWithGeographicalDetectorsToWorkspace(Mantid::API::MatrixWorkspace &workspace, const int nlat,
77 const int nlong, const double anginc,
78 const std::string &instrumentName) {
79 V3D samplePosition(0., 0., 0.);
80 V3D sourcePosition(0., 0., -14.);
81
82 BeamlineBuilder builder;
83 auto const root = builder.addAssembly(BeamlineBuilder::noParent, instrumentName);
84 builder.setReferenceFrame(std::make_shared<ReferenceFrame>(Mantid::Geometry::Y, Mantid::Geometry::Z, Right, "0,0,0"));
85
86 InstrumentCreationHelper::addSource(builder, root, sourcePosition, "source");
87 InstrumentCreationHelper::addSample(builder, root, samplePosition, "sample");
88
89 auto const stage = builder.addOutlineAssembly(root, "detector-stage", V3D(), Quat(), nullptr);
90 // set up detectors with even spacing in latitude and longitude (to match geographical angles
91 // approach used in the spatial interpolation\sparse instrument functionality)
92 int i = 0;
93 constexpr double deg2rad = M_PI / 180.0;
94 const double angincRad = anginc * deg2rad;
95 constexpr double R = 1.0;
96 for (int lat = 0; lat < nlat; ++lat) {
97 for (int lng = 0; lng < nlong; ++lng) {
98 std::stringstream buffer;
99 buffer << "detector_" << i;
100 V3D detPos;
101 auto latrad = lat * angincRad;
102 auto longrad = lng * angincRad;
103 detPos[1] = R * sin(latrad);
104 const double ct = R * cos(latrad);
105 detPos[2] = ct * cos(longrad);
106 detPos[0] = ct * sin(longrad);
107
108 InstrumentCreationHelper::addDetector(builder, stage, detPos, i, buffer.str());
109 // Link it to the workspace
110 workspace.getSpectrum(i).addDetectorID(i);
111 i++;
112 }
113 }
114 workspace.setInstrument(Instrument::fromBeamline(builder.build()));
115}
