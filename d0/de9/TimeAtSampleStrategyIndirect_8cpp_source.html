37double TimeAtSampleStrategyIndirect::getEfixed(const size_t &workspace_index) const {
38 if (!m_spectrumInfo.hasDetectors(workspace_index)) {
39 throw Kernel::Exception::NotFoundError("TimeAtSampleStrategyIndirect: no detectors for spectrum",
40 std::to_string(workspace_index));
41 }
42 // The first detector, for a spectrum of several
43 const auto detIndex = m_spectrumInfo.spectrumDefinition(workspace_index)[0].first;
44 const auto noEfixed = [&]() {
45 std::stringstream errmsg;
46 errmsg << "Inelastic instrument detector " << m_detectorInfo.detid(detIndex) << " of spectrum " << workspace_index
47 << " does not have EFixed ";
48 return std::runtime_error(errmsg.str());
49 };
50
51 double efixed{0.};
52 // A spectrum of several detectors has no parameters of its own
53 if (m_spectrumInfo.hasUniqueDetector(workspace_index)) {
54 try {
55 const auto values = m_componentInfo.getNumberParameter(detIndex, "Efixed");
56 if (!values.empty()) {
57 efixed = values.front();
58 }
59 } catch (std::runtime_error &) {
60 throw noEfixed();
61 }
62 }
63 if (efixed <= 0.) {
64 throw noEfixed();
65 }
66
67 return efixed;
68}

70Correction TimeAtSampleStrategyIndirect::calculate(const size_t &workspace_index) const {
71
72 // A constant among all spectra
73 constexpr double TWO_MEV_OVER_MASS = 2. * PhysicalConstants::meV / PhysicalConstants::NeutronMass;
74
75 if (m_spectrumInfo.isMonitor(workspace_index)) {
76 // use the same math as TimeAtSampleStrategyElastic
77 const double L1m =
78 m_beamDir.scalar_prod(m_spectrumInfo.sourcePosition() - m_spectrumInfo.position(workspace_index));
79 const double scale = std::abs(m_L1s / L1m);
80 return Correction(scale, 0.);
81 } else {
82 const double efix = this->getEfixed(workspace_index);
83 const double l2 = m_spectrumInfo.l2(workspace_index);
84 const double shift = -1. * l2 / sqrt(efix * TWO_MEV_OVER_MASS);
85
86 // 1.0 * tof + shift
87 constexpr double NO_MULTIPLY_TOF{1.};
88 return Correction(NO_MULTIPLY_TOF, shift);
89 }
90}
