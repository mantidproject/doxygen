317std::vector<double> getIntegratedCountsForWorkspaceIndices(MatrixWorkspace &self,
318 const boost::python::object &workspaceIndices,
319 const size_t numberOfWorkspaces, const double minX,
320 const double maxX, const bool entireRange) {
321
322 std::vector<size_t> sWorkspaceIndices(numberOfWorkspaces);

324 NDArrayToVector<size_t> converter(workspaceIndices);

327 PySequenceToVector<size_t> converter(workspaceIndices);
328 converter.copyTo(sWorkspaceIndices);
329 }
330
331 return self.getIntegratedCountsForWorkspaceIndices(sWorkspaceIndices, minX, maxX, entireRange);
332}

407object getSignalAtCoord(MatrixWorkspace &self, const NDArray &npCoords,
408 const Mantid::API::MDNormalization &normalization) {
409 if (npCoords.get_shape()[1] != 2) {
410 throw std::invalid_argument("MatrixWorkspace::getSignalAtCoord - Input "
411 "array must have shape (n, 2)");
412 }
413 // Create our output array
414 Py_intptr_t length = len(npCoords);
415 auto *signalValues = new Mantid::signal_t[length];
416
417 // Convert coords to a vector
418 std::vector<Mantid::coord_t> coords = NDArrayToVector<Mantid::coord_t>(npCoords)();
419
420 // Fill output array
421 for (int i = 0; i < length; ++i) {
422 std::array<Mantid::coord_t, 2> coord = {{coords[2 * i], coords[2 * i + 1]}};
423 signalValues[i] = self.getSignalAtCoord(coord.data(), normalization);
424 }
425 PyObject *npSignalArray =
426 Impl::wrapWithNDArray(signalValues, 1, &length, NumpyWrapMode::ReadOnly, OwnershipMode::Python);
427 return object(handle<>(npSignalArray));
428}
