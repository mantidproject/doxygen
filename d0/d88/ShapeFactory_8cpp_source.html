62std::vector<double> DegreesToRadians(const std::vector<double> &anglesDegrees) {
63 std::vector<double> anglesRadians;
64 std::transform(anglesDegrees.cbegin(), anglesDegrees.cend(), std::back_inserter(anglesRadians),
65 [](const auto angle) { return angle * M_PI / 180.0; });
66 return anglesRadians;
67}

108std::shared_ptr<CSGObject> ShapeFactory::createShape(Poco::XML::Element *pElem) {
109 // Write the definition to a string to store in the final object
110 std::stringstream xmlstream;
111 DOMWriter writer;
112 writer.writeNode(xmlstream, pElem);
113 std::string shapeXML = xmlstream.str();
114 auto retVal = std::make_shared<CSGObject>(shapeXML);
115
116 // if no <algebra> element then use default algebra
117 bool defaultAlgebra(false);
118 // get algebra string
119 Poco::AutoPtr<NodeList> pNL_algebra = pElem->getElementsByTagName("algebra");
120 std::string algebraFromUser;
121 if (pNL_algebra->length() == 0) {
122 defaultAlgebra = true;
123 } else if (pNL_algebra->length() == 1) {
124 auto *pElemAlgebra = static_cast<Element *>(pNL_algebra->item(0));
125 algebraFromUser = pElemAlgebra->getAttribute("val");
126 } else {
127 g_log.warning() << "More than one algebra string defined for this shape. "
128 << "Maximum one allowed. Therefore empty shape is returned.";
129 return retVal;
130 }
131
132 // <goniometer> is the TOTAL rotation applied to the surfaces below as they are parsed.
133 Poco::AutoPtr<NodeList> pNL_gonio = pElem->getElementsByTagName("goniometer");
134 auto *pElemGonio = static_cast<Element *>(pNL_gonio->item(0));
135 m_gonioRotateMatrix.identityMatrix();
136 if (pElemGonio) {
137 m_gonioRotateMatrix = parseMatrixElement(pElemGonio);
138 }
139
140 // <applied-goniometer> says how much of that total is a bake into the lab frame. Pure metadata -
141 // it rotates nothing - and the only account of which frame the parsed surfaces ended up in.
142 Poco::AutoPtr<NodeList> pNL_applied = pElem->getElementsByTagName("applied-goniometer");
143 auto *pElemApplied = static_cast<Element *>(pNL_applied->item(0));
144 Kernel::Matrix<double> bakedRotation(3, 3, true);
145 if (pElemApplied && !pElemGonio) {
146 g_log.warning() << "An <applied-goniometer> tag was given without a <goniometer> tag. There is "
147 << "no rotation for it to describe, so it is ignored.\n";
148 } else if (pElemApplied) {
149 bakedRotation = parseMatrixElement(pElemApplied);
150 } else {
151 // No <applied-goniometer>: read the whole <goniometer> as a bake, which is what CopySample and
152 // SetSample wrote. RotateSampleShape also wrote a lone tag, holding Rs * R, and that cannot be told
153 // apart here, so its Rs is misread as part of the bake.
154 bakedRotation = m_gonioRotateMatrix;
155 }
156 retVal->setAppliedGoniometerRotation(bakedRotation);
157
158 Poco::AutoPtr<NodeList> pNL_rotate_all = pElem->getElementsByTagName("rotate-all");
159 auto *pElemRotateAll = static_cast<Element *>(pNL_rotate_all->item(0));
160 m_rotateAllMatrix.identityMatrix();
161 if (pElemRotateAll) {
162 std::vector<double> rotateAngles = DegreesToRadians(parsePosition(pElemRotateAll));
163 m_rotateAllMatrix = generateMatrix(rotateAngles[0], rotateAngles[1], rotateAngles[2]);
164 }
165
166 // match id given to a shape by the user to
167 // id understandable by Mantid code
168 std::map<std::string, std::string> idMatching;
169
170 // loop over all the sub-elements of pElem
171 Poco::AutoPtr<NodeList> pNL = pElem->childNodes(); // get all child nodes
172 unsigned long pNL_length = pNL->length();
173 int numPrimitives = 0;
174 // stores the primitives that will be
175 // used to build final shape
176 std::map<int, std::shared_ptr<Surface>> primitives;
177 // used to build up unique id's for each shape added. Must start
178 // from int > zero.
179 int l_id = 1;
180 // Element of fixed complete object
181 Element *lastElement = nullptr;
182 for (unsigned int i = 0; i < pNL_length; i++) {
183 if ((pNL->item(i))->nodeType() == Node::ELEMENT_NODE) {
184 auto *pE = static_cast<Element *>(pNL->item(i));
185
186 // assume for now that if sub-element has attribute id then it is a shape
187 // element
188 if (pE->hasAttribute("id")) {
189 std::string idFromUser = pE->getAttribute("id"); // get id
190
191 std::string primitiveName = pE->tagName(); // get name of primitive
192
193 // if there are any error thrown while parsing the XML string for a
194 // given shape
195 // write out a warning to the user that this shape is ignored. If all
196 // shapes are ignored
197 // this way an empty object is returned to the user.
198 try {
199 if (primitiveName == "sphere") {
200 lastElement = pE;
201 idMatching[idFromUser] = parseSphere(pE, primitives, l_id);
202 numPrimitives++;
203 } else if (primitiveName == "infinite-plane") {
204 idMatching[idFromUser] = parseInfinitePlane(pE, primitives, l_id);
205 retVal->setFiniteGeometryFlag(false);
206 numPrimitives++;
207 } else if (primitiveName == "infinite-cylinder") {
208 idMatching[idFromUser] = parseInfiniteCylinder(pE, primitives, l_id);
209 retVal->setFiniteGeometryFlag(false);
210 numPrimitives++;
211 } else if (primitiveName == "cylinder") {
212 lastElement = pE;
213 idMatching[idFromUser] = parseCylinder(pE, primitives, l_id);
214 numPrimitives++;
215 } else if (primitiveName == "hollow-cylinder") {
216 lastElement = pE;
217 idMatching[idFromUser] = parseHollowCylinder(pE, primitives, l_id);
218 numPrimitives++;
219 } else if (primitiveName == "cuboid") {
220 lastElement = pE;
221 idMatching[idFromUser] = parseCuboid(pE, primitives, l_id);
222 numPrimitives++;
223 } else if (primitiveName == "infinite-cone") {
224 idMatching[idFromUser] = parseInfiniteCone(pE, primitives, l_id);
225 retVal->setFiniteGeometryFlag(false);
226 numPrimitives++;
227 } else if (primitiveName == "cone") {
228 lastElement = pE;
229 idMatching[idFromUser] = parseCone(pE, primitives, l_id);
230 numPrimitives++;
231 } else if (primitiveName == "hexahedron") {
232 lastElement = pE;
233 idMatching[idFromUser] = parseHexahedron(pE, primitives, l_id);
234 numPrimitives++;
235 } else if (primitiveName == "tapered-guide") {
236 idMatching[idFromUser] = parseTaperedGuide(pE, primitives, l_id);
237 numPrimitives++;
238 } else if (primitiveName == "torus") {
239 idMatching[idFromUser] = parseTorus(pE, primitives, l_id);
240 numPrimitives++;
241 } else if (primitiveName == "slice-of-cylinder-ring") {
242 idMatching[idFromUser] = parseSliceOfCylinderRing(pE, primitives, l_id);
243 numPrimitives++;
244 } else {
245 g_log.warning(primitiveName + " not a recognised geometric shape. This shape is ignored.");
246 }
247 } catch (std::invalid_argument &e) {
248 g_log.warning() << e.what() << " <" << primitiveName << "> shape is ignored.";
249 } catch (std::runtime_error &e) {
250 g_log.warning() << e.what() << " <" << primitiveName << "> shape is ignored.";
251 } catch (...) {
252 g_log.warning() << " Problem with parsing XML string for <" << primitiveName << ">. This shape is ignored.";
253 }
254 }
255 }
256 }
257
258 if (!defaultAlgebra) {
259 // Translate algebra string defined by the user into something Mantid can
260 // understand
261
262 // std::string algebra; // to hold algebra in a way Mantid can understand
263 std::map<std::string, std::string>::iterator iter;
264 std::map<size_t, std::string, std::greater<size_t>> allFound;
265 for (iter = idMatching.begin(); iter != idMatching.end(); ++iter) {
266 size_t found = algebraFromUser.find(iter->first);
267
268 if (found == std::string::npos) {
269 defaultAlgebra = true;
270 g_log.warning() << "Algebra shape Warning: " + iter->first +
271 " not found in algebra string: " + algebraFromUser + "\n" +
272 ". Default to equal shape to intersection of those defined.";
273 break;
274 } else {
275 allFound[found] = iter->first;
276 }
277 }
278
279 // Here do the actually swapping of strings
280 // but only if the algebra containes all the shapes
281 if (allFound.size() == idMatching.size()) {
282 std::map<size_t, std::string, std::greater<size_t>>::iterator iter2;
283 for (iter2 = allFound.begin(); iter2 != allFound.end(); ++iter2) {
284 // std::string kuse = iter2->second;
285 algebraFromUser.replace(iter2->first, (iter2->second).size(), idMatching[iter2->second]);
286 }
287 }
288 }
289 if (defaultAlgebra) {
290 algebraFromUser = ""; // reset in case we are overwriten invalid string
291 std::map<std::string, std::string>::iterator iter;
292 for (iter = idMatching.begin(); iter != idMatching.end(); ++iter) {
293 algebraFromUser.append(iter->second + " "); // default is intersection
294 }
295 }
296
297 // check to see if there actually were any primitives in 'type' xml element
298 // and if yes then return empty Object. Otherwise populate Object with the
299 // primitives
300
301 if (numPrimitives == 0)
302 return retVal;
303 else {
304 retVal->setObject(21, algebraFromUser);
305 retVal->populate(primitives);
306 // check whether there is only one surface/closed surface
307 if (numPrimitives == 1 && lastElement != nullptr) // special case
308 {
309 // parse the primitive and create a Geometry handler for the object
310 createGeometryHandler(lastElement, retVal);
311 }
312
313 // get bounding box string
314 Poco::AutoPtr<NodeList> pNL_boundingBox = pElem->getElementsByTagName("bounding-box");
315 if (pNL_boundingBox->length() != 1) // should probably throw an error if
316 // more than 1 bounding box is
317 // defined...
318 return retVal;
319
320 double xmin = std::stod((getShapeElement(pElem, "x-min"))->getAttribute("val"));
321 double ymin = std::stod((getShapeElement(pElem, "y-min"))->getAttribute("val"));
322 double zmin = std::stod((getShapeElement(pElem, "z-min"))->getAttribute("val"));
323 double xmax = std::stod((getShapeElement(pElem, "x-max"))->getAttribute("val"));
324 double ymax = std::stod((getShapeElement(pElem, "y-max"))->getAttribute("val"));
325 double zmax = std::stod((getShapeElement(pElem, "z-max"))->getAttribute("val"));
326
327 retVal->defineBoundingBox(xmax, ymax, zmax, xmin, ymin, zmin);
328
329 return retVal;
330 }
331}

344std::string ShapeFactory::parseSphere(Poco::XML::Element *pElem, std::map<int, std::shared_ptr<Surface>> &prim,
345 int &l_id) {
346 Element *pElemCentre = getOptionalShapeElement(pElem, "centre");
347 Element *pElemRadius = getShapeElement(pElem, "radius");
348 const double radius = getDoubleAttribute(pElemRadius, "val");
349 V3D centre = pElemCentre ? parsePosition(pElemCentre) : DEFAULT_CENTRE;
350
351 // Only rotate the normal vector by the rotate-all and goniometer tags
352 // Special case: do not obey rotate tag as rotation around the centre of sphere is unneccessary
353 if (m_rotateAllMatrix != Kernel::Matrix<double>(3, 3, true)) {
354 // Apply automatic rotation due to the rotate-all tag
355 centre.rotate(m_rotateAllMatrix);
356 }
357 if (m_gonioRotateMatrix != Kernel::Matrix<double>(3, 3, true)) {
358 // Apply automatic rotation due to the goniometer that should NOT be manually defined by the user
359 centre.rotate(m_gonioRotateMatrix);
360 }
361
362 prim[l_id] = std::make_shared<Sphere>(centre, radius);
363 const auto algebra = sphereAlgebra(l_id);
364 l_id++;
365 return algebra;
366}

402 }

404 // Apply automatic rotation due to the rotate-all tag
405 centre.rotate(m_rotateAllMatrix);
406 normVec.rotate(m_rotateAllMatrix);

409 // Apply automatic rotation due to the goniometer that should NOT be manually defined by the user
410 centre.rotate(m_gonioRotateMatrix);
411 normVec.rotate(m_gonioRotateMatrix);
412 }
413
414 // create infinite-plane
415 auto pPlane = std::make_shared<Plane>();
416 pPlane->setPlane(centre, normVec);
417 prim[l_id] = pPlane;
418
419 std::stringstream retAlgebraMatch;

518 normVec.rotate(rotateMatrix);

526 // Apply automatic rotation due to the goniometer that should NOT be manually defined by the user
527 centre.rotate(m_gonioRotateMatrix);
528 normVec.rotate(m_gonioRotateMatrix);

554 l_id++;

571std::string ShapeFactory::parseHollowCylinder(Poco::XML::Element *pElem, std::map<int, std::shared_ptr<Surface>> &prim,
572 int &l_id) {
573 Element *pElemBase = getShapeElement(pElem, "centre-of-bottom-base");
574 Element *pElemAxis = getShapeElement(pElem, "axis");
575 Element *pElemInnerRadius = getShapeElement(pElem, "inner-radius");
576 Element *pElemOuterRadius = getShapeElement(pElem, "outer-radius");
577 Element *pElemHeight = getShapeElement(pElem, "height");
578 Element *pElem_rot = getOptionalShapeElement(pElem, "rotate");
579
580 const double innerRadius = getDoubleAttribute(pElemInnerRadius, "val");
581 if (innerRadius <= 0.0) {
582 throw std::runtime_error("ShapeFactory::parseHollowCylinder(): inner-radius < 0.0");
583 }
584 const double outerRadius = getDoubleAttribute(pElemOuterRadius, "val");
585 if (outerRadius <= 0.0) {
586 throw std::runtime_error("ShapeFactory::parseHollowCylinder(): outer-radius < 0.0");
587 }
588 if (innerRadius > outerRadius) {
589 throw std::runtime_error("ShapeFactory::parseHollowCylinder(): inner-radius > outer-radius.");
590 }
591 const double height = getDoubleAttribute(pElemHeight, "val");
592 if (height <= 0.0) {
593 throw std::runtime_error("ShapeFactory::parseHollowCylinder(): height < 0.0");
594 }
595 V3D centreOfBottomBase = parsePosition(pElemBase);
596 V3D normVec = normalize(parsePosition(pElemAxis));
597 V3D centre = centreOfBottomBase + normVec * (0.5 * height);
598
599 // Rotate the normal vector by the rotate, rotate-all and goniometer tags
600 // Rotate the centre by the rotate-all and goniometer tags
601 if (pElem_rot) {
602 // Apply manual rotation supplied to rotate tag
603 std::vector<double> rotateAngles = DegreesToRadians(parsePosition(pElem_rot));
604 const std::vector<double> rotateMatrix = generateMatrix(rotateAngles[0], rotateAngles[1], rotateAngles[2]);
605 normVec.rotate(rotateMatrix);
606 }
607 if (m_rotateAllMatrix != Kernel::Matrix<double>(3, 3, true)) {
608 // Apply automatic rotation due to the rotate-all tag
609 centre.rotate(m_rotateAllMatrix);
610 normVec.rotate(m_rotateAllMatrix);
611 }
612 if (m_gonioRotateMatrix != Kernel::Matrix<double>(3, 3, true)) {
613 // Apply automatic rotation due to the goniometer that should NOT be manually defined by the user
614 centre.rotate(m_gonioRotateMatrix);
615 normVec.rotate(m_gonioRotateMatrix);
616 }
617
618 // add outer infinite cylinder surface
619 auto outerCylinder = std::make_shared<Cylinder>();
620 outerCylinder->setCentre(centre);
621 outerCylinder->setNorm(normVec);
622 outerCylinder->setRadius(outerRadius);
623 prim[l_id] = outerCylinder;
624
625 std::stringstream retAlgebraMatch;
626 retAlgebraMatch << "(-" << l_id << " ";
627 l_id++;
628
629 // add inner infinite cylinder surface
630 auto innerCylinder = std::make_shared<Cylinder>();
631 innerCylinder->setCentre(centre);
632 innerCylinder->setNorm(normVec);
633 innerCylinder->setRadius(innerRadius);
634 prim[l_id] = innerCylinder;
635 retAlgebraMatch << l_id << " ";
636 l_id++;
637
638 // add top plane
639 auto pPlaneTop = std::make_shared<Plane>();
640 // to get point in top plane
641 V3D pointInPlaneTop = centre + (normVec * height * 0.5);
642 pPlaneTop->setPlane(pointInPlaneTop, normVec);
643 prim[l_id] = pPlaneTop;
644 retAlgebraMatch << "-" << l_id << " ";
645 l_id++;
646
647 // add bottom plane
648 auto pPlaneBottom = std::make_shared<Plane>();
649 V3D pointInPlaneBottom = centre - (normVec * height * 0.5);

667CuboidCorners ShapeFactory::parseCuboid(Poco::XML::Element *pElem) {
668 // Users have two syntax options when defining cuboids:
669
670 // A - "Point" syntax.
671 Element *pElem_lfb = getOptionalShapeElement(pElem, "left-front-bottom-point");
672 Element *pElem_lft = getOptionalShapeElement(pElem, "left-front-top-point");
673 Element *pElem_lbb = getOptionalShapeElement(pElem, "left-back-bottom-point");
674 Element *pElem_rfb = getOptionalShapeElement(pElem, "right-front-bottom-point");
675
676 // B - "Alternate" syntax.
677 Element *pElem_height = getOptionalShapeElement(pElem, "height");
678 Element *pElem_width = getOptionalShapeElement(pElem, "width");
679 Element *pElem_depth = getOptionalShapeElement(pElem, "depth");
680 Element *pElem_centre = getOptionalShapeElement(pElem, "centre");
681 Element *pElem_axis = getOptionalShapeElement(pElem, "axis");
682 Element *pElem_rot = getOptionalShapeElement(pElem, "rotate");
683
684 const bool usingPointSyntax = pElem_lfb && pElem_lft && pElem_lbb && pElem_rfb;
685 const bool usingAlternateSyntax = pElem_height && pElem_width && pElem_depth;
686
687 const bool usedPointSyntaxField = pElem_lfb || pElem_lft || pElem_lbb || pElem_rfb;
688 const bool usedAlternateSyntaxField =
689 pElem_height || pElem_width || pElem_depth || pElem_centre || pElem_axis || pElem_rot;
690
691 const std::string SYNTAX_ERROR_MSG = "XML element: <" + pElem->tagName() +
692 "> may contain EITHER corner points (LFB, LFT, LBB and RFB) OR " +
693 "height, width, depth, centre and axis values.";
694
695 CuboidCorners result;
696
697 if (usingPointSyntax && !usingAlternateSyntax) {
698 if (usedAlternateSyntaxField)
699 throw std::invalid_argument(SYNTAX_ERROR_MSG);
700
701 result.lfb = parsePosition(pElem_lfb);
702 result.lft = parsePosition(pElem_lft);
703 result.lbb = parsePosition(pElem_lbb);
704 result.rfb = parsePosition(pElem_rfb);
705 } else if (usingAlternateSyntax && !usingPointSyntax) {
706 if (usedPointSyntaxField)
707 throw std::invalid_argument(SYNTAX_ERROR_MSG);
708
709 const double deltaH = getDoubleAttribute(pElem_height, "val") / 2;
710 const double deltaW = getDoubleAttribute(pElem_width, "val") / 2;
711 const double deltaD = getDoubleAttribute(pElem_depth, "val") / 2;
712
713 V3D centre = pElem_centre ? parsePosition(pElem_centre) : DEFAULT_CENTRE;
714
715 result.lfb = V3D(-deltaW, -deltaH, -deltaD);
716 result.lft = V3D(-deltaW, deltaH, -deltaD);
717 result.lbb = V3D(-deltaW, -deltaH, deltaD);
718 result.rfb = V3D(deltaW, -deltaH, -deltaD);
719
720 if (pElem_axis) {
721 // Use a quarternion to do a rotate for us, with respect to the default
722 // axis. Our "Quat" implementation requires that the vectors passed to
723 // it be normalised.
724 const V3D axis = normalize(parsePosition(pElem_axis));
725 const Quat rotate(DEFAULT_AXIS, axis);
726
727 rotate.rotate(result.lfb);
728 rotate.rotate(result.lft);
729 rotate.rotate(result.lbb);
730 rotate.rotate(result.rfb);
731 }
732
733 // Rotate the points by the rotate, rotate-all and goniometer tags
734 // Rotate the centre by the rotate-all and goniometer tags
735 if (pElem_rot) {
736 // Apply manual rotation supplied to rotate tag
737 std::vector<double> rotateAngles = DegreesToRadians(parsePosition(pElem_rot));
738 const std::vector<double> rotateMatrix = generateMatrix(rotateAngles[0], rotateAngles[1], rotateAngles[2]);
739 result.rotatePoints(rotateMatrix);
740 }
741 if (m_rotateAllMatrix != Kernel::Matrix<double>(3, 3, true)) {
742 // Apply automatic rotation due to the rotate-all tag
743 result.rotatePoints(m_rotateAllMatrix);
744 centre.rotate(m_rotateAllMatrix);
745 }
746 if (m_gonioRotateMatrix != Kernel::Matrix<double>(3, 3, true)) {
747 // Apply automatic rotation due to the goniometer that should NOT be manually defined by the user
748 result.rotatePoints(m_gonioRotateMatrix);
749 centre.rotate(m_gonioRotateMatrix);
750 }
751
752 result.lfb += centre;
753 result.lft += centre;
754 result.lbb += centre;
755 result.rfb += centre;
756 } else {
757 throw std::invalid_argument(SYNTAX_ERROR_MSG);
758 }
759
760 return result;
761}

1670std::string ShapeFactory::insertMatrixTag(const std::string &tagName, const Kernel::Matrix<double> &matrix,
1671 std::string xml) {
1672 // Keeping the '<' attached is what stops "goniometer" matching inside "<applied-goniometer".
1673 const std::string openTag = "<" + tagName;
1674
1675 // Delete any previous tag of this name. The closing bracket is part of the tag, so erase it too -
1676 // leaving it behind drops a stray '>' into the character data on every rewrite, which is legal
1677 // XML and so goes unnoticed until the tags are rewritten repeatedly.
1678 const std::size_t foundTag = xml.find(openTag);
1679 if (foundTag != std::string::npos) {
1680 xml.erase(foundTag, xml.find(">", foundTag + 1) - foundTag + 1);
1681 }
1682
1683 // Insert before the innermost enclosing end tag, or at the end if there is none
1684 std::size_t tagPlace = xml.find("</type>");
1685 if (tagPlace == std::string::npos) {
1686 tagPlace = xml.find("</samplegeometry");
1687 }
1688 if (tagPlace == std::string::npos) {
1689 tagPlace = xml.size();
1690 }
1691
1692 // Full precision: these matrices are now composed rather than overwritten, so rounding here
1693 // accumulates over repeated CopySample and RotateSampleShape calls.
1694 std::ostringstream tag;
1695 tag << std::setprecision(std::numeric_limits<double>::max_digits10);
1696 tag << " " << openTag << " ";
1697 for (size_t i = 0; i < matrix.numRows(); ++i) {
1698 for (size_t j = 0; j < matrix.numCols(); ++j) {
1699 tag << "a" << i + 1 << j + 1 << " = '" << matrix[i][j] << "' ";
1700 }
1701 }
1702 tag << "/>";
1703 xml.insert(tagPlace, tag.str());
1704
1705 return xml;
1706}

1749std::string ShapeFactory::rebakeGoniometer(const Kernel::Matrix<double> &newBake, std::string xml,
1750 const Kernel::Matrix<double> &currentBake) {
1751 // Strip the old bake off the total, leaving the definition-frame rotation, then put the new bake
1752 // on the outside of it. These are orthonormal so the transpose is the exact inverse.
1753 //
1754 // With nothing to preserve the answer is just the new bake. Taking that shortcut is not merely
1755 // cheaper but exact: the long way round multiplies by the old bake and its transpose, identity
1756 // only to within rounding, and those last bits reorder the rendered mesh's triangles and
1757 // accumulate over copies.
1758 const Kernel::Matrix<double> total = goniometerFromXML(xml);
1759 const Kernel::Matrix<double> newTotal = total == currentBake ? newBake : newBake * currentBake.Tprime() * total;
1760
1761 xml = addGoniometerTag(newTotal, std::move(xml));
1762 return addAppliedGoniometerTag(newBake, std::move(xml));
1763}
