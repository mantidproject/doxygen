140std::string cuboidXML(double xHalfLength, double yHalfLength, double zHalfLength, const V3D &centrePos,
141 const std::string &id) {
142 const double szX = xHalfLength;
143 const double szY = (yHalfLength == -1.0 ? szX : yHalfLength);
144 const double szZ = (zHalfLength == -1.0 ? szX : zHalfLength);
145
146 // top\bottom along z
147 V3D leftFrontBottom{szX, -szY, -szZ};
148 V3D leftFrontTop{szX, -szY, szZ};
149 V3D leftBackBottom{-szX, -szY, -szZ};
150 V3D rightFrontBottom{szX, szY, -szZ};
151
152 leftFrontBottom += centrePos;
153 leftFrontTop += centrePos;
154 leftBackBottom += centrePos;
155 rightFrontBottom += centrePos;
156
157 std::ostringstream xmlShapeStream;
158 xmlShapeStream << " <cuboid id=\"" << id << "\"> "
159 << "<left-front-bottom-point x=\"" << leftFrontBottom.X() << "\" y=\"" << leftFrontBottom.Y()
160 << "\" z=\"" << leftFrontBottom.Z() << "\" /> "
161 << "<left-front-top-point x=\"" << leftFrontTop.X() << "\" y=\"" << leftFrontTop.Y() << "\" z=\""
162 << leftFrontTop.Z() << "\" /> "
163 << "<left-back-bottom-point x=\"" << leftBackBottom.X() << "\" y=\"" << leftBackBottom.Y()
164 << "\" z=\"" << leftBackBottom.Z() << "\" /> "
165 << "<right-front-bottom-point x=\"" << rightFrontBottom.X() << "\" y=\"" << rightFrontBottom.Y()
166 << "\" z=\"" << rightFrontBottom.Z() << "\" /> "
167 << "</cuboid>";
168
169 return xmlShapeStream.str();
170}

190std::shared_ptr<CSGObject> createCuboid(double xHalfLength, double yHalfLength, double zHalfLength, double angle,
191 const Mantid::Kernel::V3D &axis) {
192 // top\bottom along z
193 V3D leftFrontBottom{xHalfLength, -yHalfLength, -zHalfLength};
194 V3D leftFrontTop{xHalfLength, -yHalfLength, zHalfLength};
195 V3D leftBackBottom{-xHalfLength, -yHalfLength, -zHalfLength};
196 V3D rightFrontBottom{xHalfLength, yHalfLength, -zHalfLength};
197 Quat rotation(angle, axis);
198 auto rotMatrix = Mantid::Kernel::DblMatrix(rotation.getRotation());
199 leftFrontBottom.rotate(rotMatrix);
200 leftFrontTop.rotate(rotMatrix);
201 leftBackBottom.rotate(rotMatrix);
202 rightFrontBottom.rotate(rotMatrix);
203 std::ostringstream xmlShapeStream;
204 xmlShapeStream << " <cuboid id=\"detector-shape\"> "
205 << "<left-front-bottom-point x=\"" << leftFrontBottom.X() << "\" y=\"" << leftFrontBottom.Y()
206 << "\" z=\"" << leftFrontBottom.Z() << "\" /> "
207 << "<left-front-top-point x=\"" << leftFrontTop.X() << "\" y=\"" << leftFrontTop.Y() << "\" z=\""
208 << leftFrontTop.Z() << "\" /> "
209 << "<left-back-bottom-point x=\"" << leftBackBottom.X() << "\" y=\"" << leftBackBottom.Y()
210 << "\" z=\"" << leftBackBottom.Z() << "\" /> "
211 << "<right-front-bottom-point x=\"" << rightFrontBottom.X() << "\" y=\"" << rightFrontBottom.Y()
212 << "\" z=\"" << rightFrontBottom.Z() << "\" /> "
213 << "</cuboid>";
214
215 std::string xmlCuboidShape(xmlShapeStream.str());
216 ShapeFactory shapeCreator;
217 auto cuboidShape = shapeCreator.createShape(xmlCuboidShape);
218 return cuboidShape;
219}

296std::vector<std::unique_ptr<IDetector>> createVectorOfCylindricalDetectors(const double R_min, const double R_max,
297 const double z0) {
298 std::vector<std::unique_ptr<IDetector>> allDetectors;
299 // One object
300 double R0 = 0.5;
301 double h = 1.5;
302 auto detShape = ComponentCreationHelper::createCappedCylinder(R0, h, V3D(0.0, 0.0, 0.0), V3D(0., 1.0, 0.), "tube");
303
304 auto NY = int(ceil(2 * R_max / h) + 1);
305 auto NX = int(ceil(2 * R_max / R0) + 1);
306 double y_bl = NY * h;
307 double x_bl = NX * R0;
308
309 double Rmin2(R_min * R_min), Rmax2(R_max * R_max);
310
311 int ic(0);
312 for (int j = 0; j < NY; j++) {
313 double y = -0.5 * y_bl + j * h;
314 for (int i = 0; i < NX; i++) {
315 double x = -0.5 * x_bl + i * R0;
316 double Rsq = x * x + y * y;
317 if (Rsq >= Rmin2 && Rsq < Rmax2) {
318 std::ostringstream os;
319 os << "d" << ic;
320 auto det = std::make_unique<Detector>(os.str(), ic + 1, detShape, nullptr);
321 det->setPos(x, y, z0);
322 allDetectors.emplace_back(std::move(det));
323 }
324
325 ic++;
326 }
327 }
328 return allDetectors;
329}

335std::shared_ptr<DetectorGroup> createRingOfCylindricalDetectors(const double R_min, const double R_max,
336 const double z0) {
337
338 auto vecOfDetectors = createVectorOfCylindricalDetectors(R_min, R_max, z0);
339 std::vector<std::shared_ptr<const IDetector>> groupMembers;
340 groupMembers.reserve(vecOfDetectors.size());
341 std::transform(vecOfDetectors.begin(), vecOfDetectors.end(), std::back_inserter(groupMembers),
342 [](auto &det) { return std::move(det); });
343 return std::make_shared<DetectorGroup>(std::move(groupMembers));
344}

417Instrument_sptr createTestInstrumentCylindrical(int num_banks, const Mantid::Kernel::V3D &sourcePos,
418 const Mantid::Kernel::V3D &samplePos, const double cylRadius,
419 const double cylHeight, const std::string &instrumentName) {
420 BeamlineBuilder builder;
421 auto const root = builder.addAssembly(BeamlineBuilder::noParent, instrumentName, samplePos);
422
423 // One object
424 auto pixelShape = ComponentCreationHelper::createCappedCylinder(cylRadius, cylHeight, V3D(0.0, -cylHeight / 2.0, 0.0),
425 V3D(0., 1.0, 0.), "pixel-shape");
426
427 // Just increment pixel IDs
428 int pixelID = 1;
429
430 for (int banknum = 1; banknum <= num_banks; banknum++) {
431 auto const bank = builder.addAssembly(root, "bank" + std::to_string(banknum), V3D(0.0, 0.0, 5.0 * banknum));
432 // Nine object components
433 for (int i = -1; i < 2; ++i) {
434 for (int j = -1; j < 2; ++j) {
435 std::ostringstream lexer;
436 lexer << "pixel-(" << j << ";" << i << ")";
437 const double xpos = j * (cylRadius * 2.0);
438 const double ypos = i * cylHeight;
439 builder.addDetector(bank, lexer.str(), pixelID++, V3D(xpos, ypos, 0.0), Quat(), pixelShape);
440 }
441 }
442 }
443
444 addSource(builder, root, sourcePos);
445 addSample(builder, root);
446 return instrumentFrom(builder);
447}

449Mantid::Geometry::Instrument_sptr
450createCylInstrumentWithVerticalOffsetsSpecified(size_t nTubes, const std::vector<double> &verticalOffsets,
451 size_t nDetsPerTube, double xMin, double xMax, double yMin,
452 double yMax) {
453 // Pixel shape
454 const double ySpan = (yMax - yMin);
455 const double xSpan = (xMax - xMin);
456 const double tubeDiameter = xSpan / static_cast<double>(nTubes); // No gaps between tubes
457 const double cylRadius = tubeDiameter / 2; // No gaps between tubes
458 const double cylHeight = ySpan / static_cast<double>(nDetsPerTube);
459 const double bankZPos = 2;
460 const double sourceZPos = -10;
461 const double sampleZPos = 0;
462
463 auto pixelShape = ComponentCreationHelper::createCappedCylinder(cylRadius, cylHeight, V3D(0.0, 0.0, 0.0),
464 V3D(0., 1.0, 0.), "pixel-shape");
465 BeamlineBuilder builder;
466 auto const root = builder.addAssembly(BeamlineBuilder::noParent, "instrument_with_tubes", V3D(0, 0, sampleZPos));
467 builder.setReferenceFrame(
468 std::make_shared<ReferenceFrame>(Mantid::Geometry::Y /*up*/, Mantid::Geometry::Z /*along*/, Left, "0,0,0"));
469 auto const bank = builder.addAssembly(root, "sixteenpack", V3D(0, 0, bankZPos));
470 std::vector<V3D> pixelPositions(nDetsPerTube);
471 for (size_t j = 0; j < nDetsPerTube; ++j) {
472 pixelPositions[j] = V3D(0, static_cast<double>(j) * cylHeight, 0);
473 }
474 for (size_t i = 0; i < nTubes; ++i) {
475 // Each tube has its own outline, as createOutline() gives
476 auto const tube = builder.addOutlineAssembly(
477 bank, "tube" + std::to_string(i),
478 V3D(xMin + static_cast<double>(i) * tubeDiameter, -ySpan / 2 + verticalOffsets[i], 0), Quat(),
479 tubeOutline(pixelShape, pixelPositions));
480 for (size_t j = 0; j < nDetsPerTube; ++j) {
481 auto id = static_cast<int>(i * nDetsPerTube + j);
482 builder.addDetector(tube, "det-" + std::to_string(id), id, pixelPositions[j], Quat(), pixelShape);
483 }
484 }
485 addSource(builder, root, V3D(0, 0, sourceZPos));
486 addSample(builder, root);
487 return instrumentFrom(builder);
488}

494bool double_cmprsn(double x1, double x2) {
495 const double TOL(1.e-4);
496 return Mantid::Kernel::withinRelativeDifference(x1, x2, TOL);

498Mantid::Geometry::Instrument_sptr createCylInstrumentWithDetInGivenPositions(const std::vector<double> &L2,
499 const std::vector<double> &polar,
500 const std::vector<double> &azim,
501 const std::string &instrumentName) {
502 // find characteristic sizes of the detectors;
503 double dAzi_min(FLT_MAX);
504 double dPol_min(FLT_MAX);
505 double L2_min(FLT_MAX);
506 double dAzi, dPol;
507 std::vector<double> az(azim);
508 std::vector<double> po(polar);
509 std::sort(az.begin(), az.end());
510 std::sort(po.begin(), po.end());
511 // very crude identification of interdetector distance; no need in more
512 // accurate caluclations for example;
513 for (size_t i = 0; i < L2.size(); i++) {
514 if (L2[i] < L2_min)
515 L2_min = L2[i];
516 for (size_t j = i + 1; j < L2.size(); j++) {
517 if (!double_cmprsn(az[i], az[j])) {
518 dAzi = std::fabs(az[i] - az[j]);
519 if (dAzi < dAzi_min)
520 dAzi_min = dAzi;
521 }
522 if (!double_cmprsn(po[i], po[j])) {
523 dPol = std::fabs(po[i] - po[j]);
524 if (dPol < dPol_min)
525 dPol_min = dPol;
526 }
527 }
528 }
529 double cylRadius = L2_min * sin(dAzi_min * 0.5);
530 double cylHeight = 2 * L2_min * sin(dPol_min * 0.5);
531
532 // One object
533 auto pixelShape = ComponentCreationHelper::createCappedCylinder(cylRadius, cylHeight, V3D(0.0, -cylHeight / 2.0, 0.0),
534 V3D(0., 1.0, 0.), "pixel-shape");
535 BeamlineBuilder builder;
536 auto const root = builder.addAssembly(BeamlineBuilder::noParent, instrumentName);
537 // one bank
538 auto const bank = builder.addAssembly(root, "det_ass");
539 // Just increment pixel ID's
540 int pixelID = 1;
541 for (size_t i = 0; i < azim.size(); i++) {
542 double zpos = L2[i] * cos(polar[i]);
543 double xpos = L2[i] * sin(polar[i]) * cos(azim[i]);
544 double ypos = L2[i] * sin(polar[i]) * sin(azim[i]);
545 builder.addDetector(bank, "det" + std::to_string(i), pixelID++, V3D(xpos, ypos, zpos), Quat(), pixelShape);
546 }
547
548 addSource(builder, root, V3D(0.0, 0.0, -L2_min));
549 addSample(builder, root);
550 return instrumentFrom(builder);
551}

556void addRectangularBank(Instrument &testInstrument, int idStart, int pixels, double pixelSpacing,
557 const std::string &bankName, const V3D &bankPos, const Quat &bankRot) {
558
559 const double cylRadius(pixelSpacing / 2);
560 const double cylHeight(0.0002);
561 // One object
562 auto pixelShape = ComponentCreationHelper::createCappedCylinder(cylRadius, cylHeight, V3D(0.0, -cylHeight / 2.0, 0.0),
563 V3D(0., 1.0, 0.), "pixel-shape");
564
565 auto *bank = new RectangularDetector(bankName);
566 bank->initialize(pixelShape, pixels, 0.0, pixelSpacing, pixels, 0.0, pixelSpacing, idStart, true, pixels);
567
568 // Mark them all as detectors
569 for (int x = 0; x < pixels; x++)
570 for (int y = 0; y < pixels; y++) {
571 std::shared_ptr<Detector> detector = bank->getAtXY(x, y);
572 if (detector)
573 // Mark it as a detector (add to the instrument cache)
574 testInstrument.markAsDetector(detector.get());
575 }
576
577 testInstrument.add(bank);
578 bank->setPos(bankPos);
579 bank->setRot(bankRot);
580}

584BeamlineBuilder rectangularBuilder(int num_banks, int pixels, double pixelSpacing, double bankDistanceFromSample,

586 BeamlineBuilder builder;

881Mantid::Geometry::Instrument_sptr createInstrumentWithPSDTubes(const size_t nTubes, const size_t nPixelsPerTube,
882 bool mirrorTubes) {
883 // Need a tube based instrument.
884 //
885 // Pixels will be numbered simply from 1->nTubes*nPixelsPerTube with a 1:1
886 // mapping
887 //
888 // Tubes will be located at 1 m from the sample (0, 0, 0) from 0 -> 90 deg
889 // If mirror is set to true they will go from 0 -> -90 deg
890 BeamlineBuilder builder;
891 auto const root = builder.addAssembly(BeamlineBuilder::noParent, "PSDTubeInst");
892 int xDirection(1);
893 if (mirrorTubes)
894 xDirection = -1;
895
896 builder.setReferenceFrame(
897 std::make_shared<ReferenceFrame>(Mantid::Geometry::Y, Mantid::Geometry::Z, Mantid::Geometry::X, Right, "0,0,0"));
898
899 // Pixel shape
900 const double pixelRadius(0.01);
901 const double pixelHeight(0.003);
902 const double radius(1.0);
903 const auto pixelShape = ComponentCreationHelper::createCappedCylinder(
904 pixelRadius, pixelHeight, V3D(0.0, -0.5 * pixelHeight, 0.0), V3D(0.0, 1.0, 0.0), "pixelShape");
905
906 const auto tubeShape = ComponentCreationHelper::createCappedCylinder(
907 pixelRadius, pixelHeight, V3D(0.0, -0.5 * pixelHeight * (double)nPixelsPerTube, 0.0), V3D(0.0, 1.0, 0.0),
908 "tubeShape");
909
910 for (size_t i = 0; i < nTubes; ++i) {
911 const auto theta = (M_PI / 2.0) * double(i) / (double(nTubes) - 1);
912 auto x = xDirection * radius * sin(theta);
913 // A small correction to make testing easier where the instrument is
914 // mirrored
915 if (i == 0 && xDirection < 0)
916 x = -1e-32;
917 const auto z = radius * cos(theta);
918 auto const tube = builder.addOutlineAssembly(root, "tube-" + std::to_string(i), V3D(x, 0.0, z), Quat(), tubeShape);
919 for (size_t j = 0; j < nPixelsPerTube; ++j) {
920 const double ypos = double(j) * pixelHeight;
921 builder.addDetector(tube, "pixel-" + std::to_string(i * nPixelsPerTube + j), int(i * nPixelsPerTube + j + 1),
922 V3D(0.0, ypos, 0.0), Quat(), pixelShape);
923 }
924 }
925 addSource(builder, root, V3D(0.0, 0.0, -1.0));
926 addSample(builder, root);
927 return instrumentFrom(builder);
928}
