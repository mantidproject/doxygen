97template <typename TYPE> void PropertyWithValue<TYPE>::saveProperty(Nexus::File *file) {
98 if constexpr (std::is_same_v<TYPE, float> || std::is_same_v<TYPE, double> || std::is_same_v<TYPE, int32_t> ||
99 std::is_same_v<TYPE, uint32_t> || std::is_same_v<TYPE, int64_t> || std::is_same_v<TYPE, uint64_t> ||
100 std::is_same_v<TYPE, long> || std::is_same_v<TYPE, std::string> ||
101 std::is_same_v<TYPE, std::vector<double>> || std::is_same_v<TYPE, std::vector<int32_t>>) {
102 file->makeGroup(this->name(), "NXlog", true);
103 file->writeData("value", m_value);
104 file->openData("value");
105 file->putAttr("units", this->units());
106 file->closeData();
107 file->closeGroup();
108 } else {
109 // AppleClang 7.3 and later gives a -Winfinite-recursion warning if I call the
110 // base class method. The function is small enough that reimplementing it
111 // isn't a big deal.
112 throw std::invalid_argument("PropertyWithValue::saveProperty - Cannot save '" + this->name() + "', property type " +
113 typeid(TYPE).name() + " not implemented.");
114 }
115}

179template <typename TYPE> std::string PropertyWithValue<TYPE>::setValue(const std::string &value) {
180 try {
181 TYPE result = m_value;
182 std::string valueCopy = value;
183 if (autoTrim()) {
184 boost::trim(valueCopy);
185 }
186 toValue(valueCopy, result);
187 // Uses the assignment operator defined below which runs isValid() and
188 // throws based on the result
189 *this = result;
190 return "";
191 } catch (boost::bad_lexical_cast &) {
192 std::string error = "Could not set property " + name() + ". Can not convert \"" + value + "\" to " + type();
193 g_logger.debug() << error;
194 return error;
195 } catch (std::invalid_argument &except) {
196 g_logger.debug() << "Could not set property " << name() << ": " << except.what();
197 return except.what();
198 }
199}

272template <typename TYPE> PropertyWithValue<TYPE> &PropertyWithValue<TYPE>::operator=(const TYPE &value) {
273 TYPE oldValue = m_value;
274 if constexpr (std::is_same<TYPE, std::string>::value) {
275 std::string valueCopy = toString(value);
276 if (autoTrim()) {
277 boost::trim(valueCopy);
278 }
279 toValue(valueCopy, m_value);
280 } else {
281 m_value = value;
282 }
283 std::string problem = this->isValid();
284 if (problem.empty()) {
285 return *this;
286 } else if (problem == "_alias") {
287 m_value = getValueForAlias(value);
288 return *this;
289 } else {
290 m_value = oldValue;
291 throw std::invalid_argument("When setting value of property \"" + this->name() + "\": " + problem);
292 }
293}
