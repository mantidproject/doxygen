257void IFunction::addTies(const std::string &ties, bool isDefault) {
258 std::map<size_t, std::pair<size_t, std::string>> oldTies;
259
260 Expression list;
261 list.parse(ties);
262 list.toList();
263 for (const auto &t : list) {
264 if (t.name() == "=" && t.size() >= 2) {
265 size_t n = t.size() - 1;
266 const std::string expr = t[n].str();
267 for (size_t i = n; i != 0;) {
268 --i;
269 const auto &parName = t[i].name();
270 auto parTie = createAndProcessTie(parName, expr, isDefault);
271
272 if (parTie) {
273 auto iPar = getParameterIndex(*parTie);
274 oldTies[iPar] = insertTie(std::move(parTie));
275 }
276 }
277 }
278 }
279
280 try {
281 sortTies(true);
282 } catch (std::runtime_error &) {
283 for (const auto &[iPar, oldTie] : oldTies) {
284 auto [oldIdx, oldExp] = oldTie;
285
286 if (!oldExp.empty()) {
287 m_ties[oldIdx] = std::make_unique<ParameterTie>(this, parameterName(iPar), oldExp);
288 } else {
289 removeTie(iPar);
290 }
291 }
292 throw;
293 }
294 applyTies();
295}

329std::pair<std::size_t, std::string> IFunction::insertTie(std::unique_ptr<ParameterTie> tie) {
330 auto iPar = getParameterIndex(*tie);
331 std::size_t existingTieIndex =
332 std::distance(m_ties.begin(), std::find_if(m_ties.begin(), m_ties.end(),
333 [&](const auto &m_tie) { return getParameterIndex(*m_tie) == iPar; }));
334 std::string oldExp = "";
335 const auto oldTie = getTie(iPar);
336 if (oldTie) {
337 const auto oldTieStr = oldTie->asString();
338 oldExp = oldTieStr.substr(oldTieStr.find("=") + 1);
339 }
340
341 if (existingTieIndex < m_ties.size()) {
342 m_ties[existingTieIndex] = std::move(tie);
343 } else {
344 m_ties.emplace_back(std::move(tie));
345 setParameterStatus(iPar, Tied);
346 }
347
348 if (oldTie) {
349 return {existingTieIndex, oldExp};
350 } else {
351 return {existingTieIndex, ""};
352 }
353}

360void IFunction::addTie(std::unique_ptr<ParameterTie> tie) {
361 auto iPar = getParameterIndex(*tie);
362 const auto [oldIdx, oldExp] = insertTie(std::move(tie));
363
364 try {
365 // sortTies checks for circular and self ties
366 sortTies(true);
367 } catch (std::runtime_error &) {
368 // revert / remove tie if invalid
369 if (!oldExp.empty()) {
370 m_ties[oldIdx] = std::make_unique<ParameterTie>(this, parameterName(iPar), oldExp);
371 } else {
372 removeTie(iPar);
373 }
374 throw;
375 }

553std::string IFunction::writeToString(const std::string &parentLocalAttributesStr) const {
554 std::ostringstream ostr;
555 ostr << "name=" << this->name();
556 // print the attributes
557 std::vector<std::string> attr = this->getAttributeNames();
558 for (const auto &attName : attr) {
559 std::string attValue = this->getAttribute(attName).value();
560 if (!attValue.empty() && attValue != "\"\"") {
561 ostr << ',' << attName << '=' << attValue;
562 }
563 }
564 std::vector<std::string> ties;
565 // print the parameters
566 for (size_t i = 0; i < nParams(); i++) {
567 std::ostringstream paramOut;
568 paramOut << parameterName(i) << '=' << getParameter(i);
569 ostr << ',' << paramOut.str();
570 // Output non-default ties only.
571 if (getParameterStatus(i) == Fixed) {
572 ties.emplace_back(paramOut.str());
573 }
574 }
575
576 // collect non-default constraints
577 std::string constraints = writeConstraints();
578 // print constraints
579 if (!constraints.empty()) {
580 ostr << ",constraints=(" << constraints << ")";
581 }
582
583 // collect the non-default ties
584 auto tiesString = writeTies();
585 if (!tiesString.empty()) {
586 ties.emplace_back(tiesString);
587 }
588 // print the ties
589 if (!ties.empty()) {
590 ostr << ",ties=(" << Kernel::Strings::join(ties.begin(), ties.end(), ",") << ")";
591 }
592 // "local" attributes of a parent composite function
593 ostr << parentLocalAttributesStr;
594 return ostr.str();
595}

603void IFunction::addConstraints(const std::string &str, bool isDefault) {
604 Expression list;
605 list.parse(str);
606 list.toList();
607 for (auto it = list.begin(); it != list.end(); ++it) {
608 auto expr = (*it);
609 if (expr.terms()[0].str().compare("penalty") == 0) {
610 continue;
611 }
612 if ((it + 1) != list.end()) {
613 auto next_expr = *(it + 1);
614 if (next_expr.terms()[0].str().compare("penalty") == 0) {
615 auto c = std::unique_ptr<IConstraint>(ConstraintFactory::Instance().createInitialized(this, expr, isDefault));
616 double penalty_factor = std::stof(next_expr.terms()[1].str(), NULL);
617 c->setPenaltyFactor(penalty_factor);
618 this->addConstraint(std::move(c));
619 } else {
620 auto c = std::unique_ptr<IConstraint>(ConstraintFactory::Instance().createInitialized(this, expr, isDefault));
621 this->addConstraint(std::move(c));
622 }
623 } else {
624 auto c = std::unique_ptr<IConstraint>(ConstraintFactory::Instance().createInitialized(this, expr, isDefault));
625 this->addConstraint(std::move(c));
626 }
627 }

1113void IFunction::calNumericalDeriv(const FunctionDomain &domain, Jacobian &jacobian) {
1114 /*
1115 * There is a similar more specialized method for 1D functions in IFunction1D
1116 * but the method takes different parameters and uses slightly different
1117 * function calls in places making it difficult to share code. Please also
1118 * consider that method when updating this.
1119 */
1120
1121 const size_t nParam = nParams();
1122 size_t nData = getValuesSize(domain);
1123
1124 FunctionValues minusStep(nData);
1125 FunctionValues plusStep(nData);
1126
1127 applyTies(); // just in case
1128 function(domain, minusStep);
1129
1130 if (nData == 0) {
1131 nData = minusStep.size();
1132 }
1133
1134 double step;
1135 for (size_t iP = 0; iP < nParam; iP++) {
1136 if (isActive(iP)) {
1137 const double val = activeParameter(iP);
1138 step = calculateStepSize(iP, val);
1139
1140 const double paramPstep = val + step;
1141 setActiveParameter(iP, paramPstep);
1142 applyTies();
1143 function(domain, plusStep);
1144 setActiveParameter(iP, val);
1145 applyTies();
1146
1147 step = paramPstep - val;
1148 for (size_t i = 0; i < nData; i++) {
1149 jacobian.set(i, iP, (plusStep.getCalculated(i) - minusStep.getCalculated(i)) / step);
1150 }
1151 }
1152 }

1183void IFunction::setMatrixWorkspace(std::shared_ptr<const API::MatrixWorkspace> workspace, size_t wi, double startX,
1184 double endX) {
1185 UNUSED_ARG(startX);
1186 UNUSED_ARG(endX);
1187
1188 if (!workspace)
1189 return; // unset the workspace
1190
1191 try {
1192
1193 // check if parameter are specified in instrument definition file
1194
1195 // Note JZ oct 2011 - I'm not sure why the code uses the first detector
1196 // and not the group. Ask Roman.
1197 // Note RB oct 2026 - Roman was not asked. And now, he never will be.
1198 const auto &specDef = workspace->spectrumInfo().spectrumDefinition(wi);
1199 if (specDef.size() == 0) {
1200 g_log.information() << "MatrixWorkspace has not been set for index: " << std::to_string(wi)
1201 << " as no detector ids can be found for this workspace on the instrument\n";
1202 return;
1203 }
1204 // Detector indices in ComponentInfo match those in the SpectrumDefinition
1205 const auto &componentInfo = workspace->componentInfo();
1206 const auto &paramInfo = *(componentInfo.sharedParameterInfo());
1207 const auto detIndex = specDef[0].first;
1208
1209 for (size_t i = 0; i < nParams(); i++) {
1210 if (!isExplicitlySet(i)) {
1211 // Use the fitting-function-aware lookup: two functions on the same component can declare a
1212 // parameter with the same short name (e.g. Bk2BkExpConvPV:Gamma and IkedaCarpenterPV:Gamma).
1213 // The plain getRecursive() short-circuits on the first match and would pick the wrong one.
1214 Geometry::Parameter_sptr param =
1215 paramInfo.getRecursiveFittingParameter(componentInfo, detIndex, parameterName(i), this->name());
1216 if (!param) {
1217 // Fall back for IDFs that omit the function prefix (no embedded function name to match).
1218 param = paramInfo.getRecursive(componentInfo, detIndex, parameterName(i), "fitting");
1219 }
1220 if (param != Geometry::Parameter_sptr()) {
1221 // get FitParameter
1222 const auto &fitParam = param->value<Geometry::FitParameter>();
1223
1224 // check first if this parameter is actually specified for this
1225 // function
1226 if (name() == fitParam.getFunction()) {
1227 // update value
1228 const auto *testWithLocation = dynamic_cast<IFunctionWithLocation *>(this);
1229 if (testWithLocation == nullptr ||
1230 (!fitParam.getLookUpTable().containData() && fitParam.getFormula().empty())) {
1231 setParameter(i, fitParam.getValue());
1232 } else {
1233 double centreValue = testWithLocation->centre();
1234 Kernel::Unit_sptr centreUnit; // unit of value used in formula or
1235 // to look up value in lookup table
1236 if (fitParam.getFormula().empty())
1237 centreUnit = fitParam.getLookUpTable().getXUnit(); // from table
1238 else {
1239 if (!fitParam.getFormulaUnit().empty()) {
1240 try {
1241 centreUnit = Kernel::UnitFactory::Instance().create(fitParam.getFormulaUnit()); // from formula
1242 } catch (...) {
1243 g_log.warning() << fitParam.getFormulaUnit() << " Is not an recognised formula unit for parameter "
1244 << fitParam.getName() << "\n";
1245 }
1246 }
1247 }
1248
1249 // if unit specified convert centre value to unit required by
1250 // formula or look-up-table
1251 if (centreUnit) {
1252 g_log.debug() << "For FitParameter " << parameterName(i)
1253 << " centre of peak before any unit conversion is " << centreValue << '\n';
1254 centreValue = convertValue(centreValue, centreUnit, workspace, wi);
1255 g_log.debug() << "For FitParameter " << parameterName(i)
1256 << " centre of peak after any unit conversion is " << centreValue << '\n';
1257 }
1258
1259 double paramValue = fitParam.getValue(centreValue);
1260
1261 // this returned param value by a formula or a look-up-table may
1262 // have
1263 // a unit of its own. If set convert param value
1264 // See section 'Using fitting parameters in
1265 // docs/source/concepts/InstrumentDefinitionFile.rst
1266 if (fitParam.getFormula().empty()) {
1267 // so from look up table
1268 Kernel::Unit_sptr resultUnit = fitParam.getLookUpTable().getYUnit(); // from table
1269 g_log.debug() << "The FitParameter " << parameterName(i) << " = " << paramValue
1270 << " before y-unit conversion\n";
1271 paramValue /= convertValue(1.0, resultUnit, workspace, wi);
1272 g_log.debug() << "The FitParameter " << parameterName(i) << " = " << paramValue
1273 << " after y-unit conversion\n";
1274 } else {
1275 // so from formula
1276
1277 std::string resultUnitStr = fitParam.getResultUnit();
1278
1279 if (!resultUnitStr.empty()) {
1280 std::vector<std::string> allUnitStr = Kernel::UnitFactory::Instance().getKeys();
1281 for (auto &iUnit : allUnitStr) {
1282 size_t found = resultUnitStr.find(iUnit);
1283 if (found != std::string::npos) {
1284 size_t len = iUnit.size();
1285 std::stringstream readDouble;
1286 Kernel::Unit_sptr unt = Kernel::UnitFactory::Instance().create(iUnit);
1287 readDouble << 1.0 / convertValue(1.0, unt, workspace, wi);
1288 resultUnitStr.replace(found, len, readDouble.str());
1289 }
1290 } // end for
1291
1292 try {
1293 mu::Parser p;
1294 p.SetExpr(resultUnitStr);
1295 g_log.debug() << "The FitParameter " << parameterName(i) << " = " << paramValue
1296 << " before result-unit conversion (using " << resultUnitStr << ")\n";
1297 paramValue *= p.Eval();
1298 g_log.debug() << "The FitParameter " << parameterName(i) << " = " << paramValue
1299 << " after result-unit conversion\n";
1300 } catch (mu::Parser::exception_type &e) {
1301 g_log.error() << "Cannot convert formula unit to workspace unit"
1302 << " Formula unit which cannot be passed is " << resultUnitStr
1303 << ". Muparser error message is: " << e.GetMsg() << '\n';
1304 }
1305 } // end if
1306 } // end trying to convert result-unit from formula or y-unit for
1307 // lookuptable
1308
1309 setParameter(i, paramValue);
1310 } // end of update parameter value
1311
1312 // add tie if specified for this parameter in instrument definition
1313 // file
1314 if (!fitParam.getTie().empty()) {
1315 std::ostringstream str;
1316 str << getParameter(i);
1317 tie(parameterName(i), str.str());
1318 }
1319
1320 // add constraint if specified for this parameter in instrument
1321 // definition file
1322 if (!fitParam.getConstraint().empty()) {
1323 IConstraint *constraint = ConstraintFactory::Instance().createInitialized(this, fitParam.getConstraint());
1324 if (!fitParam.getConstraintPenaltyFactor().empty()) {
1325 try {
1326 double penalty = std::stod(fitParam.getConstraintPenaltyFactor());
1327 constraint->setPenaltyFactor(penalty);
1328 } catch (...) {
1329 g_log.warning() << "Can't set penalty factor for constraint\n";
1330 }
1331 }
1332 addConstraint(std::unique_ptr<IConstraint>(constraint));
1333 }
1334 }
1335 }
1336 }
1337 }
1338 } catch (...) {
1339 }

1350double IFunction::convertValue(double value, Kernel::Unit_sptr &outUnit,
1351 const std::shared_ptr<const MatrixWorkspace> &ws, size_t wsIndex) const {
1352 // only required if formula or look-up-table different from ws unit
1353 const auto &wsUnit = ws->getAxis(0)->unit();
1354 if (outUnit->unitID() == wsUnit->unitID())
1355 return value;
1356
1357 // first check if it is possible to do a quick conversion and convert
1358 // slight duplication to below to avoid instantiating vector unless necessary
1359 double factor(0.0), power(0.0);
1360 if (wsUnit->quickConversion(*outUnit, factor, power)) {
1361 return factor * std::pow(value, power);
1362 } else {
1363 std::vector<double> singleValue(1, value);
1364 convertValue(singleValue, outUnit, ws, wsIndex);
1365 return singleValue.front();
1366 }

1377void IFunction::convertValue(std::vector<double> &values, Kernel::Unit_sptr &outUnit,
1378 const std::shared_ptr<const MatrixWorkspace> &ws, size_t wsIndex) const {
1379 // only required if formula or look-up-table different from ws unit
1380 const auto &wsUnit = ws->getAxis(0)->unit();
1381 if (outUnit->unitID() == wsUnit->unitID())
1382 return;
1383
1384 // first check if it is possible to do a quick conversion convert
1385 double factor, power;
1386 if (wsUnit->quickConversion(*outUnit, factor, power)) {
1387 auto iend = values.end();
1388 for (auto itr = values.begin(); itr != iend; ++itr)
1389 (*itr) = factor * std::pow(*itr, power);
1390 } else {
1391 // Get l1, l2 and theta (see also RemoveBins.calculateDetectorPosition())
1392 Instrument_const_sptr instrument = ws->getInstrument();
1393 Geometry::IComponent_const_sptr sample = instrument->getSample();
1394 if (sample == nullptr) {
1395 g_log.error() << "No sample defined instrument. Cannot convert units for function\n"
1396 << "Ignore conversion.";
1397 return;
1398 }
1399 const auto &spectrumInfo = ws->spectrumInfo();
1400 double l1 = spectrumInfo.l1();
1401 // If this is a monitor then l1+l2 = source-detector distance and twoTheta=0
1402 auto emode = ws->getEMode();
1403
1404 Kernel::UnitParametersMap pmap{};
1405 spectrumInfo.getDetectorValues(*wsUnit, *outUnit, emode, false, wsIndex, pmap);
1406 try {
1407 std::vector<double> emptyVec;
1408 wsUnit->toTOF(values, emptyVec, l1, emode, pmap);
1409 outUnit->fromTOF(values, emptyVec, l1, emode, pmap);
1410 } catch (std::exception &) {
1411 throw std::runtime_error("Unable to perform unit conversion to " + outUnit->unitID());
1412 }
1413 }

1669void IFunction::sortTies(const bool checkOnly) {
1670 if (!checkOnly) {
1671 m_orderedTies.clear();
1672 }
1673
1674 std::list<TieNode> orderedTieNodes;
1675 for (size_t i = 0; i < nParams(); ++i) {
1676 auto const parTie = getTie(i);
1677 if (!parTie || ignoreTie(*parTie)) {
1678 continue;
1679 }
1680
1681 TieNode newNode;
1682 newNode.left = getParameterIndex(*parTie);
1683 auto const rhsParameters = parTie->getRHSParameters();
1684 newNode.right.reserve(rhsParameters.size());
1685 for (auto &&p : rhsParameters) {
1686 newNode.right.emplace_back(this->getParameterIndex(p));
1687 }
1688 if (newNode < newNode) {
1689 throw std::runtime_error("Parameter is tied to itself: " + parTie->asString(this));
1690 }
1691 bool before(false), after(false);
1692 size_t indexBefore(0), indexAfter(0);
1693 for (auto &&node : orderedTieNodes) {
1694 if (newNode < node) {
1695 before = true;
1696 indexBefore = node.left;
1697 }
1698 if (node < newNode) {
1699 after = true;
1700 indexAfter = node.left;
1701 }
1702 }
1703 if (before) {
1704 if (after) {
1705 std::string message = "Circular dependency in ties:\n" + parTie->asString(this) + '\n';
1706 message += getTie(indexBefore)->asString(this);
1707 if (indexAfter != indexBefore) {
1708 message += '\n' + getTie(indexAfter)->asString(this);
1709 }
1710 throw std::runtime_error(message);
1711 }
1712 orderedTieNodes.push_front(newNode);
1713 } else {
1714 orderedTieNodes.emplace_back(newNode);
1715 }
1716 }
1717 if (!checkOnly) {
1718 for (auto &&node : orderedTieNodes) {
1719 auto const parTie = getTie(node.left);
1720 m_orderedTies.emplace_back(parTie);
1721 }
1722 }
1723}
