95 std::string fileType;
96 if (searchTerm == "_Definition")
97 fileType = "instrument file";
98 else if (searchTerm == "_Parameters")
99 fileType = "parameter file";
100 else
101 fileType = searchTerm + " file";
102
103 g_log.debug() << "Looking for " << fileType << " for " << instrumentName << " that is valid on '" << date << "'\n";
104 // Lookup the instrument (long) name, falling back to the provided name if not found in any facility
105 std::string instrument;
106 try {
107 instrument = Kernel::ConfigService::Instance().getInstrument(instrumentName).name();
108 } catch (const Kernel::Exception::NotFoundError &) {
109 instrument = instrumentName;
110 }
111
112 // Build the directory search list: dirHint (if any) is checked first so that
113 // parameter files co-located with the IDF in a non-standard directory (e.g.
114 // unit_testing/) are preferred over files in the standard instrument dirs.
115 const std::vector<std::string> &configDirs = Kernel::ConfigService::Instance().getInstrumentDirectories();
116 std::vector<std::string> directoryNames;
117 if (!dirHint.empty()) {
118 directoryNames.push_back(dirHint);
119 directoryNames.insert(directoryNames.end(), configDirs.begin(), configDirs.end());
120 } else {
121 directoryNames = configDirs;
122 }
123
124 // matching files sorted with newest files coming first
125 const std::vector<std::string> matchingFiles =
126 getResourceFilenames(instrument + searchTerm, fileFormats, directoryNames, date);
127 std::string foundFile;
128 if (!matchingFiles.empty()) {
129 foundFile = matchingFiles[0];
130 g_log.debug() << "The " << fileType << " selected is " << foundFile << '\n';
131 } else {

217std::string InstrumentFileFinder::lookupIPF(const std::string &dir, std::string filename) {
218 const std::string ext = ".xml";
219 // Remove .xml for example if abc.xml was passed
220 boost::algorithm::ierase_all(filename, ext);
221
222 const std::string suffixSeperator("_Definition");
223
224 std::string prefix;
225 std::string suffix;
226
227 if (auto sepPos = boost::algorithm::ifind_first(filename, suffixSeperator)) {
228 prefix = std::string(filename.begin(), sepPos.begin());
229 suffix = std::string(sepPos.end(), filename.end());
230 } else {
231 prefix = filename;
232 }
233
234 std::filesystem::path directoryPath(dir);
235
236 // Assemble parameter file name
237 std::string fullPathParamIDF = (directoryPath / (prefix + "_Parameters" + suffix + ext)).string();
238
239 if (std::filesystem::exists(fullPathParamIDF)) {
240 return fullPathParamIDF;
241 }
242
243 fullPathParamIDF = (directoryPath / (prefix + "_Parameters" + ext)).string();
244 if (std::filesystem::exists(fullPathParamIDF)) {
245 return fullPathParamIDF;
246 }
247
248 return "";
249}

266std::vector<std::string> InstrumentFileFinder::getResourceFilenames(const std::string &prefix,
267 const std::vector<std::string> &fileFormats,
268 const std::vector<std::string> &directoryNames,
269 const std::string &date) {
270
271 if (date.empty()) {
272 // Just use the current date
273 g_log.debug() << "No date specified, using current date and time.\n";
274 const std::string now = Types::Core::DateAndTime::getCurrentTime().toISO8601String();
275 // Recursively call this method, but with all parameters.
276 return InstrumentFileFinder::getResourceFilenames(prefix, fileFormats, directoryNames, now);
277 }
278
279 // Join all the file formats into a single string
280 std::stringstream ss;
281 ss << "(";
282 for (size_t i = 0; i < fileFormats.size(); ++i) {
283 if (i != 0)
284 ss << "|";
285 ss << fileFormats[i];
286 }
287 ss << ")";
288 const std::string allFileFormats = ss.str();
289
290 const boost::regex regex(prefix + ".*\\." + allFileFormats, boost::regex_constants::icase);
291
292 // Normalise date: if only YYYY-MM-DD was provided (no time component), append midnight so
293 // DateAndTime can parse it. Parameter files commonly store date-only valid-from attributes.
294 const std::string normalisedDate = getNormalisedDate(date);
295
296 DateAndTime d;
297 try {
298 d = DateAndTime(normalisedDate);
299 } catch (const std::invalid_argument &) {
300 // Some legacy data files store dates in non-ISO8601 formats.
301 // In this case fall back to the current time so we select the most recent matching file.
302 g_log.warning() << "Could not parse date '" << date
303 << "' as ISO8601; using current time for instrument file lookup.\n";
304 d = DateAndTime::getCurrentTime();
305 }
306
307 DateAndTime refDate("1899-01-01 23:59:00"); // used to help determine the most
308 // recently starting file, if none match
309 DateAndTime refDateGoodFile("1899-01-01 23:59:00"); // used to help determine the most recently

311 struct MatchingFile {
312 DateAndTime validFrom;
313 size_t directoryIndex;
314 std::string path;
315 };
316 std::vector<MatchingFile> matchingFiles;
317 bool foundFile = false;
318 std::string mostRecentFile; // path to the file with most recent "valid-from"
319 for (size_t directoryIndex = 0; directoryIndex < directoryNames.size(); ++directoryIndex) {
320 const auto &directoryName = directoryNames[directoryIndex];
321 // Iterate over the directories from user ->etc ->install, and find the
322 // first beat file
323 for (const auto &dir_entry : std::filesystem::directory_iterator(directoryName)) {
324
325 const auto &filePath = dir_entry.path();
326 if (!std::filesystem::is_regular_file(filePath))
327 continue;
328
329 const std::string l_filenamePart = filePath.filename().string();
330 if (regex_match(l_filenamePart, regex)) {
331 const std::string pathName = filePath.string();
332 g_log.debug() << "Found file: '" << pathName << "'\n";
333
334 std::string validFrom, validTo;
335 getValidFromTo(pathName, validFrom, validTo);
336 g_log.debug() << "File '" << pathName << " valid dates: from '" << validFrom << "' to '" << validTo << "'\n";
337 // Use default valid "from" and "to" dates if none were found.
338 // Normalise date-only strings (YYYY-MM-DD) to full datetimes before parsing.
339 // Some legacy instrument files store dates in non-ISO8601 formats; treat them as lowest-priority
340 // catch-alls (valid for all time) so they are still considered but ranked last.
341 DateAndTime to, from;
342 try {
343 if (validFrom.length() > 0) {
344 const std::string normFrom = getNormalisedDate(validFrom);
345 from.setFromISO8601(normFrom);
346 } else {
347 from = refDate;
348 }
349 } catch (const std::invalid_argument &) {
350 g_log.debug() << "Could not parse valid-from='" << validFrom << "' in '" << pathName
351 << "'; treating as lowest priority.\n";
352 from = refDate;
353 }
354 try {
355 if (validTo.length() > 0) {
356 const std::string normTo = getNormalisedDate(validTo);
357 to.setFromISO8601(normTo);
358 } else {
359 to.setFromISO8601("2100-01-01T00:00:00");
360 }
361 } catch (const std::invalid_argument &) {
362 g_log.debug() << "Could not parse valid-to='" << validTo << "' in '" << pathName
363 << "'; treating as lowest priority.\n";
364 to.setFromISO8601("2100-01-01T00:00:00");
365 }
366
367 if (from <= d && d <= to) {
368 foundFile = true;
369 matchingFiles.emplace_back(from, directoryIndex, pathName);
370 }
371 // Consider the most recent file in the absence of matching files
372 if (!foundFile && (from >= refDate)) {
373 refDate = from;
374 mostRecentFile = pathName;
375 }
376 }
377 }
378 }
379
380 // Retrieve the file names only
381 std::vector<std::string> pathNames;
382 if (!matchingFiles.empty()) {
383 std::sort(matchingFiles.begin(), matchingFiles.end(), [](const auto &lhs, const auto &rhs) {
384 if (lhs.validFrom != rhs.validFrom)
385 return lhs.validFrom > rhs.validFrom;
386 if (lhs.directoryIndex != rhs.directoryIndex)
387 return lhs.directoryIndex < rhs.directoryIndex;
388 return lhs.path < rhs.path;
389 });
390 pathNames.reserve(matchingFiles.size());
391
392 std::transform(matchingFiles.begin(), matchingFiles.end(), std::back_inserter(pathNames),
393 [](const auto &elem) { return elem.path; });
394 } else {
395 pathNames.emplace_back(std::move(mostRecentFile));
396 }
397
398 return pathNames;
399}
