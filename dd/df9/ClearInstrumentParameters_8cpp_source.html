45void ClearInstrumentParameters::exec() {
46 const MatrixWorkspace_sptr ws = getProperty("Workspace");
47 auto &detectorInfo = ws->mutableDetectorInfo();
48 if (detectorInfo.isScanning()) {
49 // The base instrument has a single time index, so there is no unparametrized geometry to restore each scan
50 // point to, and the scan positions are data rather than parameters.
51 throw std::runtime_error("ClearInstrumentParameters cannot be used on a workspace with a detector scan.");
52 }
53
54 auto &componentInfo = ws->mutableComponentInfo();
55 componentInfo.clearParameters();
56 detectorInfo.clearMaskFlags();
57
58 // Restore the geometry as loaded from the instrument definition. Moving or rotating a component carries its
59 // subtree with it, so parents must be restored before their children. Every parent has a higher component index
60 // than its children, so walking down from the root visits them in that order.
61 const auto &baseComponentInfo = ws->baseComponentInfo();
62 // Set unconditionally: V3D and Quat compare with a tolerance, which would leave small calibration offsets behind.
63 for (size_t index = componentInfo.size(); index-- > 0;) {
64 componentInfo.setRotation(index, baseComponentInfo.rotation(index));
65 componentInfo.setPosition(index, baseComponentInfo.position(index));
66 componentInfo.setScaleFactor(index, baseComponentInfo.scaleFactor(index));
67 }
68}
