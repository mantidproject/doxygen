450 template <typename T> IPropertyManager *doSetProperty(const std::string &name, const std::shared_ptr<T> &value) {
451 // CAREFUL: is_convertible has undefined behavior for incomplete types. If T
452 // is forward-declared in the calling code, e.g., an algorithm that calls
453 // setProperty, compilation and linking do work. However, the BEHAVIOR IS
454 // UNDEFINED and the compiler will not complain, but things crash or go
455 // wrong badly. To circumvent this we call `sizeof` here to force a compiler
456 // error if T is an incomplete type.
457 static_cast<void>(sizeof(T)); // DO NOT REMOVE, enforces complete type
458 setTypedProperty(name, value, std::is_convertible<T *, DataItem *>());
459 this->afterPropertySet(name);
460 return this;
461 }

471 template <typename T>
472 IPropertyManager *setTypedProperty(const std::string &name, const T &value, const std::false_type &) {
473 PropertyWithValue<T> *prop = dynamic_cast<PropertyWithValue<T> *>(getPointerToProperty(name));
474 if (prop) {
475 *prop = value;

477 throw std::invalid_argument("Attempt to assign to property (" + name + ") of incorrect type");
478 }
479 return this;

491 // T is convertible to DataItem_sptr
492 std::shared_ptr<DataItem> data = std::static_pointer_cast<DataItem>(value);
493 std::string error = getPointerToProperty(name)->setDataItem(data);
494 if (!error.empty()) {
