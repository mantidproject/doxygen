57void DetectorGroup::addDetector(const IDetector_const_sptr &det) {
58 // the topology of the group become undefined and needs recalculation if new
59 // detector has been added to the group
60 group_topology = undef;
61
62 // For now at least, the ID is the same as the first detector that is added
63 if (m_detectors.empty())
64 m_id = det->getID();
65
66 m_detectors.insert(DetCollection::value_type(det->getID(), det));
67}

81V3D DetectorGroup::getPos() const {
82 V3D newPos = std::accumulate(m_detectors.cbegin(), m_detectors.cend(), V3D(),
83 [](const V3D &pos, const auto &detector) { return pos + detector.second->getPos(); });
84
85 // We can have very small values (< Tolerance) of each component that should
86 // be zero
87 if (std::abs(newPos[0]) < Mantid::Kernel::Tolerance)
88 newPos[0] = 0.0;
89 if (std::abs(newPos[1]) < Mantid::Kernel::Tolerance)
90 newPos[1] = 0.0;
91 if (std::abs(newPos[2]) < Mantid::Kernel::Tolerance)
92 newPos[2] = 0.0;
93
94 return newPos /= static_cast<double>(m_detectors.size()); // protection against divide by zero in V3D
95}

110double DetectorGroup::getTwoTheta(const V3D &observer, const V3D &axis) const {
111 double result = 0.0;
112 DetCollection::const_iterator it;
113 for (it = m_detectors.begin(); it != m_detectors.end(); ++it) {
114 const V3D sampleDetVec = (*it).second->getPos() - observer;
115 result += sampleDetVec.angle(axis);
116 }
117 return result / static_cast<double>(m_detectors.size());
118}

120/*
121Gives the average angle of a group of detectors from the observation point,
122relative to the axis given.
123Returned values are signed according to the rotation direction relative to the
124axis and the instrument up direction.
125@param observer : observer (usually sample)
126@param axis : scattering axis.
127@param instrumentUp : Instrument up direction
128@return signed theta
129*/
130double DetectorGroup::getSignedTwoTheta(const Kernel::V3D &observer, const Kernel::V3D &axis,
131 const Kernel::V3D &instrumentUp) const {
132 double result = 0.0;
133 DetCollection::const_iterator it;
134 for (it = m_detectors.begin(); it != m_detectors.end(); ++it) {
135 const V3D sampleDetVec = it->second->getPos() - observer;
136 double angle = sampleDetVec.angle(axis);
137
138 V3D cross = axis.cross_prod(sampleDetVec);
139 V3D normToSurface = axis.cross_prod(instrumentUp);
140 if (normToSurface.scalar_prod(cross) < 0) {
141 angle *= -1;
142 }
143 result += angle;
144 }
145 return result / static_cast<double>(m_detectors.size());
146}

149double DetectorGroup::getPhi() const {
150 V3D avgPos = this->getPos();
151 double phi(0.0), dummy1(0.0), dummy2(0.0);
152 avgPos.getSpherical(dummy1, dummy2, phi);
153 return phi * M_PI / 180.0;

366void DetectorGroup::calculateGroupTopology() const {
367 if (m_detectors.size() == 1) {
368 group_topology = rect;
369
370 return;
371 }
372 this->groupCentre = this->getPos();
373 if (this->isValid(groupCentre)) {
374 group_topology = rect;
375 } else {
376 // the topology can still be rectangular, but randomisation errors or small
377 // gaps between
378 // detectors caused central point not to belong to detectors; need more
379 // accrurate estinations
380 // assuming that distance between detectors can not be bigger then
381 // detector's half size
382 // get detector's size:
383 IDetector_const_sptr spFirstDet = this->m_detectors.begin()->second;
384
385 BoundingBox bbox;
386 spFirstDet->getBoundingBox(bbox);
387 V3D width = bbox.width();
388
389 // loop if any near point belongs to group;
390 for (int i = 0; i < 6; i++) {
391 int ic = int(i) / 2;
392 int is = (i % 2 == 0) ? -1 : 1;
393 V3D cs = groupCentre;
394 cs[ic] += is * width[ic] / 4;
395 if (this->isValid(cs)) { // if it is, finish and end
396 group_topology = rect;
397 break;
398 }
399 }
400 // if not, consider this group to be a ring;
401 if (this->group_topology == undef) {
402 group_topology = cyl;
403 }
404 }
405}
