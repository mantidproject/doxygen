77void CubicSpline::setupInput(int const n) const {
78 // Populate data points from the input attributes and parameters
79 bool isSorted = true;
80
81 std::vector<double> x(n);
82 std::vector<double> y(n);
83
84 for (int i = 0; i < n; ++i) {
85
86 x[i] = getAttribute("x" + std::to_string(i)).asDouble();
87 y[i] = getParameter(i);
88
89 if (isSorted) {
90 // if x[i] is out of order with its neighbours
91 if (i > 1 && i < n && (x[i - 1] < x[i - 2] || x[i - 1] > x[i])) {
92 isSorted = false;
93 }
94 }
95 }
96
97 // sort the data points if necessary
98 if (!isSorted) {
99 g_log.warning() << "Spline x parameters are not in ascending order. Values "
100 "will be sorted.\n";
101
102 std::ranges::sort(std::views::zip(x, y), {}, [](const auto &xy) { return std::get<0>(xy); });
103 }
104
105 // initialize the spline
106 int status = gsl_spline_init(m_spline.get(), x.data(), y.data(), n);
107 checkGSLError(status, GSL_EINVAL);
108 m_recalculateSpline = false;
109}

141void CubicSpline::calculateSpline(double *out, const double *xValues, const size_t nData) const {
142 // calculate spline for given input set
143 bool outOfRange(false);
144 for (size_t i = 0; i < nData; ++i) {
145 if (checkXInRange(xValues[i])) {
146 // calculate the y value
147 out[i] = splineEval(xValues[i]);
148 } else {
149 // if out of range, set it to constant of fringe values
150 outOfRange = true;
151 if (xValues[i] < m_spline->interp->xmin) {
152 out[i] = splineEval(m_spline->interp->xmin);
153 } else {
154 out[i] = splineEval(m_spline->interp->xmax);
155 }
156 }
157 }
158
159 // inform user that some values weren't calculated
160 if (outOfRange) {
161 g_log.information() << "Some x values where out of range and will not be calculated.\n";
162 }
163}

188void CubicSpline::calculateDerivative(double *out, const double *xValues, const size_t nData,
189 const size_t order) const {
190 double xDeriv = 0;
191 int errorCode = 0;
192 bool outOfRange(false);
193
194 // throw error if the order is not the 1st or 2nd derivative
195 if (order < 1)
196 throw std::invalid_argument("CubicSpline: order of derivative must be 1 or greater");
197
198 for (size_t i = 0; i < nData; ++i) {
199 if (checkXInRange(xValues[i])) {
200 // choose the order of the derivative
201 if (order == 1) {
202 errorCode = gsl_spline_eval_deriv_e(m_spline.get(), xValues[i], m_acc.get(), &xDeriv);
203 } else if (order == 2) {
204 errorCode = gsl_spline_eval_deriv2_e(m_spline.get(), xValues[i], m_acc.get(), &xDeriv);
205 }
206 } else {
207 // if out of range, just set it to zero
208 outOfRange = true;
209 xDeriv = 0;
210 }
211
212 // check GSL functions didn't return an error
213 checkGSLError(errorCode, GSL_EDOM);
214
215 // record the value
216 out[i] = xDeriv;
217 }
218
219 // inform user that some values weren't calculated
220 if (outOfRange) {
221 g_log.information() << "Some x values were out of range and will not be calculated.";
222 }
223}

244void CubicSpline::setAttribute(const std::string &attName, const API::IFunction::Attribute &att) {
245
246 if (attName == "n") {
247 // get the new and old number of data points
248 int n = att.asInt();
249 int oldN = getAttribute("n").asInt();
250
251 // check that the number of data points is in a valid range
252 if (n > oldN) {
253 // get the name of the last x data point
254 std::string oldXName = "x" + std::to_string(oldN - 1);
255 double oldX = getAttribute(oldXName).asDouble();
256
257 // reallocate gsl object to new size
258 reallocGSLObjects(n);
259
260 // create blank a number of new blank parameters and attributes
261 for (int i = oldN; i < n; ++i) {
262 std::string num = std::to_string(i);
263
264 std::string newXName = "x" + num;
265 std::string newYName = "y" + num;
266
267 declareAttribute(newXName, Attribute(oldX + static_cast<double>(i - oldN + 1)));
268 declareParameter(newYName, 0);
269 }
270
271 // flag that the spline + derivatives will now need to be recalculated
272 m_recalculateSpline = true;
273 } else if (n < oldN) {
274 throw std::invalid_argument("Cubic Spline: Can't decrease the number of attributes");
275 }
276 }
277
278 storeAttributeValue(attName, att);
279}
